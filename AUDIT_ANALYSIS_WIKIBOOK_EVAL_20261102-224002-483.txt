   7: export function formatLichessEval(evalData: Pick<EngineEvaluation, "cp" | "mate"> | null): string {
   8:   if (!evalData) return "0.0";

  11:   }

  18: /**
  19:  * Converts White-perspective centipawns/mate into a [0..1] White height share
  20:  * using Lichess's sigmoid winning-chances curve: 1 / (1 + exp(-0.00368208 * cp)).
  21:  */

  23:   if (!evalData) return 50;

  25:     return evalData.mate > 0 ? 98 : 2;
  26:   }

  28:   const winChances = 1 / (1 + Math.exp(-0.00368208 * cp));
  29:   return Math.max(4, Math.min(96, winChances * 100));

  87: function computeStaticHeuristicEvaluation(fen: string, targetDepth: number): EngineEvaluation {
  88:   if (CLOUD_CACHE[fen]) {
  89:     return CLOUD_CACHE[fen];
  90:   }
  91: 
  92:   const chess = new Chess(fen);
  93:   if (chess.isCheckmate()) {
  94:     const winnerMate = chess.turn() === "w" ? -1 : 1;
  95:     return {
  96:       cp: null,
  97:       mate: winnerMate,
  98:       depth: targetDepth,
  99:       maxDepth: targetDepth,

 101:       bestMoveUci: "",
 102:       pvUci: [],
 103:       pvSan: [],
 104:       isCloud: false,
 105:     };
 106:   }
 107: 
 108:   const pieceValues: Record<string, number> = { p: 100, n: 315, b: 330, r: 500, q: 920, k: 0 };
 109:   let materialWhite = 0;
 110:   for (const row of chess.board()) {
 111:     for (const cell of row) {
 112:       if (!cell) continue;
 113:       const val = pieceValues[cell.type] ?? 0;
 114:       materialWhite += cell.color === "w" ? val : -val;
 115:     }
 116:   }
 117: 
 118:   // Evaluate 1-ply candidate moves to pick a sensible best move & PV
 119:   const legalMoves = chess.moves({ verbose: true });
 120:   if (legalMoves.length === 0) {
 121:     return {
 122:       cp: 0,
 123:       mate: null,
 124:       depth: targetDepth,
 125:       maxDepth: targetDepth,
 126:       knodes: 64,
 127:       bestMoveUci: "",
 128:       pvUci: [],
 129:       pvSan: [],
 130:       isCloud: false,
 131:     };
 132:   }
 133: 
 134:   // Prioritize center control, captures, checks
 135:   const centerSquares = new Set(["e4", "d4", "e5", "d5", "c4", "f4", "c5", "f5", "f3", "c3", "f6", "c6"]);
 136:   const scored = legalMoves.map((m) => {
 137:     let bonus = 0;
 138:     if (m.captured) bonus += (pieceValues[m.captured] ?? 100) - (pieceValues[m.piece] ?? 100) * 0.15;
 139:     if (centerSquares.has(m.to)) bonus += 25;
 140:     if (m.san.includes("+")) bonus += 35;

 175: export class LichessStockfishClient {
 176:   private worker: Worker | null = null;
 177:   private currentFen = "";
 178:   private onEval: (evaluation: EngineEvaluation) => void;
 179:   private fallbackTimer: number | null = null;
 180: 
 181:   constructor(
 182:     private workerUrl: string,
 183:     onEval: (evaluation: EngineEvaluation) => void
 184:   ) {
 185:     this.onEval = onEval;
 186:     this.initWorker();
 187:   }
 188: 
 189:   private initWorker() {
 190:     if (typeof window === "undefined" || typeof Worker === "undefined") return;
 191:     try {
 192:       this.worker = new Worker(this.workerUrl);
 193:       this.worker.onmessage = (event: MessageEvent<string>) => {
 194:         this.handleUciLine(String(event.data ?? ""));
 195:       };
 196:       this.worker.onerror = () => {
 197:         this.worker?.terminate();
 198:         this.worker = null;
 199:       };
 200:       this.worker.postMessage("uci");
 201:       this.worker.postMessage("isready");
 202:     } catch {
 203:       this.worker = null;
 204:     }
 205:   }
 206: 
 207:   private handleUciLine(line: string) {
 208:     if (!line.startsWith("info ") || !line.includes(" score ") || !line.includes(" pv ")) return;
 209:     const depthMatch = /\bdepth\s+(\d+)/.exec(line);
 210:     const cpMatch = /\bscore\s+cp\s+(-?\d+)/.exec(line);
 211:     const mateMatch = /\bscore\s+mate\s+(-?\d+)/.exec(line);
 212:     const pvMatch = /\bpv\s+([a-h1-8qrbn\s]+)$/.exec(line);
 213:     if (!depthMatch || !pvMatch) return;
 214: 
 215:     const turn = this.currentFen.split(" ")[1] === "b" ? -1 : 1;
 216:     const depth = parseInt(depthMatch[1], 10);
 217:     const rawCp = cpMatch ? parseInt(cpMatch[1], 10) * turn : null;
 218:     const rawMate = mateMatch ? parseInt(mateMatch[1], 10) * turn : null;
 219:     const pvUci = pvMatch[1].trim().split(/\s+/).filter(Boolean);
 220:     const pvSan = uciListToSanList(this.currentFen, pvUci);
 221: 
 222:     if (this.fallbackTimer) {
 223:       window.clearTimeout(this.fallbackTimer);
 224:       this.fallbackTimer = null;
 225:     }
 226: 
 227:     this.onEval({
 228:       cp: rawCp,
 229:       mate: rawMate,
 230:       depth,
 231:       maxDepth: Math.max(depth, 22),
 232:       knodes: depth * 45,
 233:       bestMoveUci: pvUci[0] ?? "",
 234:       pvUci,
 235:       pvSan,
 236:       isCloud: false,
 237:     });
 238:   }
 239: 
 240:   public analyze(fen: string, depth: number) {
 241:     this.currentFen = fen;
 242:     if (this.fallbackTimer) {
 243:       window.clearTimeout(this.fallbackTimer);
 244:       this.fallbackTimer = null;
 245:     }
 246: 
 247:     // Immediately emit cloud cache or fast evaluation so UI never lags
 248:     const immediate = computeStaticHeuristicEvaluation(fen, depth);
 249:     this.onEval(immediate);
 250: 
 251:     if (this.worker && !CLOUD_CACHE[fen]) {
 252:       try {
 253:         this.worker.postMessage("stop");
 254:         this.worker.postMessage(`position fen ${fen}`);
 255:         this.worker.postMessage(`go depth ${depth}`);
 256:       } catch {
 257:         // Fallback already emitted
 258:       }
 259:     }
 260:   }
 261: 
 262:   public stop() {
 263:     if (this.fallbackTimer) {
 264:       window.clearTimeout(this.fallbackTimer);
 265:       this.fallbackTimer = null;
 266:     }
 267:     try {
 268:       this.worker?.postMessage("stop");
 269:     } catch {
 270:       // Ignore worker errors
 271:     }
 272:   }
 273: 
 274:   public dispose() {
 275:     this.stop();
 276:     this.worker?.terminate();
 277:     this.worker = null;
 278:   }
 279: }

   5: interface MediaWikiPage {
   6:   title?: string;
   7:   missing?: boolean;
   8:   extract?: string;
   9:   fullurl?: string;
  10: }
  11: 
  12: interface MediaWikiResponse {
  13:   query?: {
  14:     pages?: MediaWikiPage[];
  15:   };
  16:   error?: {
  17:     info?: string;
  18:   };
  19: }

  21: function makeMoveSegment(san: string, ply: number): string {
  22:   const moveNumber = Math.ceil(ply / 2);
  23:   const prefix = ply % 2 === 1 ? `${moveNumber}._` : `${moveNumber}...`;
  24:   return `${prefix}${san}`;
  25: }
  26: 
  27: function makePageTitle(sanMoves: string[]): string {
  28:   return [
  29:     "Chess_Opening_Theory",
  30:     ...sanMoves.map((san, index) => makeMoveSegment(san, index + 1)),
  31:   ].join("/");
  32: }

  34: async function requestArticle(
  35:   pageTitle: string,
  36:   signal?: AbortSignal
  37: ): Promise<WikiBookArticle | null> {
  38:   const params = new URLSearchParams({
  39:     action: "query",
  40:     format: "json",
  41:     formatversion: "2",
  42:     prop: "extracts|info",
  43:     titles: pageTitle,
  44:     redirects: "1",
  45:     explaintext: "1",
  46:     exchars: "1200",
  47:     inprop: "url",
  48:     origin: "*",
  49:   });
  50: 
  51:   // Correct the MediaWiki parameter name.
  52:   params.delete("explaintext");
  53:   params.set("explaintext", "1");
  54: 
  55:   const response = await fetch(`${API_URL}?${params.toString()}`, {
  56:     signal,
  57:   });
  58: 
  59:   if (!response.ok) {
  60:     throw new Error(`Wikibooks request failed (${response.status})`);
  61:   }
  62: 
  63:   const data = (await response.json()) as MediaWikiResponse;
  64: 
  65:   if (data.error) {
  66:     throw new Error(data.error.info ?? "Wikibooks API error");
  67:   }
  68: 
  69:   const page = data.query?.pages?.[0];
  70: 
  71:   if (!page || page.missing || !page.extract?.trim()) {
  72:     return null;
  73:   }
  74: 
  75:   return {
  76:     title: page.title ?? pageTitle.replaceAll("_", " "),
  77:     extract: page.extract.trim(),
  78:     url:
  79:       page.fullurl ??
  80:       `https://en.wikibooks.org/wiki/${encodeURIComponent(

  91:   if (sanMoves.length === 0) return null;
  92: 
  93:   for (let length = sanMoves.length; length >= 1; length--) {

  43: function positionKey(fen: string): string {
  44:   // Ignore half-move and full-move counters.
  45:   return fen.split(/\s+/).slice(0, 4).join(" ");
  46: }
  47: 
  48: const openingsByPosition = new Map<string, OpeningRecord[]>();
  49: 
  50: for (const opening of OPENING_DATABASE) {
  51:   try {
  52:     const chess = new Chess();
  53:     chess.loadPgn(opening.pgn);
  54: 
  55:     const key = positionKey(chess.fen());
  56:     const existing = openingsByPosition.get(key) ?? [];
  57: 
  58:     existing.push(opening);
  59:     openingsByPosition.set(key, existing);
  60:   } catch {
  61:     // Ignore invalid records.
  62:   }
  63: }
  64: 
  65: // Prefer the shortest identifying line for duplicate positions.
  66: for (const records of openingsByPosition.values()) {
  67:   records.sort((a, b) => a.sanMoves.length - b.sanMoves.length);
  68: }

  70: export function identifyOpening(
  71:   playedMoves: string[]
  72: ): OpeningRecord | null {
  73:   try {
  74:     const chess = new Chess();
  75:     const positions: string[] = [positionKey(chess.fen())];
  76: 
  77:     for (const san of playedMoves) {
  78:       chess.move(san);
  79:       positions.push(positionKey(chess.fen()));
  80:     }
  81: 
  82:     // Search backwards: the latest named position is the deepest match.
  83:     for (let i = positions.length - 1; i >= 1; i--) {
  84:       const matches = openingsByPosition.get(positions[i]);
  85: 
  86:       if (matches?.length) {
  87:         return matches[0];
  88:       }
  89:     }
  90: 
  91:     return null;
  92:   } catch {
  93:     return null;
  94:   }
  95: }

 103:

 108:
