import React, { useEffect, useMemo, useRef, useState } from "react";
import { Chess, type Square } from "chess.js";
import { Search, Users, Crosshair, Bell } from "lucide-react";
import type {
  BoardArrowShape,
  BoardCircleShape,
  ChessVariant,
  EngineEvaluation,
  MoveNode,
  OpeningExplorerData,
  WikiBookArticle,
} from "./types";
import { WikiBookPanel } from "./WikiBookPanel";
import { ChessboardView } from "./ChessboardView";
import { EvaluationBar } from "./EvaluationBar";
import { EnginePanel } from "./EnginePanel";
import { MoveListPanel } from "./MoveListPanel";
import { OpeningExplorerModal } from "./OpeningExplorerModal";
import { BoardControlsBar } from "./BoardControlsBar";
import { fetchLichessOpeningStats } from "./openingWikiData";
import { fetchWikibooksArticle } from "./wikibooksService";
import { identifyOpening } from "./openingDatabase";
import { formatLichessEval, LichessStockfishClient } from "./stockfishWorkerClient";
import "./lichess-analysis.css";

const START_FEN = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1";


export interface LichessAnalysisBoardProps {
  showSiteHeader?: boolean;
  initialPgnMoves?: string[];
  stockfishWorkerPath?: string;
}

export const LichessAnalysisBoard: React.FC<LichessAnalysisBoardProps> = ({
  showSiteHeader = true,
  initialPgnMoves = [],
  stockfishWorkerPath = "/stockfish/stockfish-19-lite-single.js",
}) => {
  const [variant, setVariant] = useState<ChessVariant>("Standard");
  const [baseFen, setBaseFen] = useState<string>(START_FEN);
  const [moves, setMoves] = useState<MoveNode[]>(() => {
    const chess = new Chess(START_FEN);
    const list: MoveNode[] = [];
    initialPgnMoves.forEach((san, i) => {
      try {
        const res = chess.move(san);
        list.push({
          ply: i + 1,
          moveNumber: Math.floor(i / 2) + 1,
          color: res.color,
          san: res.san,
          uci: `${res.from}${res.to}${res.promotion ?? ""}`,
          from: res.from as Square,
          to: res.to as Square,
          fen: chess.fen(),
          evalFormatted: i === 0 ? "+0.2" : undefined,
          evalCp: i === 0 ? 20 : undefined,
        });
      } catch {
        // Ignore invalid initial move
      }
    });
    return list;
  });

  const [currentPly, setCurrentPly] = useState<number>(() => initialPgnMoves.length);
  const [flipped, setFlipped] = useState<boolean>(false);
  const [engineEnabled, setEngineEnabled] = useState<boolean>(true);
  const [engineDepth, setEngineDepth] = useState<number>(70);
  const [evaluation, setEvaluation] = useState<EngineEvaluation | null>(null);
  const [evaluationFen, setEvaluationFen] = useState<string>("");
  const [showThreat, setShowThreat] = useState<boolean>(false);
  const [showEngineArrow, setShowEngineArrow] = useState<boolean>(true);

  const [userArrows, setUserArrows] = useState<BoardArrowShape[]>([]);
  const [userCircles, setUserCircles] = useState<BoardCircleShape[]>([]);

  const [wikiArticle, setWikiArticle] = useState<WikiBookArticle | null>(null);
  const [wikiLoading, setWikiLoading] = useState<boolean>(false);
  const [wikiError, setWikiError] = useState<string>("");
  const [explorerOpen, setExplorerOpen] = useState<boolean>(false);
  const [explorerData, setExplorerData] = useState<OpeningExplorerData | null>(null);
  const [explorerLoading, setExplorerLoading] = useState<boolean>(false);
  const [practiceMode, setPracticeMode] = useState<boolean>(false);
  const [actionMenuOpen, setActionMenuOpen] = useState<boolean>(false);

  const [fenDraft, setFenDraft] = useState<string>(START_FEN);
  const [pgnDraft, setPgnDraft] = useState<string>("");
  const [statusToast, setStatusToast] = useState<string>("");

  const engineClientRef = useRef<LichessStockfishClient | null>(null);

  const activeFen = useMemo(() => {
    if (currentPly === 0) return baseFen;
    return moves[currentPly - 1]?.fen ?? baseFen;
  }, [baseFen, moves, currentPly]);

  const game = useMemo(() => {
    try {
      return new Chess(activeFen);
    } catch {
      return new Chess(START_FEN);
    }
  }, [activeFen]);

  const lastMove = useMemo(() => {
    if (currentPly === 0) return null;
    const node = moves[currentPly - 1];
    return node ? { from: node.from, to: node.to } : null;
  }, [moves, currentPly]);

  const activeSanHistory = useMemo(
    () => moves.slice(0, currentPly).map((m) => m.san),
    [moves, currentPly]
  );

  const selectedMoveEvaluation = currentPly > 0 ? moves[currentPly - 1] : null;

  const barEvaluation: EngineEvaluation | null = useMemo(() => {
    if (evaluationFen === activeFen && evaluation) return evaluation;

    if (selectedMoveEvaluation?.evalFormatted !== undefined) {
      return {
        cp: selectedMoveEvaluation.evalCp ?? null,
        mate: selectedMoveEvaluation.evalMate ?? null,
        depth: 0,
        maxDepth: 0,
        knodes: 0,
        bestMoveUci: "",
        pvUci: [],
        pvSan: [],
        isCloud: false,
      };
    }

    return null;
  }, [evaluationFen, activeFen, evaluation, selectedMoveEvaluation]);
  const activeHistoryKey = activeSanHistory.join("\u001f");

  useEffect(() => {
    const controller = new AbortController();
    const sanHistory = activeHistoryKey ? activeHistoryKey.split("\u001f") : [];

    setWikiError("");

    if (sanHistory.length === 0) {
      setWikiArticle(null);
      setWikiLoading(false);
      return () => controller.abort();
    }

    setWikiLoading(true);

    fetchWikibooksArticle(sanHistory, controller.signal)
      .then((article) => {
        if (!controller.signal.aborted) setWikiArticle(article);
      })
      .catch((error: unknown) => {
        if (controller.signal.aborted) return;
        setWikiError(
          error instanceof Error
            ? error.message
            : "Could not retrieve the Wikibooks article."
        );
      })
      .finally(() => {
        if (!controller.signal.aborted) setWikiLoading(false);
      });

    return () => controller.abort();
  }, [activeHistoryKey]);
  const formattedPgnString = useMemo(() => {
    if (moves.length === 0) return "";
    const parts: string[] = [];
    moves.forEach((m, idx) => {
      if (m.color === "w") {
        parts.push(`${m.moveNumber}. ${m.san}`);
      } else if (idx === 0) {
        parts.push(`${m.moveNumber}... ${m.san}`);
      } else {
        parts.push(m.san);
      }
    });
    return parts.join(" ");
  }, [moves]);

  useEffect(() => {
    setFenDraft(activeFen);
  }, [activeFen]);

  useEffect(() => {
    setPgnDraft(formattedPgnString);
  }, [formattedPgnString]);

  // Initialize Stockfish client.
  // The active move evaluation is updated by the separate effect below.
  useEffect(() => {
    const client = new LichessStockfishClient(stockfishWorkerPath, (ev) => {
      setEvaluation(ev);
    });
    engineClientRef.current = client;
    return () => client.dispose();
  }, [stockfishWorkerPath]);

  useEffect(() => {
    if (!engineEnabled) {
      engineClientRef.current?.stop();
      return;
    }
    setEvaluationFen(activeFen);
    engineClientRef.current?.analyze(activeFen, engineDepth);
  }, [engineEnabled, activeFen, engineDepth]);

  useEffect(() => {
    if (!engineEnabled || !evaluation || currentPly === 0) return;
    const formatted = formatLichessEval(evaluation);

    setMoves((prev) => {
      const index = currentPly - 1;
      const node = prev[index];

      if (!node) return prev;
      if (
        node.evalFormatted === formatted &&
        node.evalCp === evaluation.cp &&
        node.evalMate === evaluation.mate
      ) {
        return prev;
      }

      const next = [...prev];
      next[index] = {
        ...node,
        evalFormatted: formatted,
        evalCp: evaluation.cp,
        evalMate: evaluation.mate,
      };
      return next;
    });
  }, [evaluation, engineEnabled, currentPly]);

  useEffect(() => {
    if (!explorerOpen) return;

    const controller = new AbortController();
    const sanHistory = activeHistoryKey ? activeHistoryKey.split("\u001f") : [];

    setExplorerLoading(true);

    fetchLichessOpeningStats(activeFen, sanHistory, controller.signal)
      .then((res) => {
        if (controller.signal.aborted) return;
        setExplorerData(res);
        setExplorerLoading(false);
      })
      .catch(() => {
        if (!controller.signal.aborted) setExplorerLoading(false);
      });

    return () => controller.abort();
  }, [explorerOpen, activeFen, activeHistoryKey]);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      const target = e.target as HTMLElement | null;
      if (target && ["INPUT", "TEXTAREA", "SELECT"].includes(target.tagName)) return;

      if (e.key === "ArrowLeft") {
        e.preventDefault();
        setCurrentPly((p) => Math.max(0, p - 1));
      } else if (e.key === "ArrowRight") {
        e.preventDefault();
        setCurrentPly((p) => Math.min(moves.length, p + 1));
      } else if (e.key === "ArrowUp" || e.key === "Home") {
        e.preventDefault();
        setCurrentPly(0);
      } else if (e.key === "ArrowDown" || e.key === "End") {
        e.preventDefault();
        setCurrentPly(moves.length);
      } else if (e.key.toLowerCase() === "f") {
        e.preventDefault();
        setFlipped((f) => !f);
      } else if (e.key.toLowerCase() === "e") {
        e.preventDefault();
        setExplorerOpen((o) => !o);
      }
    };
    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [moves.length]);

  const handleMakeMove = (from: Square, to: Square, promotion?: "q" | "r" | "b" | "n"): boolean => {
    try {
      const nextGame = new Chess(activeFen);
      const res = nextGame.move({ from, to, ...(promotion ? { promotion } : {}) });
      const truncated = moves.slice(0, currentPly);
      const nextPly = truncated.length + 1;
      const moveNumber =
        res.color === "w"
          ? Math.floor(truncated.length / 2) + 1
          : truncated[truncated.length - 1]?.moveNumber ?? 1;

      const newNode: MoveNode = {
        ply: nextPly,
        moveNumber,
        color: res.color,
        san: res.san,
        uci: `${res.from}${res.to}${res.promotion ?? ""}`,
        from: res.from as Square,
        to: res.to as Square,
        fen: nextGame.fen(),
      };

      const updatedMoves = [...truncated, newNode];
      setMoves(updatedMoves);
      setCurrentPly(nextPly);
      setUserArrows([]);
      setUserCircles([]);

      if (practiceMode && !nextGame.isGameOver()) {
        window.setTimeout(() => {
          const replyGame = new Chess(nextGame.fen());
          const legal = replyGame.moves({ verbose: true });
          if (legal.length > 0) {
            const pick = legal[0];
            const replyRes = replyGame.move(pick);
            const replyPly = nextPly + 1;
            const replyMoveNumber =
              replyRes.color === "w" ? moveNumber + 1 : moveNumber;
            setMoves((curr) => [
              ...curr,
              {
                ply: replyPly,
                moveNumber: replyMoveNumber,
                color: replyRes.color,
                san: replyRes.san,
                uci: `${replyRes.from}${replyRes.to}${replyRes.promotion ?? ""}`,
                from: replyRes.from as Square,
                to: replyRes.to as Square,
                fen: replyGame.fen(),
              },
            ]);
            setCurrentPly(replyPly);
          }
        }, 350);
      }
      return true;
    } catch {
      return false;
    }
  };

  const handlePlayUciMove = (uci: string) => {
    if (uci.length < 4) return;
    const from = uci.slice(0, 2) as Square;
    const to = uci.slice(2, 4) as Square;
    const promo = uci[4] as "q" | "r" | "b" | "n" | undefined;
    handleMakeMove(from, to, promo);
  };

  const handleApplyFen = (fenText: string) => {
    const trimmed = fenText.trim();
    try {
      const parsed = new Chess(trimmed);
      setBaseFen(parsed.fen());
      setMoves([]);
      setCurrentPly(0);
      setUserArrows([]);
      setUserCircles([]);
      setStatusToast("Loaded custom FEN position");
      window.setTimeout(() => setStatusToast(""), 2400);
    } catch {
      setStatusToast("Invalid FEN string");
      window.setTimeout(() => setStatusToast(""), 2400);
    }
  };

  const handleApplyPgn = (pgnText: string) => {
    const trimmed = pgnText.trim();
    if (!trimmed) {
      setBaseFen(START_FEN);
      setMoves([]);
      setCurrentPly(0);
      return;
    }
    try {
      const parser = new Chess();
      parser.loadPgn(trimmed);
      const history = parser.history({ verbose: true });
      const replay = new Chess(START_FEN);
      const built: MoveNode[] = [];
      history.forEach((h, i) => {
        replay.move(h);
        built.push({
          ply: i + 1,
          moveNumber: Math.floor(i / 2) + 1,
          color: h.color,
          san: h.san,
          uci: `${h.from}${h.to}${h.promotion ?? ""}`,
          from: h.from as Square,
          to: h.to as Square,
          fen: replay.fen(),
        });
      });
      setBaseFen(START_FEN);
      setMoves(built);
      setCurrentPly(built.length);
      setStatusToast(`Loaded PGN (${built.length} plies)`);
      window.setTimeout(() => setStatusToast(""), 2400);
    } catch {
      setStatusToast("Could not parse PGN");
      window.setTimeout(() => setStatusToast(""), 2400);
    }
  };
const identifiedOpening = useMemo(
    () => identifyOpening(activeSanHistory),
    [activeSanHistory]
  );

  return (
    <div className="li-analysis-root">
      {showSiteHeader && (
        <header className="li-site-header">
          <div className="li-site-header-left">
            <a href="#analysis" className="li-brand-logo" onClick={(e) => e.preventDefault()}>
              <svg viewBox="0 0 50 50" width="28" height="28" className="li-knight-logo">
                <circle cx="25" cy="25" r="22" fill="none" stroke="#bababa" strokeWidth="3.2" />
                <path
                  d="M18 36c1-8 9-10 8-19 0 0-5 4-8 2-2-1 2-6 5-8 3-2 9-1 12 4 3 5 3 13 2 21H18z"
                  fill="#bababa"
                />
              </svg>
              <span className="li-brand-text">
                lichess<span>.org</span>
              </span>
            </a>

            <nav className="li-site-nav" aria-label="Main navigation">
              <a href="#play" onClick={(e) => e.preventDefault()}>PLAY</a>
              <a href="#puzzles" onClick={(e) => e.preventDefault()}>PUZZLES</a>
              <a href="#learn" onClick={(e) => e.preventDefault()}>LEARN</a>
              <a href="#watch" onClick={(e) => e.preventDefault()}>WATCH</a>
              <a href="#community" onClick={(e) => e.preventDefault()}>COMMUNITY</a>
              <a href="#tools" className="active" onClick={(e) => e.preventDefault()}>TOOLS</a>
              <a href="#donate" className="li-donate-link" onClick={(e) => e.preventDefault()}>
                <span className="li-wing-icon">ðŸª½</span> DONATE
              </a>
            </nav>
          </div>

          <div className="li-site-header-right">
            <button type="button" className="li-header-icon-btn" aria-label="Search">
              <Search size={18} />
            </button>
            <button type="button" className="li-header-icon-btn" aria-label="Friends">
              <Users size={18} />
            </button>
            <button type="button" className="li-header-icon-btn" aria-label="Challenges">
              <Crosshair size={18} />
            </button>
            <button type="button" className="li-header-icon-btn" aria-label="Notifications">
              <Bell size={18} />
            </button>
            <span className="li-user-tag">Vinayayayay</span>
            <span className="li-user-avatar" aria-hidden="true">â—</span>
          </div>
        </header>
      )}

      <main className="li-analyse-grid">
        <WikiBookPanel
          variant={variant}
          onVariantChange={setVariant}
          wikiArticle={wikiArticle}
          wikiLoading={wikiLoading}
          wikiError={wikiError}
          identifiedOpening={identifiedOpening}
          hasPlayedMoves={activeSanHistory.length > 0}
          onSelectOpening={(opening) => handleApplyPgn(opening.pgn)}
        />

        <section className="li-center-column" aria-label="Chessboard and FEN/PGN">
          <ChessboardView
            game={game}
            flipped={flipped}
            lastMove={lastMove}
            engineBestMoveUci={engineEnabled ? evaluation?.bestMoveUci ?? null : null}
            threatMoveUci={engineEnabled ? evaluation?.threatMoveUci ?? null : null}
            showEngineArrow={engineEnabled && showEngineArrow}
            showThreatArrow={engineEnabled && showThreat}
            userArrows={userArrows}
            userCircles={userCircles}
            onMakeMove={handleMakeMove}
            onToggleUserArrow={(arrow) =>
              setUserArrows((prev) => {
                const exists = prev.some(
                  (a) => a.from === arrow.from && a.to === arrow.to && a.brush === arrow.brush
                );
                return exists
                  ? prev.filter(
                      (a) => !(a.from === arrow.from && a.to === arrow.to && a.brush === arrow.brush)
                    )
                  : [...prev, arrow];
              })
            }
            onToggleUserCircle={(circle) =>
              setUserCircles((prev) => {
                const exists = prev.some(
                  (c) => c.square === circle.square && c.brush === circle.brush
                );
                return exists
                  ? prev.filter((c) => !(c.square === circle.square && c.brush === circle.brush))
                  : [...prev, circle];
              })
            }
            onClearUserShapes={() => {
              setUserArrows([]);
              setUserCircles([]);
            }}
          />

          <div className="li-underboard">
            <div className="li-fen-pgn-row">
              <label htmlFor="li-fen-input" className="li-fen-pgn-label">
                FEN
              </label>
              <input
                id="li-fen-input"
                type="text"
                className="li-fen-input"
                value={fenDraft}
                onChange={(e) => setFenDraft(e.target.value)}
                onKeyDown={(e) => {
                  if (e.key === "Enter") handleApplyFen(fenDraft);
                }}
                onBlur={() => {
                  if (fenDraft.trim() !== activeFen) handleApplyFen(fenDraft);
                }}
                spellCheck={false}
              />
            </div>

            <div className="li-fen-pgn-row li-pgn-row">
              <label htmlFor="li-pgn-input" className="li-fen-pgn-label">
                PGN
              </label>
              <textarea
                id="li-pgn-input"
                className="li-pgn-textarea"
                rows={2}
                value={pgnDraft}
                onChange={(e) => setPgnDraft(e.target.value)}
                onKeyDown={(e) => {
                  if (e.key === "Enter" && (e.ctrlKey || e.metaKey)) {
                    handleApplyPgn(pgnDraft);
                  }
                }}
                onBlur={() => {
                  if (pgnDraft.trim() !== formattedPgnString) {
                    handleApplyPgn(pgnDraft);
                  }
                }}
                spellCheck={false}
              />
            </div>
          </div>
        </section>

        <EvaluationBar
          evaluation={barEvaluation}
          enabled={engineEnabled}
          flipped={flipped}
        />

        <aside className="li-right-column" aria-label="Engine evaluation and move tree">
          <div className="li-tools-card">
            <EnginePanel
              enabled={engineEnabled}
              onToggleEnabled={() => setEngineEnabled((e) => !e)}
              evaluation={evaluation}
              depth={engineDepth}
              onChangeDepth={setEngineDepth}
              showThreat={showThreat}
              onToggleThreat={() => setShowThreat((t) => !t)}
              currentTurn={game.turn()}
              currentMoveNumber={moves[currentPly - 1]?.moveNumber ?? 1}
              onPlayPvFirstMove={handlePlayUciMove}
            />



            <MoveListPanel
              moves={moves}
              currentPly={currentPly}
              onSelectPly={setCurrentPly}
            />
            {explorerOpen && (
              <OpeningExplorerModal
                data={explorerData}
                loading={explorerLoading}
                recognisedOpening={identifiedOpening}
                onPlayUciMove={handlePlayUciMove}
              />
            )}

            {actionMenuOpen && (
              <div className="li-action-menu-drawer">
                <button
                  type="button"
                  onClick={() => {
                    setFlipped((f) => !f);
                    setActionMenuOpen(false);
                  }}
                >
                  Flip board (f)
                </button>
                <button
                  type="button"
                  onClick={() => {
                    setShowEngineArrow((a) => !a);
                    setActionMenuOpen(false);
                  }}
                >
                  {showEngineArrow ? "Hide best move arrow" : "Show best move arrow"}
                </button>
                <button
                  type="button"
                  onClick={() => {
                    setBaseFen(START_FEN);
                    setMoves([]);
                    setCurrentPly(0);
                    setActionMenuOpen(false);
                  }}
                >
                  Reset to starting position
                </button>
                <button
                  type="button"
                  onClick={() => {
                    void navigator.clipboard?.writeText(activeFen);
                    setStatusToast("FEN copied to clipboard");
                    window.setTimeout(() => setStatusToast(""), 2000);
                    setActionMenuOpen(false);
                  }}
                >
                  Copy FEN
                </button>
                <button
                  type="button"
                  onClick={() => {
                    void navigator.clipboard?.writeText(formattedPgnString);
                    setStatusToast("PGN copied to clipboard");
                    window.setTimeout(() => setStatusToast(""), 2000);
                    setActionMenuOpen(false);
                  }}
                >
                  Copy PGN
                </button>
              </div>
            )}
          </div>

          <BoardControlsBar
            explorerOpen={explorerOpen}
            onToggleExplorer={() => setExplorerOpen((o) => !o)}
            practiceMode={practiceMode}
            onTogglePractice={() => setPracticeMode((p) => !p)}
            canGoBack={currentPly > 0}
            canGoForward={currentPly < moves.length}
            onFirst={() => setCurrentPly(0)}
            onPrev={() => setCurrentPly((p) => Math.max(0, p - 1))}
            onNext={() => setCurrentPly((p) => Math.min(moves.length, p + 1))}
            onLast={() => setCurrentPly(moves.length)}
            menuOpen={actionMenuOpen}
            onToggleMenu={() => setActionMenuOpen((m) => !m)}
          />
        </aside>
      </main>

      {statusToast && (
        <div className="li-toast" role="status">
          {statusToast}
        </div>
      )}
    </div>
  );
};

export default LichessAnalysisBoard;

import React from "react";
import type { EngineEvaluation } from "./types";
import { computeWhiteBarPercentage, formatLichessEval } from "./stockfishWorkerClient";

interface EvaluationBarProps {
  evaluation: EngineEvaluation | null;
  enabled: boolean;
  flipped: boolean;
}

export const EvaluationBar: React.FC<EvaluationBarProps> = ({
  evaluation,
  enabled,
  flipped,
}) => {
  const whitePct = enabled ? computeWhiteBarPercentage(evaluation) : 50;
  const blackPct = 100 - whitePct;
  const label = enabled ? formatLichessEval(evaluation) : "0.0";

  return (
    <div
      className={`li-eval-gauge ${flipped ? "li-eval-flipped" : ""}`}
      title={`Evaluation: ${label}`}
      aria-label={`Evaluation bar ${label}`}
    >
      <div
        className="li-eval-black-bar"
        style={{ height: `${blackPct}%` }}
      />
      <div
        className="li-eval-white-bar"
        style={{ height: `${whitePct}%` }}
      />
      {/* Tick marks every 12.5% and orange 50% center marker matching Lichess */}
      <div className="li-eval-tick" style={{ top: "12.5%" }} />
      <div className="li-eval-tick" style={{ top: "25%" }} />
      <div className="li-eval-tick" style={{ top: "37.5%" }} />
      <div className="li-eval-zero-marker" />
      <div className="li-eval-tick" style={{ top: "62.5%" }} />
      <div className="li-eval-tick" style={{ top: "75%" }} />
      <div className="li-eval-tick" style={{ top: "87.5%" }} />
    </div>
  );
};

import React, { useState } from "react";
import { Target, Settings, Plus } from "lucide-react";
import type { EngineEvaluation } from "./types";
import { formatLichessEval } from "./stockfishWorkerClient";
import { toFigurativeSan } from "./CburnettPieces";

interface EnginePanelProps {
  enabled: boolean;
  onToggleEnabled: () => void;
  evaluation: EngineEvaluation | null;
  depth: number;
  onChangeDepth: (d: number) => void;
  showThreat: boolean;
  onToggleThreat: () => void;
  currentTurn: "w" | "b";
  currentMoveNumber: number;
  onPlayPvFirstMove?: (uci: string) => void;
}

export const EnginePanel: React.FC<EnginePanelProps> = ({
  enabled,
  onToggleEnabled,
  evaluation,
  depth,
  onChangeDepth,
  showThreat,
  onToggleThreat,
  currentTurn,
  currentMoveNumber,
  onPlayPvFirstMove,
}) => {
  const [settingsOpen, setSettingsOpen] = useState(false);
  const [pvExpanded, setPvExpanded] = useState(false);
  const [multiPvCount, setMultiPvCount] = useState(1);

  const evalFormatted = enabled ? formatLichessEval(evaluation) : "—";

  // Build formatted PV string matching `1... e5 2. ♘f3 ♘c6 3. ♗b5 ♘f6 4. O-O ♘xe4 5. ♖e1 ♘d6 6...`
  const formattedPvTokens = React.useMemo(() => {
    if (!enabled || !evaluation || evaluation.pvSan.length === 0) return [];
    const tokens: { label: string; uci?: string }[] = [];
    let moveNum = currentTurn === "b" ? Math.max(1, currentMoveNumber) : currentMoveNumber + 1;
    let side = currentTurn;

    evaluation.pvSan.forEach((san, idx) => {
      const fig = toFigurativeSan(san);
      const uci = evaluation.pvUci[idx];
      if (idx === 0 && side === "b") {
        tokens.push({ label: `${moveNum}... ${fig}`, uci });
        side = "w";
        moveNum++;
      } else if (side === "w") {
        tokens.push({ label: `${moveNum}. ${fig}`, uci });
        side = "b";
      } else {
        tokens.push({ label: fig, uci });
        side = "w";
        moveNum++;
      }
    });
    return tokens;
  }, [enabled, evaluation, currentTurn, currentMoveNumber]);

  return (
    <div className="li-ceval-section">
      {/* Top thin green engine progress bar */}
      <div className="li-ceval-progress-track">
        <div
          className="li-ceval-progress-bar"
          style={{ width: enabled ? `${Math.min(100, ((evaluation?.depth ?? depth) / 70) * 100)}%` : "0%" }}
        />
      </div>

      {/* Main Engine Header Strip */}
      <div className="li-ceval-header">
        {/* Lichess Green Toggle Switch with Checkmark */}
        <button
          type="button"
          role="switch"
          aria-checked={enabled}
          aria-label="Toggle Stockfish evaluation"
          className={`li-ceval-switch ${enabled ? "on" : "off"}`}
          onClick={onToggleEnabled}
        >
          <span className="li-ceval-switch-knob">
            {enabled ? "✓" : ""}
          </span>
        </button>

        {/* Big Score Readout (+0.2) */}
        <output className="li-ceval-score">{evalFormatted}</output>

        {/* Engine Metadata Block (SF 19 1MB NNUE / Depth 70 CLOUD) */}
        <div className="li-ceval-meta">
          <div className="li-ceval-meta-top">
            <span className="li-sf-title">SF 19 1MB</span>
            <span className="li-nnue-badge">NNUE</span>
          </div>
          <div className="li-ceval-meta-bottom">
            <button
              type="button"
              className="li-depth-plus-btn"
              title="Increase search depth"
              onClick={() => onChangeDepth(Math.min(99, depth + 4))}
            >
              <Plus size={11} strokeWidth={3} />
            </button>
            <span className="li-depth-text">
              Depth {enabled ? evaluation?.depth ?? depth : 0}
            </span>
            {enabled && evaluation?.isCloud && (
              <span className="li-cloud-badge">CLOUD</span>
            )}
          </div>
        </div>

        {/* Right Controls: Threat Target & Settings Gear */}
        <div className="li-ceval-actions">
          <button
            type="button"
            className={`li-ceval-icon-btn ${showThreat ? "active" : ""}`}
            title="Show threat (x)"
            aria-label="Show threat"
            onClick={onToggleThreat}
          >
            <Target size={17} />
          </button>
          <button
            type="button"
            className={`li-ceval-icon-btn ${settingsOpen ? "active" : ""}`}
            title="Engine settings"
            aria-label="Engine settings"
            onClick={() => setSettingsOpen((s) => !s)}
          >
            <Settings size={17} />
          </button>
        </div>
      </div>

      {/* Collapsible Engine Settings Drawer */}
      {settingsOpen && (
        <div className="li-ceval-settings-drawer">
          <label className="li-ceval-setting-row">
            <span>Search depth ({depth})</span>
            <input
              type="range"
              min={10}
              max={70}
              value={depth}
              onChange={(e) => onChangeDepth(Number(e.target.value))}
            />
          </label>
          <label className="li-ceval-setting-row">
            <span>Multiple lines ({multiPvCount})</span>

              max={3}
              value={multiPvCount}

      )}

        {enabled && formattedPvTokens.length > 0 ? (

            <div

                  onPlayPvFirstMove(evaluation.bestMoveUci);
                }

import React from "react";
import type { MoveNode } from "./types";
import { toFigurativeSan } from "./CburnettPieces";

interface MoveListPanelProps {
  moves: MoveNode[];
  currentPly: number;
  onSelectPly: (ply: number) => void;
}

export const MoveListPanel: React.FC<MoveListPanelProps> = ({
  moves,
  currentPly,
  onSelectPly,
}) => {
  // Group moves into pairs (White + Black per full move number)
  const movePairs = React.useMemo(() => {
    const rows: { moveNumber: number; white?: MoveNode; black?: MoveNode }[] = [];
    for (const node of moves) {
      let row = rows[rows.length - 1];
      if (!row || row.moveNumber !== node.moveNumber) {
        row = { moveNumber: node.moveNumber };
        rows.push(row);
      }
      if (node.color === "w") row.white = node;
      else row.black = node;
    }
    return rows;
  }, [moves]);

  return (
    <div className="li-moves-container" role="region" aria-label="Move list">
      {movePairs.map((pair) => (
        <div key={pair.moveNumber} className="li-move-row">
          <div className="li-move-index">{pair.moveNumber}</div>

          {pair.white ? (
            <button
              type="button"
              className={`li-move-cell ${currentPly === pair.white.ply ? "active" : ""}`}
              onClick={() => onSelectPly(pair.white!.ply)}
            >
              <span className="li-move-san">{toFigurativeSan(pair.white.san)}</span>
              {pair.white.evalFormatted && (
                <span className="li-move-eval">{pair.white.evalFormatted}</span>
              )}
            </button>
          ) : (
            <div className="li-move-cell li-move-empty">...</div>
          )}

          {pair.black ? (
            <button
              type="button"
              className={`li-move-cell ${currentPly === pair.black.ply ? "active" : ""}`}
              onClick={() => onSelectPly(pair.black!.ply)}
            >
              <span className="li-move-san">{toFigurativeSan(pair.black.san)}</span>
              {pair.black.evalFormatted && (
                <span className="li-move-eval">{pair.black.evalFormatted}</span>
              )}
            </button>
          ) : (
            <div className="li-move-cell li-move-empty" />
          )}
        </div>
      ))}
    </div>
  );

import { Chess } from "chess.js";
import rawOpenings from "./openings.json";

export interface OpeningRecord {
  eco: string;
  name: string;
  pgn: string;
  sanMoves: string[];
}

interface RawOpening {
  eco: string;
  name: string;
  pgn: string;
}

const parsedOpenings: OpeningRecord[] = [];

for (const record of rawOpenings as RawOpening[]) {
  try {
    const chess = new Chess();
    chess.loadPgn(record.pgn);

    const sanMoves = chess.history();

    if (sanMoves.length === 0) continue;

    parsedOpenings.push({
      eco: record.eco,
      name: record.name,
      pgn: record.pgn,
      sanMoves,
    });
  } catch {
    // Ignore malformed database entries.
  }
}

export const OPENING_DATABASE: OpeningRecord[] = parsedOpenings;

export type EcoGroup = "ALL" | "A" | "B" | "C" | "D" | "E";

function positionKey(fen: string): string {
  // Ignore half-move and full-move counters.
  return fen.split(/\s+/).slice(0, 4).join(" ");
}

const openingsByPosition = new Map<string, OpeningRecord[]>();

for (const opening of OPENING_DATABASE) {
  try {
    const chess = new Chess();
    chess.loadPgn(opening.pgn);

    const key = positionKey(chess.fen());
    const existing = openingsByPosition.get(key) ?? [];

    existing.push(opening);
    openingsByPosition.set(key, existing);
  } catch {
    // Ignore invalid records.
  }
}

// Prefer the shortest identifying line for duplicate positions.
for (const records of openingsByPosition.values()) {
  records.sort((a, b) => a.sanMoves.length - b.sanMoves.length);
}

export function identifyOpening(
  playedMoves: string[]
): OpeningRecord | null {
  try {
    const chess = new Chess();
    const positions: string[] = [positionKey(chess.fen())];

    for (const san of playedMoves) {
      chess.move(san);
      positions.push(positionKey(chess.fen()));
    }

    // Only recognise the current board position.
    // Do not keep displaying an old opening after the played line diverges.
    const currentPosition = positions[positions.length - 1];
    const matches = openingsByPosition.get(currentPosition);

    return matches?.[0] ?? null;
  } catch {
    return null;
  }
}

export function searchOpenings(
  query: string,
  group: EcoGroup = "ALL",
  limit = 40
): OpeningRecord[] {
  const normalizedQuery = query.trim().toLowerCase();

  return OPENING_DATABASE
    .filter((opening) => {
      const groupMatches =
        group === "ALL" || opening.eco.startsWith(group);

      if (!groupMatches) return false;
      if (!normalizedQuery) return true;

      return (
        opening.name.toLowerCase().includes(normalizedQuery) ||
        opening.eco.toLowerCase().includes(normalizedQuery) ||
        opening.pgn.toLowerCase().includes(normalizedQuery)
      );
    })
    .sort(
      (a, b) =>
        a.eco.localeCompare(b.eco) ||
        a.name.localeCompare(b.name)
    )
    .slice(0, limit);
}

import { Chess, type Square } from "chess.js";
import type { EngineEvaluation } from "./types";

/**
 * Formats an engine evaluation into Lichess's "+0.2" / "-1.4" / "#3" display string.
 */
export function formatLichessEval(evalData: Pick<EngineEvaluation, "cp" | "mate"> | null): string {
  if (!evalData) return "0.0";
  if (evalData.mate !== null) {
    return `#${evalData.mate}`;
  }
  if (evalData.cp === null) return "0.0";
  const pawns = evalData.cp / 100;
  const sign = pawns > 0 ? "+" : "";
  return `${sign}${pawns.toFixed(1)}`;
}

/**
 * Converts White-perspective centipawns/mate into a [0..1] White height share
 * using Lichess's sigmoid winning-chances curve: 1 / (1 + exp(-0.00368208 * cp)).
 */
export function computeWhiteBarPercentage(evalData: Pick<EngineEvaluation, "cp" | "mate"> | null): number {
  if (!evalData) return 50;
  if (evalData.mate !== null) {
    return evalData.mate > 0 ? 98 : 2;
  }
  const cp = Math.max(-1000, Math.min(1000, evalData.cp ?? 0));
  const winChances = 1 / (1 + Math.exp(-0.00368208 * cp));
  return Math.max(4, Math.min(96, winChances * 100));
}

function uciListToSanList(fen: string, uciMoves: string[]): string[] {
  const chess = new Chess(fen);
  const out: string[] = [];
  for (const uci of uciMoves) {
    const m = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
    if (!m) break;
    try {
      const moveObj = chess.move({
        from: m[1] as Square,
        to: m[2] as Square,
        ...(m[3] ? { promotion: m[3] as "q" | "r" | "b" | "n" } : {}),
      });
      out.push(moveObj.san);
    } catch {
      break;
    }
  }
  return out;
}

 * 2) If a real Stockfish Web Worker is present at `/stockfish/stockfish-19-lite-single.js`,
 *    it streams live UCI `info depth ... score cp ... pv ...` updates from the worker.
 */

    pvSan: ["e5", "Nf3", "Nc6", "Bb5", "Nf6", "O-O", "Nxe4", "Re1", "Nd6", "Nxe5"],

function computeStaticHeuristicEvaluation(fen: string, targetDepth: number): EngineEvaluation {
  if (CLOUD_CACHE[fen]) {
    return CLOUD_CACHE[fen];
  }

  const chess = new Chess(fen);
  if (chess.isCheckmate()) {
    const winnerMate = chess.turn() === "w" ? -1 : 1;
    return {
      cp: null,
      mate: winnerMate,
      depth: targetDepth,
      maxDepth: targetDepth,
      knodes: 120,
      bestMoveUci: "",
      pvUci: [],
      pvSan: [],
      isCloud: false,
    };
  }

  const pieceValues: Record<string, number> = { p: 100, n: 315, b: 330, r: 500, q: 920, k: 0 };
  let materialWhite = 0;
  for (const row of chess.board()) {
    for (const cell of row) {
      if (!cell) continue;
      const val = pieceValues[cell.type] ?? 0;
      materialWhite += cell.color === "w" ? val : -val;
    }
  }

  // Evaluate 1-ply candidate moves to pick a sensible best move & PV
  const legalMoves = chess.moves({ verbose: true });
  if (legalMoves.length === 0) {
    return {
      cp: 0,
      mate: null,
      depth: targetDepth,
      maxDepth: targetDepth,
      knodes: 64,
      bestMoveUci: "",
      pvUci: [],
      pvSan: [],
      isCloud: false,
    };
  }

  // Prioritize center control, captures, checks
  const centerSquares = new Set(["e4", "d4", "e5", "d5", "c4", "f4", "c5", "f5", "f3", "c3", "f6", "c6"]);
  const scored = legalMoves.map((m) => {
    let bonus = 0;
    if (m.captured) bonus += (pieceValues[m.captured] ?? 100) - (pieceValues[m.piece] ?? 100) * 0.15;
    if (centerSquares.has(m.to)) bonus += 25;
    if (m.san.includes("+")) bonus += 35;
    return { move: m, score: bonus };
  });
  scored.sort((a, b) => b.score - a.score);

  const best = scored[0].move;
  const pvUci: string[] = [`${best.from}${best.to}${best.promotion ?? ""}`];
  const sim = new Chess(fen);
  sim.move(best);

  for (let ply = 0; ply < 6; ply++) {
    const replies = sim.moves({ verbose: true });
    if (!replies.length) break;
    const reply = replies[Math.min(ply, replies.length - 1)];
    pvUci.push(`${reply.from}${reply.to}${reply.promotion ?? ""}`);
    sim.move(reply);
  }

  const pvSan = uciListToSanList(fen, pvUci);
  const turnSign = chess.turn() === "w" ? 1 : -1;
  const finalCp = Math.round(materialWhite + turnSign * 18);

  return {
    cp: finalCp,
    mate: null,
    depth: targetDepth,
    maxDepth: targetDepth,
    knodes: 340,
    bestMoveUci: pvUci[0] ?? "",
    pvUci,
    pvSan,
    isCloud: false,
  };
}

export class LichessStockfishClient {
  private worker: Worker | null = null;
  private currentFen = "";
  private onEval: (evaluation: EngineEvaluation) => void;
  private fallbackTimer: number | null = null;

  constructor(
    private workerUrl: string,
    onEval: (evaluation: EngineEvaluation) => void
  ) {
    this.onEval = onEval;
    this.initWorker();
  }

  private initWorker() {
    if (typeof window === "undefined" || typeof Worker === "undefined") return;
    try {
      this.worker = new Worker(this.workerUrl);
      this.worker.onmessage = (event: MessageEvent<string>) => {
        this.handleUciLine(String(event.data ?? ""));
      };
      this.worker.onerror = () => {
        this.worker?.terminate();
        this.worker = null;
      };
      this.worker.postMessage("uci");
      this.worker.postMessage("isready");
    } catch {
      this.worker = null;
    }
  }

  private handleUciLine(line: string) {
    if (!line.startsWith("info ") || !line.includes(" score ") || !line.includes(" pv ")) return;
    const depthMatch = /\bdepth\s+(\d+)/.exec(line);
    const cpMatch = /\bscore\s+cp\s+(-?\d+)/.exec(line);
    const mateMatch = /\bscore\s+mate\s+(-?\d+)/.exec(line);
    const pvMatch = /\bpv\s+([a-h1-8qrbn\s]+)$/.exec(line);
    if (!depthMatch || !pvMatch) return;

    const turn = this.currentFen.split(" ")[1] === "b" ? -1 : 1;
    const depth = parseInt(depthMatch[1], 10);
    const rawCp = cpMatch ? parseInt(cpMatch[1], 10) * turn : null;
    const rawMate = mateMatch ? parseInt(mateMatch[1], 10) * turn : null;
    const pvUci = pvMatch[1].trim().split(/\s+/).filter(Boolean);
    const pvSan = uciListToSanList(this.currentFen, pvUci);

    if (this.fallbackTimer) {
      window.clearTimeout(this.fallbackTimer);
      this.fallbackTimer = null;
    }

    this.onEval({
      cp: rawCp,
      mate: rawMate,
      depth,
      maxDepth: Math.max(depth, 22),
      knodes: depth * 45,
      bestMoveUci: pvUci[0] ?? "",
      pvUci,
      pvSan,
      isCloud: false,
    });
  }

  public analyze(fen: string, depth: number) {
    this.currentFen = fen;
    if (this.fallbackTimer) {
      window.clearTimeout(this.fallbackTimer);
      this.fallbackTimer = null;
    }

    // Immediately emit cloud cache or fast evaluation so UI never lags
    const immediate = computeStaticHeuristicEvaluation(fen, depth);
    this.onEval(immediate);

    if (this.worker && !CLOUD_CACHE[fen]) {
      try {
        this.worker.postMessage("stop");
        this.worker.postMessage(`position fen ${fen}`);
        this.worker.postMessage(`go depth ${depth}`);
      } catch {
        // Fallback already emitted
      }
    }
  }

  public stop() {
    if (this.fallbackTimer) {
      window.clearTimeout(this.fallbackTimer);
      this.fallbackTimer = null;
    }
    try {
      this.worker?.postMessage("stop");
    } catch {
      // Ignore worker errors
    }
  }

  public dispose() {
    this.stop();
    this.worker?.terminate();
    this.worker = null;
  }
}
