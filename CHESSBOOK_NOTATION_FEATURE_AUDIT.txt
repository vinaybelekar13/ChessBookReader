import { Chess } from "chess.js";

export function makePosition(baseFen: string, moves: string[], moveIndex: number): Chess {
  const game = new Chess(baseFen);
  for (const san of moves.slice(0, moveIndex)) {
    const result = game.move(san);
    if (!result) throw new Error(`Saved move could not be replayed: ${san}`);
  }
  return game;
}

export function fenWithTurn(fen: string, turn: "w" | "b"): string {
  const fields = fen.split(/\s+/);
  if (fields.length !== 6) throw new Error("FEN must contain six fields.");
  fields[1] = turn;
  fields[3] = "-";
  fields[4] = "0";
  return fields.join(" ");
}

import { Chess, type Square } from "chess.js";

export type OpeningLineResult = {
  fen: string;
  sanMoves: string[];
  uciMoves: string[];
  error: string | null;
};

export function parseOpeningLine(input: string): OpeningLineResult {
  const chess = new Chess();
  const sanMoves: string[] = [];
  const uciMoves: string[] = [];

  const tokens = input
    .replace(/\{[^}]*\}/g, " ")
    .replace(/\([^)]*\)/g, " ")
    .replace(/\d+\.(\.\.)?/g, " ")
    .replace(/\$\d+/g, " ")
    .split(/\s+/)
    .filter(Boolean)
    .filter((token) => !["1-0", "0-1", "1/2-1/2", "*"].includes(token));

  for (const token of tokens) {
    try {
      const uci = token.match(/^([a-h][1-8])([a-h][1-8])([qrbn])?$/i);
      const move = uci
        ? chess.move({
            from: uci[1].toLowerCase() as Square,
            to: uci[2].toLowerCase() as Square,
            promotion: uci[3]?.toLowerCase(),
          })
        : chess.move(token);

      sanMoves.push(move.san);
      uciMoves.push(`${move.from}${move.to}${move.promotion ?? ""}`);
    } catch {
      return {
        fen: chess.fen(),
        sanMoves,
        uciMoves,
        error: `Invalid or illegal move: ${token}`,
      };
    }
  }

  return { fen: chess.fen(), sanMoves, uciMoves, error: null };
}

export function openingPositionAt(input: string, ply: number): OpeningLineResult {
  const parsed = parseOpeningLine(input);
  if (parsed.error) return parsed;

  const limit = Math.max(0, Math.min(parsed.sanMoves.length, Math.floor(ply)));
  const chess = new Chess();
  const sanMoves: string[] = [];
  const uciMoves: string[] = [];

  for (let i = 0; i < limit; i++) {
    const move = chess.move(parsed.sanMoves[i]);
    sanMoves.push(move.san);
    uciMoves.push(`${move.from}${move.to}${move.promotion ?? ""}`);
  }

  return { fen: chess.fen(), sanMoves, uciMoves, error: null };
}

import type { BookRecord, BookSession } from "./model";

const DB_NAME = "chessbook-reader";
const DB_VERSION = 1;
const ACTIVE_KEY = "activeBookId";

function openDatabase(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, DB_VERSION);
    request.onupgradeneeded = () => {
      const db = request.result;
      if (!db.objectStoreNames.contains("books")) {
        db.createObjectStore("books", { keyPath: "id" });
      }
      if (!db.objectStoreNames.contains("sessions")) {
        db.createObjectStore("sessions", { keyPath: "bookId" });
      }
      if (!db.objectStoreNames.contains("settings")) {
        db.createObjectStore("settings", { keyPath: "key" });
      }
    };
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error ?? new Error("Could not open local library."));
  });
}

async function transact<T>(
  storeName: string,
  mode: IDBTransactionMode,
  run: (store: IDBObjectStore) => IDBRequest<T>,
): Promise<T> {
  const db = await openDatabase();
  try {
    return await new Promise<T>((resolve, reject) => {
      const transaction = db.transaction(storeName, mode);
      let outcome: { value: T } | null = null;
      transaction.oncomplete = () => {
        if (outcome) resolve(outcome.value);
        else reject(new Error("Local library transaction completed without a result."));
      };
      transaction.onerror = () => reject(transaction.error ?? new Error("Local library transaction failed."));
      transaction.onabort = () => reject(transaction.error ?? new Error("Local library transaction aborted."));
      const request = run(transaction.objectStore(storeName));
      request.onsuccess = () => { outcome = { value: request.result }; };
      request.onerror = () => reject(request.error ?? new Error("Local library operation failed."));
    });
  } finally {
    db.close();
  }
}

  const item = await transact<{ key: string; value: string } | undefined>(

    "readonly",
