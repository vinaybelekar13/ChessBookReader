  772:   async function openTransfer(candidate: Candidate) {
  773:     const canvas = canvasRef.current;
  774:     if (!canvas || canvas.width === 0 || canvas.height === 0) {
  775:       setError("The rendered page image is unavailable. Rerender the page before using this crop.");
  776:       return;
  777:     }
  778:     let preview: HTMLCanvasElement;
  779:     try {
  780:       const crop = cropToPixels(candidate, canvas.width, canvas.height);
  781:       preview = document.createElement("canvas");
  782:       preview.width = crop.width;
  783:       preview.height = crop.height;
  784:       const context = preview.getContext("2d");
  785:       if (!context) throw new Error("Could not prepare a crop preview in this browser.");
  786:       context.drawImage(canvas, crop.x, crop.y, crop.width, crop.height, 0, 0, crop.width, crop.height);
  787:     } catch (cause) {
  788:       setError(`Could not create the crop preview: ${messageOf(cause)}`);
  789:       return;
  790:     }
  791:     setTransferCandidate(candidate);
  792:     setTransferPreview(preview.toDataURL("image/png"));
  793:     setTransferFen("");
  794:     setRecognitionResult(null);
  795:         setRecognitionError("");
  796:         const requestId = ++transferRequestRef.current;
  797:     setRecognitionStatus("loading");
  798:     try {
  799:       const image = await new Promise<Blob>((resolve, reject) => {
  800:         preview.toBlob((blob) => blob ? resolve(blob) : reject(new Error("Could not encode the selected crop.")), "image/png");
  801:       });
  802:       const form = new FormData();
  803:       form.append("image", image, `board-crop-${candidate.id}.png`);
  804:       const response = await fetch(`${API_BASE}/recognize`, { method: "POST", body: form });
  805:       const result = await response.json() as RecognitionResult & { detail?: string };
  806:       if (!response.ok) throw new Error(result.detail ?? `Recognition service returned HTTP ${response.status}.`);
  807:       if (requestId !== transferRequestRef.current) return;
  808:       setTransferFen(result.fen);
  809:       setRecognitionResult(result);
  810:       setRecognitionStatus("complete");
  811:       // A successful scan immediately becomes the active analysis position.
  812:       // Keep the review dialog open so the user can correct uncertain squares;
  813:       // the explicit confirmation button remains available for corrected FENs.
  814:       if (moveToPosition(result.fen)) {
  815:         setNotice("Scanned position loaded automatically onto the analysis board. Review the diagram and correct any uncertain squares before continuing.");
  816:       }
  817:     } catch (cause) {
  818:       if (requestId !== transferRequestRef.current) return;
  819:       setRecognitionStatus("failed");
  820:       setRecognitionError(messageOf(cause));
  821:     }
  822:   }
