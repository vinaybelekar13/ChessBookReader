export function lookupWikiBook(sanMoves: string[]): WikiBookEntry {
  for (let len = sanMoves.length; len >= 0; len--) {
    const key = sanMoves.slice(0, len).join(" ");
    if (WIKIBOOK_BY_LINE[key]) {
      return WIKIBOOK_BY_LINE[key];
    }
  }
  const moveListStr = sanMoves
    .map((m, i) => (i % 2 === 0 ? `${Math.floor(i / 2) + 1}. ${m}` : m))
    .join(" ");
  return {
    eco: "B00",
    title: `${moveListStr || "Custom Position"} · Opening Theory`,
    paragraphs: [
      "The position has left the primary introductory WikiBook overview or entered a specific middlegame continuation.",
      "Use the Stockfish 19 NNUE engine lines on the right or toggle the Opening Explorer (book icon in the bottom toolbar) to inspect master game frequencies and continuations.",
    ],
  };
}

export async function fetchLichessOpeningStats(
  fen: string,
  sanMoves: string[],
  signal?: AbortSignal
): Promise<OpeningExplorerData> {
  try {
    const url = `https://explorer.lichess.ovh/masters?fen=${encodeURIComponent(fen)}&moves=8`;
    const res = await fetch(url, { signal });
    if (res.ok) {
      const data = await res.json();
      return {
        opening: data.opening ?? null,
        white: data.white ?? 0,
        draws: data.draws ?? 0,
        black: data.black ?? 0,
        moves: (data.moves ?? []).map((m: Record<string, unknown>) => ({
          uci: String(m.uci ?? ""),
          san: String(m.san ?? ""),
          white: Number(m.white ?? 0),
          draws: Number(m.draws ?? 0),
          black: Number(m.black ?? 0),
          averageRating: Number(m.averageRating ?? 2450),
        })),
        source: "lichess-api",
      };
    }
  } catch {
    // Fallback to built-in master book when offline or rate-limited
  }

  const key = sanMoves.join(" ");
  return (
    BUILTIN_EXPLORER_BY_LINE[key] ?? {
      opening: {
        eco: lookupWikiBook(sanMoves).eco,
        name: lookupWikiBook(sanMoves).title.split("·").pop()?.trim() ?? "Custom Line",
      },
      white: 1240,
      draws: 1580,
      black: 980,
      source: "builtin-master-book",
      moves: [],
    }
  );
}

import { Chess } from "chess.js";
import rawOpenings from "./openings.json";

export interface OpeningRecord {
  eco: string;
  name: string;
  pgn: string;
  sanMoves: string[];
}

interface RawOpening {
  eco: string;
  name: string;
  pgn: string;
}

const parsedOpenings: OpeningRecord[] = [];

for (const record of rawOpenings as RawOpening[]) {
  try {
    const chess = new Chess();
    chess.loadPgn(record.pgn);

    const sanMoves = chess.history();

    if (sanMoves.length === 0) continue;

    parsedOpenings.push({
      eco: record.eco,
      name: record.name,
      pgn: record.pgn,
      sanMoves,
    });
  } catch {
    // Ignore malformed database entries.
  }
}

export const OPENING_DATABASE: OpeningRecord[] = parsedOpenings;

export type EcoGroup = "ALL" | "A" | "B" | "C" | "D" | "E";

function positionKey(fen: string): string {
  // Ignore half-move and full-move counters.
  return fen.split(/\s+/).slice(0, 4).join(" ");
}

const openingsByPosition = new Map<string, OpeningRecord[]>();

for (const opening of OPENING_DATABASE) {
  try {
    const chess = new Chess();
    chess.loadPgn(opening.pgn);

    const key = positionKey(chess.fen());
    const existing = openingsByPosition.get(key) ?? [];

    existing.push(opening);
    openingsByPosition.set(key, existing);
  } catch {
    // Ignore invalid records.
  }
}

// Prefer the shortest identifying line for duplicate positions.
for (const records of openingsByPosition.values()) {
  records.sort((a, b) => a.sanMoves.length - b.sanMoves.length);
}

export function identifyOpening(
  playedMoves: string[]
): OpeningRecord | null {
  try {
    const chess = new Chess();
    const positions: string[] = [positionKey(chess.fen())];

    for (const san of playedMoves) {
      chess.move(san);
      positions.push(positionKey(chess.fen()));
    }

    // Search backwards: the latest named position is the deepest match.
    for (let i = positions.length - 1; i >= 1; i--) {
      const matches = openingsByPosition.get(positions[i]);

      if (matches?.length) {
        return matches[0];
      }
    }

    return null;
  } catch {
    return null;
  }
}

export function searchOpenings(
  query: string,
  group: EcoGroup = "ALL",
  limit = 40
): OpeningRecord[] {
  const normalizedQuery = query.trim().toLowerCase();

  return OPENING_DATABASE
    .filter((opening) => {
      const groupMatches =
        group === "ALL" || opening.eco.startsWith(group);

      if (!groupMatches) return false;
      if (!normalizedQuery) return true;

      return (
        opening.name.toLowerCase().includes(normalizedQuery) ||
        opening.eco.toLowerCase().includes(normalizedQuery) ||
        opening.pgn.toLowerCase().includes(normalizedQuery)
      );
    })
    .sort(
      (a, b) =>
        a.eco.localeCompare(b.eco) ||
        a.name.localeCompare(b.name)
    )
    .slice(0, limit);
}
