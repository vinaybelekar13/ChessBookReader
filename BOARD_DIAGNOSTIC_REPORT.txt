   87: export function getArrowPath(
   88:   fromSquare: Square,
   89:   toSquare: Square,
   90:   flipped = false,
   91:   shaftWidth = 0.22,
   92:   headLength = 0.36,
   93:   headWidth = 0.50,
   94: ): string {
   95:   const from = squareCenter(fromSquare, flipped);
   96:   const to = squareCenter(toSquare, flipped);
   97:   const dx = to.x - from.x;
   98:   const dy = to.y - from.y;
   99:   const len = Math.hypot(dx, dy);
  100:   if (len < 0.2) return "";
  101: 
  102:   const ux = dx / len;
  103:   const uy = dy / len;
  104:   const nx = -uy;
  105:   const ny = ux;
  106: 
  107:   const actualHeadLen = len < 0.7 ? len * 0.48 : headLength;
  108:   const actualHeadWidth = len < 0.7 ? len * 0.65 : headWidth;
  109:   const actualShaftWidth = len < 0.7 ? len * 0.28 : shaftWidth;
  110: 
  111:   const hs = actualShaftWidth / 2;
  112:   const hw = actualHeadWidth / 2;
  113: 
  114:   const bx = to.x - actualHeadLen * ux;
  115:   const by = to.y - actualHeadLen * uy;
  116: 
  117:   const tipX = to.x;
  118:   const tipY = to.y;
  119: 
  120:   const headLeftX = bx + hw * nx;
  121:   const headLeftY = by + hw * ny;
  122:   const headRightX = bx - hw * nx;
  123:   const headRightY = by - hw * ny;
  124: 
  125:   const shaftEndLeftX = bx + hs * nx;
  126:   const shaftEndLeftY = by + hs * ny;
  127:   const shaftEndRightX = bx - hs * nx;
  128:   const shaftEndRightY = by - hs * ny;
  129: 
  130:   const shaftStartLeftX = from.x + hs * nx;
  131:   const shaftStartLeftY = from.y + hs * ny;
  132:   const shaftStartRightX = from.x - hs * nx;
  133:   const shaftStartRightY = from.y - hs * ny;
  134: 
  135:   return [
  136:     "M " + shaftStartLeftX.toFixed(4) + " " + shaftStartLeftY.toFixed(4),
  137:     "L " + shaftEndLeftX.toFixed(4) + " " + shaftEndLeftY.toFixed(4),
  138:     "L " + headLeftX.toFixed(4) + " " + headLeftY.toFixed(4),
  139:     "L " + tipX.toFixed(4) + " " + tipY.toFixed(4),
  140:     "L " + headRightX.toFixed(4) + " " + headRightY.toFixed(4),
  141:     "L " + shaftEndRightX.toFixed(4) + " " + shaftEndRightY.toFixed(4),
  142:     "L " + shaftStartRightX.toFixed(4) + " " + shaftStartRightY.toFixed(4),
  143:     "A " + hs.toFixed(4) + " " + hs.toFixed(4) + " 0 0 0 " + shaftStartLeftX.toFixed(4) + " " + shaftStartLeftY.toFixed(4),
  144:     "Z",
  145:   ].join(" ");
  146: }
