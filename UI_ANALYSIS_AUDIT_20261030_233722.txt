function formatUciMove(fen: string, uci: string): string {
  const match = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
  if (!match) return uci;
  try {
    const position = new Chess(fen);
    const move = position.move({
      from: match[1] as Square,
      to: match[2] as Square,
      ...(match[3] ? { promotion: match[3] as "q" | "r" | "b" | "n" } : {}),
    });
    return move.san;
  } catch {
    return uci;
  }
}

function formatUciPv(fen: string, pv: string[]): string {
  const position = new Chess(fen);
  const san: string[] = [];
  for (const uci of pv) {
    const match = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
    if (!match) break;
    try {
      const move = position.move({
        from: match[1] as Square,
        to: match[2] as Square,
        ...(match[3] ? { promotion: match[3] as "q" | "r" | "b" | "n" } : {}),
      });
      san.push(move.san);
    } catch {
      break;
    }
  }
  return san.join(" ");
}

function readableDate(timestamp: number): string {
  if (!timestamp) return "Not opened";
  const elapsed = Date.now() - timestamp;
  if (elapsed < 60_000) return "Just now";
  if (elapsed < 3_600_000) return `${Math.floor(elapsed / 60_000)}m ago`;
  if (elapsed < 86_400_000) return `${Math.floor(elapsed / 3_600_000)}h ago`;
  return new Date(timestamp).toLocaleDateString();
}

import type { Square } from "chess.js";

export type BoardArrow = {
  from: Square;
  to: Square;
  color?: string;
};

export type SquareHighlight = {
  square: Square;
  color?: string;
};

export const ANNOTATION_COLORS = {
  red: "#EF5350",
  yellow: "#FFB52E",
  blue: "#29B6E6",
  green: "#8BC34A",
  engine: "#f0a23b",
} as const;

export function getAnnotationColor(event: {
  shiftKey?: boolean;
  ctrlKey?: boolean;
  altKey?: boolean;
  metaKey?: boolean;
}): string {
  if (event.shiftKey) return ANNOTATION_COLORS.green;
  if (event.ctrlKey || event.metaKey) return ANNOTATION_COLORS.yellow;
  if (event.altKey) return ANNOTATION_COLORS.blue;
  return ANNOTATION_COLORS.red;
}

export function boardSquareAtPoint(
  clientX: number,
  clientY: number,
  rect: Pick<DOMRect, "left" | "top" | "width" | "height">,
  flipped = false,
): Square | null {
  if (rect.width <= 0 || rect.height <= 0) return null;
  const x = clientX - rect.left;
  const y = clientY - rect.top;
  if (x < 0 || y < 0 || x >= rect.width || y >= rect.height) return null;
  const column = Math.floor((x / rect.width) * 8);
  const row = Math.floor((y / rect.height) * 8);
  const file = flipped ? 7 - column : column;
  const rank = flipped ? row + 1 : 8 - row;
  return ("abcdefgh"[file] + rank) as Square;
}

export function squareCenter(square: Square, flipped = false): { x: number; y: number } {
  const file = square.charCodeAt(0) - 97;
  const rank = Number(square[1]) - 1;
  return {
    x: (flipped ? 7 - file : file) + 0.5,
    y: (flipped ? rank : 7 - rank) + 0.5,
  };
}

export function parseBestMove(move: string | undefined): BoardArrow | null {
  const match = move?.match(/^([a-h][1-8])([a-h][1-8])[qrbn]?$/);
  return match ? { from: match[1] as Square, to: match[2] as Square, color: ANNOTATION_COLORS.engine } : null;
}

export function toggleArrow(arrows: BoardArrow[], next: BoardArrow): BoardArrow[] {
  const existing = arrows.findIndex((arrow) => arrow.from === next.from && arrow.to === next.to);
  if (existing < 0) return [...arrows, next];
  if (next.color && arrows[existing].color !== next.color) {
    return arrows.map((arrow, index) => (index === existing ? next : arrow));
  }
  return arrows.filter((_, index) => index !== existing);
}

export function toggleSquareHighlight(
  highlights: SquareHighlight[],
  square: Square,
  color: string = ANNOTATION_COLORS.red,
): SquareHighlight[] {
  const existing = highlights.findIndex((h) => h.square === square);
  if (existing < 0) return [...highlights, { square, color }];
  if (highlights[existing].color !== color) {
    return highlights.map((h, index) => (index === existing ? { square, color } : h));
  }
  return highlights.filter((_, index) => index !== existing);
}

export function getArrowPath(
  fromSquare: Square,
  toSquare: Square,
  flipped = false,
  shaftWidth = 0.22,
  headLength = 0.36,
  headWidth = 0.50,
): string {
  const from = squareCenter(fromSquare, flipped);
  const to = squareCenter(toSquare, flipped);
  const dx = to.x - from.x;
  const dy = to.y - from.y;
  const len = Math.hypot(dx, dy);
  if (len < 0.2) return "";

  const ux = dx / len;
  const uy = dy / len;
  const nx = -uy;
  const ny = ux;

  const actualHeadLen = len < 0.7 ? len * 0.48 : headLength;
  const actualHeadWidth = len < 0.7 ? len * 0.65 : headWidth;
  const actualShaftWidth = len < 0.7 ? len * 0.28 : shaftWidth;

  const hs = actualShaftWidth / 2;
  const hw = actualHeadWidth / 2;

  const bx = to.x - actualHeadLen * ux;
  const by = to.y - actualHeadLen * uy;

  const tipX = to.x;
  const tipY = to.y;

  const headLeftX = bx + hw * nx;
  const headLeftY = by + hw * ny;
  const headRightX = bx - hw * nx;
  const headRightY = by - hw * ny;

  const shaftEndLeftX = bx + hs * nx;
  const shaftEndLeftY = by + hs * ny;
  const shaftEndRightX = bx - hs * nx;
  const shaftEndRightY = by - hs * ny;

  const shaftStartLeftX = from.x + hs * nx;
  const shaftStartLeftY = from.y + hs * ny;
  const shaftStartRightX = from.x - hs * nx;
  const shaftStartRightY = from.y - hs * ny;

  return [
    "M " + shaftStartLeftX.toFixed(4) + " " + shaftStartLeftY.toFixed(4),
    "L " + shaftEndLeftX.toFixed(4) + " " + shaftEndLeftY.toFixed(4),
    "L " + headLeftX.toFixed(4) + " " + headLeftY.toFixed(4),
    "L " + tipX.toFixed(4) + " " + tipY.toFixed(4),
    "L " + headRightX.toFixed(4) + " " + headRightY.toFixed(4),
    "L " + shaftEndRightX.toFixed(4) + " " + shaftEndRightY.toFixed(4),
    "L " + shaftStartRightX.toFixed(4) + " " + shaftStartRightY.toFixed(4),
    "A " + hs.toFixed(4) + " " + hs.toFixed(4) + " 0 0 0 " + shaftStartLeftX.toFixed(4) + " " + shaftStartLeftY.toFixed(4),
    "Z",
  ].join(" ");
}

import { Chess } from "chess.js";

export function makePosition(baseFen: string, moves: string[], moveIndex: number): Chess {
  const game = new Chess(baseFen);
  for (const san of moves.slice(0, moveIndex)) {
    const result = game.move(san);
    if (!result) throw new Error(`Saved move could not be replayed: ${san}`);
  }
  return game;
}

export function fenWithTurn(fen: string, turn: "w" | "b"): string {
  const fields = fen.split(/\s+/);
  if (fields.length !== 6) throw new Error("FEN must contain six fields.");
  fields[1] = turn;
  fields[3] = "-";
  fields[4] = "0";
  return fields.join(" ");
}

import { useEffect, useMemo, useState } from "react";
import "./coordinate-training.css";

type Orientation = "white" | "black";
type Mode = "timed" | "untimed" | "streak";
type SessionResult = { date: string; score: number; accuracy: number; duration: number };

const FILES = "abcdefgh";
const RANKS = "12345678";
const randomSquare = (avoid?: string) => {
  let next = `${FILES[Math.floor(Math.random() * 8)]}${RANKS[Math.floor(Math.random() * 8)]}`;
  while (next === avoid) next = `${FILES[Math.floor(Math.random() * 8)]}${RANKS[Math.floor(Math.random() * 8)]}`;
  return next;
};
const STORAGE_KEY = "chessbook-coordinate-training-v1";

export default function CoordinateTraining() {
  const [orientation, setOrientation] = useState<Orientation>("white");
  const [mode, setMode] = useState<Mode>("timed");
  const [duration, setDuration] = useState(60);
  const [remaining, setRemaining] = useState(60);
  const [target, setTarget] = useState(() => randomSquare());
  const [score, setScore] = useState(0);
  const [correct, setCorrect] = useState(0);
  const [wrong, setWrong] = useState(0);
  const [streak, setStreak] = useState(0);
  const [bestStreak, setBestStreak] = useState(0);
  const [startedAt, setStartedAt] = useState<number | null>(null);
  const [feedback, setFeedback] = useState<"correct" | "wrong" | "">("");
  const [sound, setSound] = useState(false);
  const [showCoordinates, setShowCoordinates] = useState(true);
  const [history, setHistory] = useState<SessionResult[]>(() => {
    try { return JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]") as SessionResult[]; } catch { return []; }
  });
  const finished = mode === "timed" && remaining <= 0;
  const attempts = correct + wrong;
  const accuracy = attempts ? Math.round((correct / attempts) * 100) : 0;
  const squares = useMemo(() => {
    const files = orientation === "white" ? [...FILES] : [...FILES].reverse();
    const ranks = orientation === "white" ? [...RANKS].reverse() : [...RANKS];
    return ranks.flatMap((rank) => files.map((file) => `${file}${rank}`));
  }, [orientation]);

  useEffect(() => {
    if (!startedAt || finished || mode !== "timed") return;
    const timer = window.setInterval(() => setRemaining((value) => Math.max(0, value - 1)), 1000);
    return () => window.clearInterval(timer);
  }, [startedAt, finished, mode]);

  useEffect(() => {
    if (finished && startedAt) {
      const result = { date: new Date().toISOString(), score, accuracy, duration: duration - remaining };
      setHistory((previous) => {
        const next = [result, ...previous].slice(0, 30);
        try { localStorage.setItem(STORAGE_KEY, JSON.stringify(next)); } catch { /* storage may be unavailable */ }
        return next;
      });
      setStartedAt(null);
    }
  }, [finished]);

  function restart(nextMode = mode, nextDuration = duration) {
    setMode(nextMode);
    setDuration(nextDuration);
    setRemaining(nextDuration);
    setTarget(randomSquare());
    setScore(0); setCorrect(0); setWrong(0); setStreak(0);
    setFeedback(""); setStartedAt(null);
  }

  function chooseSquare(square: string) {
    if (finished) return;
    if (!startedAt) setStartedAt(Date.now());
    if (square === target) {
      setCorrect((value) => value + 1);
      setScore((value) => value + 1);
      setStreak((value) => { const next = value + 1; setBestStreak((best) => Math.max(best, next)); return next; });
      setFeedback("correct");
    } else {
      setWrong((value) => value + 1);
      setStreak(0);
      setFeedback("wrong");
    }
    setTarget(randomSquare(target));
  }

  return <main className="coordinate-training-workspace">
    <header className="ct-header">
      <div><span className="ct-eyebrow">CHESSBOOK TRAINING</span><h1>Coordinate Training</h1><p>Build instant board vision. Find the square before the clock runs out.</p></div>
      <div className="ct-header-actions"><button onClick={() => restart()}>↻ Restart</button><button className="ct-primary" onClick={() => restart(mode, duration)}>New session</button></div>
    </header>
    <div className="ct-layout">
      <section className="ct-board-card">
        <div className="ct-board-top"><span>Find this square</span><strong className="ct-target">{finished ? "TIME!" : target}</strong><span className={`ct-feedback ${feedback}`}>{feedback === "correct" ? "Correct" : feedback === "wrong" ? "Try the next one" : "Select a square"}</span></div>
        <div className="ct-board" role="group" aria-label="Coordinate training chessboard">
          {squares.map((square) => {
            const file = FILES.indexOf(square[0]);
            const rank = Number(square[1]);
            const dark = (file + rank) % 2 === 0;
            return <button key={square} type="button" disabled={finished} onClick={() => chooseSquare(square)} className={`ct-square ${dark ? "dark" : "light"}`} aria-label={square}>
              {showCoordinates && ((orientation === "white" && file === 0) || (orientation === "black" && file === 7)) && <span className="ct-rank">{rank}</span>}
              {showCoordinates && ((orientation === "white" && rank === 1) || (orientation === "black" && rank === 8)) && <span className="ct-file">{square[0]}</span>}
            </button>;
          })}
        </div>
        <div className="ct-board-footer"><span>{orientation === "white" ? "White perspective" : "Black perspective"}</span><button onClick={() => setOrientation((value) => value === "white" ? "black" : "white")}>⇅ Flip board</button></div>
      </section>
      <aside className="ct-side">
        <section className="ct-card ct-stats"><div className="ct-card-heading">SESSION</div><div className="ct-timer">{mode === "untimed" ? "∞" : mode === "streak" ? "—" : `00:${String(remaining).padStart(2, "0")}`}</div><div className="ct-progress"><i style={{ width: `${mode === "timed" ? ((duration - remaining) / duration) * 100 : 0}%` }} /></div>
          <div className="ct-stat-grid"><div><strong>{score}</strong><span>Score</span></div><div><strong>{accuracy}%</strong><span>Accuracy</span></div><div><strong>{streak}</strong><span>Streak</span></div><div><strong>{bestStreak}</strong><span>Best streak</span></div></div>
          {!startedAt && <p className="ct-start-note">Your timer starts with your first answer.</p>}
          {finished && <div className="ct-finished">Session complete · {score} correct squares</div>}
        </section>
        <section className="ct-card"><div className="ct-card-heading">TRAINING SETTINGS</div><label>Mode<select value={mode} onChange={(e) => restart(e.target.value as Mode)}><option value="timed">Timed challenge</option><option value="untimed">Untimed practice</option><option value="streak">Streak challenge</option></select></label>

export type CropBounds = { x: number; y: number; width: number; height: number };
export type CropHandle = "move" | "n" | "s" | "e" | "w" | "ne" | "nw" | "se" | "sw";

export function pointOnCanvas(
  clientX: number,
  clientY: number,
  rect: Pick<DOMRect, "left" | "top" | "width" | "height">,
): { x: number; y: number } | null {
  if (rect.width <= 0 || rect.height <= 0) return null;
  return {
    x: Math.max(0, Math.min(1, (clientX - rect.left) / rect.width)),
    y: Math.max(0, Math.min(1, (clientY - rect.top) / rect.height)),
  };
}

export function editCrop(
  start: CropBounds,
  handle: CropHandle,
  deltaX: number,
  deltaY: number,
  minimum = 0.04,
): CropBounds {
  if (handle === "move") {
    const x = Math.max(0, Math.min(1 - start.width, start.x + deltaX));
    const y = Math.max(0, Math.min(1 - start.height, start.y + deltaY));
    return { ...start, x, y };
  }
  let left = start.x;
  let top = start.y;
  let right = start.x + start.width;
  let bottom = start.y + start.height;
  if (handle.includes("w")) left = Math.max(0, Math.min(right - minimum, left + deltaX));
  if (handle.includes("e")) right = Math.min(1, Math.max(left + minimum, right + deltaX));
  if (handle.includes("n")) top = Math.max(0, Math.min(bottom - minimum, top + deltaY));
  if (handle.includes("s")) bottom = Math.min(1, Math.max(top + minimum, bottom + deltaY));
  return { x: left, y: top, width: right - left, height: bottom - top };
}

export function cropToPixels(bounds: CropBounds, sourceWidth: number, sourceHeight: number) {
  const x = Math.max(0, Math.min(sourceWidth - 1, Math.floor(bounds.x * sourceWidth)));
  const y = Math.max(0, Math.min(sourceHeight - 1, Math.floor(bounds.y * sourceHeight)));
  const right = Math.max(x + 1, Math.min(sourceWidth, Math.ceil((bounds.x + bounds.width) * sourceWidth - 1e-7)));
  const bottom = Math.max(y + 1, Math.min(sourceHeight, Math.ceil((bounds.y + bounds.height) * sourceHeight - 1e-7)));
  return { x, y, width: right - x, height: bottom - y };
}

    const fields = trimmed ? trimmed.split(/\s+/) : [];

    const turn = fields[1] === "b" ? ("b" as const) : ("w" as const);
    const castling = fields[2] ?? "-";
    const ep = fields[3] ?? "-";
    const halfmove = Number(fields[4]) || 0;
    const fullmove = Number(fields[5]) || 1;
    const fen = buildFullFen(placement, turn, castling, ep, halfmove, fullmove);
    return { placement, turn, castling, ep, halfmove, fullmove, fen };
  }, [initialFen]);

  const [placement, setPlacement] = useState(parsedInitial.placement);
  const [turn, setTurn] = useState<"w" | "b">(parsedInitial.turn);
  const [castling, setCastling] = useState(parsedInitial.castling);
  const [activeTool, setActiveTool] = useState<string>("P"); // Piece symbol or "" for erase
  const [selectedSquare, setSelectedSquare] = useState<{ row: number; col: number } | null>(null);
  const [flipped, setFlipped] = useState(false);
  const [history, setHistory] = useState<string[]>([parsedInitial.fen]);
  const [historyIndex, setHistoryIndex] = useState(0);
  const [fenInputText, setFenInputText] = useState(parsedInitial.fen);
  const [validationError, setValidationError] = useState<string>("");

  // When initialFen changes externally (e.g. recognition completes or retries), reset editor
  useEffect(() => {
    setPlacement(parsedInitial.placement);
    setTurn(parsedInitial.turn);
    setCastling(parsedInitial.castling);
    setHistory([parsedInitial.fen]);
    setHistoryIndex(0);
    setFenInputText(parsedInitial.fen);
    setValidationError("");
    setSelectedSquare(null);
  }, [parsedInitial]);

  const currentBoard = useMemo(() => parseFenPlacement(placement) ?? parseFenPlacement(EMPTY_PLACEMENT)!, [placement]);
  const currentFen = useMemo(() => buildFullFen(placement, turn, castling), [placement, turn, castling]);
  const validation = useMemo(() => validatePositionFen(currentFen), [currentFen]);
  const kingCounts = useMemo(() => countKings(placement), [placement]);

  // Push new state to history
  function pushState(newPlacement: string, newTurn: "w" | "b" = turn, newCastling: string = castling) {
    const board = parseFenPlacement(newPlacement);
    const sanitizedCastling = board ? sanitizeCastlingRights(board, newCastling) : "-";
    const newFen = buildFullFen(newPlacement, newTurn, sanitizedCastling);

    setPlacement(newPlacement);
    setTurn(newTurn);
    setCastling(sanitizedCastling);
    setFenInputText(newFen);
    setValidationError("");

    if (newFen !== history[historyIndex]) {
      const nextHistory = history.slice(0, historyIndex + 1);
      nextHistory.push(newFen);
      setHistory(nextHistory);
      setHistoryIndex(nextHistory.length - 1);
    }
  }

  function handleUndo() {
    if (historyIndex > 0) {
      const prevIndex = historyIndex - 1;
      const prevFen = history[prevIndex];
      const fields = prevFen.split(/\s+/);
      setPlacement(fields[0]);
      setTurn(fields[1] === "b" ? "b" : "w");
      setCastling(fields[2] ?? "-");
      setFenInputText(prevFen);
      setHistoryIndex(prevIndex);
      setValidationError("");
    }
  }

  function handleRedo() {
    if (historyIndex < history.length - 1) {
      const nextIndex = historyIndex + 1;
      const nextFen = history[nextIndex];
      const fields = nextFen.split(/\s+/);
      setPlacement(fields[0]);
      setTurn(fields[1] === "b" ? "b" : "w");
      setCastling(fields[2] ?? "-");
      setFenInputText(nextFen);
      setHistoryIndex(nextIndex);
      setValidationError("");
    }
  }

  function handleSquareClick(row: number, col: number) {
    setSelectedSquare({ row, col });
    if (activeTool === "") {
      // Erase tool active
      const nextPlacement = setFenPiece(placement, row, col, "");
      pushState(nextPlacement);
    } else {
      // Place / Replace with active piece
      const nextPlacement = setFenPiece(placement, row, col, activeTool);
      pushState(nextPlacement);
    }
  }

  function handleToolSelect(tool: string) {
    setActiveTool(tool);
    // If a square is already selected and user picks a piece or erase, apply to that square
    if (selectedSquare) {
      const nextPlacement = setFenPiece(placement, selectedSquare.row, selectedSquare.col, tool);
      pushState(nextPlacement);
    }
  }

  function handleClearBoard() {
    pushState(EMPTY_PLACEMENT, turn, "-");
    setSelectedSquare(null);
  }

  function handleStartingPosition() {
    pushState(START_PLACEMENT, "w", "KQkq");
    setSelectedSquare(null);
  }

  function handleFlipBoard() {
    setFlipped((f) => !f);
  }

  function handleInvertPieces() {
    const nextPlacement = flipFenPlacement(placement);
    pushState(nextPlacement);
  }

  function handleTurnChange(newTurn: "w" | "b") {
    pushState(placement, newTurn, castling);
  }

  function handleFenInputChange(value: string) {
    setFenInputText(value);
    const trimmed = value.trim();
    const fields = trimmed.split(/\s+/);
    if (fields.length === 6) {
      const parsed = parseFenPlacement(fields[0]);
      if (parsed) {
        setPlacement(fields[0]);
        setTurn(fields[1] === "b" ? "b" : "w");
        setCastling(fields[2] ?? "-");
        setValidationError("");
        if (trimmed !== history[historyIndex]) {
          const nextHistory = history.slice(0, historyIndex + 1);
          nextHistory.push(trimmed);
          setHistory(nextHistory);
          setHistoryIndex(nextHistory.length - 1);
        }
        return;
      }
    }
    const val = validatePositionFen(trimmed);
    if (!val.isValid) {
      setValidationError(val.error || "Invalid FEN syntax.");
    }
  }

  // Keyboard navigation for selected square
  function handleKeyDown(event: React.KeyboardEvent) {
    if (!selectedSquare) return;
    const { row, col } = selectedSquare;
    if (event.key === "Delete" || event.key === "Backspace") {
      event.preventDefault();
      const nextPlacement = setFenPiece(placement, row, col, "");
      pushState(nextPlacement);
      return;
    }
    const key = event.key;
    if (/^[prnbqkPRNBQK]$/.test(key)) {
      event.preventDefault();
      const nextPlacement = setFenPiece(placement, row, col, key);
      pushState(nextPlacement);
      setActiveTool(key);
      return;
    }
    if (event.key === "Escape") {
      setSelectedSquare(null);
    }
  }

  function handleConfirmClick() {
    if (validationError) {
      return;
    }
    if (!validation.isValid) {
      setValidationError(validation.error || "Cannot confirm: position is not valid.");
      return;
    }
    onConfirm(currentFen);
  }

  const pieceAssetUrl = (piece: string) => {
    const color = piece === piece.toUpperCase() ? "w" : "b";
    const type = piece.toLowerCase();
    return `${import.meta.env.BASE_URL}assets/pieces/standalone/${color}${type}.svg`;
  };

  return (
    <div className="modal-backdrop" role="presentation" onKeyDown={handleKeyDown}>
      <section
        className="transfer-dialog manual-editor-modal"
        role="dialog"
        aria-modal="true"
        aria-label={title}
      >
        <header>
          <div>
            <h2>{title}</h2>
            {subtitle && <p>{subtitle}</p>}
          </div>
          <button
            type="button"
            aria-label="Close dialog"
            title="Cancel and close"
            onClick={onCancel}
          >
            ×
          </button>
        </header>

        {showPresets && (

            <span>START FROM:</span>
            {currentAnalysisFen && (

                type="button"
                className="preset-btn"
                onClick={() => {
                  const fields = currentAnalysisFen.trim().split(/\s+/);
                  if (fields.length === 6 && parseFenPlacement(fields[0])) {
                    pushState(fields[0], fields[1] === "b" ? "b" : "w", fields[2]);
                  }
                }}
              >
                Edit Current Board
              </button>
            )}
            <button type="button" className="preset-btn" onClick={handleClearBoard}>
              Empty Board
            </button>
            <button type="button" className="preset-btn" onClick={handleStartingPosition}>
              Standard Starting Board
            </button>
          </div>
        )}

        {cropPreviewUrl && (
          <div className="recognition-layout">
            <img className="crop-preview" src={cropPreviewUrl} alt="Selected chess diagram crop" />
            <div className="recognition-summary" role="status">
              {recognitionStatus === "loading" && (
                <>
                  <span className="spinner" /> Recognizing pieces…
                </>
              )}
              {recognitionStatus === "complete" && (
                <>
                  <strong>Model-recognized position · Editable below</strong>
                  <span>
                    {recognitionResult?.piece_count ?? 0} pieces recognized · confidence{" "}
                    {recognitionResult?.board_confidence?.toFixed(2) ?? "—"}
                  </span>
                  <span>
                    Orientation: {recognitionResult?.orientation ?? "normal"}. Use "Flip position 180°" if needed.
                  </span>
                  {onRetryRecognition && (
                    <button
                      type="button"
                      className="mini-button retry-btn"
                      onClick={onRetryRecognition}
                    >
                      ↻ Retry recognition
                    </button>
                  )}
                </>
              )}
              {recognitionStatus === "failed" && (
                <>
                  <strong>Automatic recognition did not complete.</strong>

            onClick={handleConfirmClick}
          >
            {confirmLabel}
          </button>
        </footer>
      </section>
    </div>
  );
}

import type { Candidate } from "./model";

export type NotationHighlightKind = "continuous" | "variation" | "prose";

export type NotationHighlight = {
  id: string;
  candidateId: string;
  kind: NotationHighlightKind;
  text: string;
  x: number;
  y: number;
  width: number;
  height: number;
};

export type PDFTextItemInput = {
  str: string;
  transform: number[];
  width: number;
  height: number;
};

const MOVE_NUMBER_REGEX = /\b\d{1,3}\s*(\.{1,3}|…)/;
const FIGURINE_REGEX = /[♔♕♖♗♘♙♚♛♜♝♞♟]/;
const SAN_MOVE_REGEX = /\b([KQRBN]?[a-h]?[1-8]?x?[a-h][1-8](?:=[QRBN])?[\+#]?|O-O(?:-O)?[\+#]?)\b/;
const FIGURINE_MOVE_REGEX = /[♔♕♖♗♘♙♚♛♜♝♞♟]x?[a-h][1-8](?:=[QRBN])?[\+#]?/;
const CHESS_SYMBOL_REGEX = /([!?]{1,2}|(\+\/?-|-\/?\+|\+\/=|=\/\+|±|∓|⩲|⩱|=|∞|⨁))/;
const CHESS_PROSE_REGEX = /\b(diagram|position|white to|black to|wins|draw|mate in|play|move|variation|defense|attack|advantage)\b/i;

export function isChessNotationText(text: string): { isNotation: boolean; kind: NotationHighlightKind } {
  const trimmed = text.trim();
  if (!trimmed || trimmed.length < 2) return { isNotation: false, kind: "continuous" };

  const hasMoveNum = MOVE_NUMBER_REGEX.test(trimmed);
  const hasSan = SAN_MOVE_REGEX.test(trimmed);
  const hasFigurine = FIGURINE_REGEX.test(trimmed) || FIGURINE_MOVE_REGEX.test(trimmed);
  const hasSymbol = CHESS_SYMBOL_REGEX.test(trimmed);
  const hasProse = CHESS_PROSE_REGEX.test(trimmed);

  if (trimmed.startsWith("(") || (trimmed.includes("(") && (hasSan || hasFigurine))) {
    return { isNotation: true, kind: "variation" };
  }

  if (hasMoveNum && (hasSan || hasFigurine)) {
    return { isNotation: true, kind: "continuous" };
  }

  if (hasSan && (hasFigurine || hasSymbol || trimmed.split(/\s+/).length >= 2)) {
    return { isNotation: true, kind: "continuous" };
  }

  if (hasFigurine && (hasSan || hasSymbol || hasMoveNum)) {
    return { isNotation: true, kind: "continuous" };
  }

  if (hasProse && (hasMoveNum || hasSan || hasFigurine || trimmed.toLowerCase().includes("diagram"))) {
    return { isNotation: true, kind: "prose" };
  }

  return { isNotation: false, kind: "continuous" };
}

type NormalizedTextLine = {
  text: string;
  x: number;
  y: number;
  width: number;
  height: number;
  kind: NotationHighlightKind;
};

export function extractNotationHighlights(
  items: PDFTextItemInput[],
  viewport: { width: number; height: number; convertToViewportPoint?: (x: number, y: number) => [number, number] },
  candidates: Candidate[],
): NotationHighlight[] {
  if (!items.length || !candidates.length || viewport.width <= 0 || viewport.height <= 0) {
    return [];
  }

  const textItems: { text: string; x: number; y: number; width: number; height: number }[] = [];

  for (const item of items) {
    if (!item.str || !item.str.trim()) continue;
    const tx = item.transform[4];
    const ty = item.transform[5];
    const w = item.width || Math.abs(item.transform[0]) || 8;
    const h = item.height || Math.abs(item.transform[3]) || 10;

    let x1: number, y1: number, x2: number, y2: number;
    if (typeof viewport.convertToViewportPoint === "function") {
      [x1, y1] = viewport.convertToViewportPoint(tx, ty + h);
      [x2, y2] = viewport.convertToViewportPoint(tx + w, ty);
    } else {
      x1 = tx;
      y1 = viewport.height - (ty + h);
      x2 = tx + w;
      y2 = viewport.height - ty;
    }

    const normX = Math.max(0, Math.min(1, Math.min(x1, x2) / viewport.width));
    const normY = Math.max(0, Math.min(1, Math.min(y1, y2) / viewport.height));
    const normW = Math.max(0.005, Math.min(1 - normX, Math.abs(x2 - x1) / viewport.width));
    const normH = Math.max(0.005, Math.min(1 - normY, Math.abs(y2 - y1) / viewport.height));

    textItems.push({
      text: item.str,
      x: normX,
      y: normY,
      width: normW,
      height: normH,
    });
  }

  if (!textItems.length) return [];

  textItems.sort((a, b) => (Math.abs(a.y - b.y) < 0.008 ? a.x - b.x : a.y - b.y));

  const lines: NormalizedTextLine[] = [];
  let currentGroup: typeof textItems = [];

  function flushGroup() {
    if (!currentGroup.length) return;
    const text = currentGroup.map((i) => i.text).join(" ").trim();
    const { isNotation, kind } = isChessNotationText(text);

    if (isNotation) {
      const minX = Math.min(...currentGroup.map((i) => i.x));
      const minY = Math.min(...currentGroup.map((i) => i.y));
      const maxX = Math.max(...currentGroup.map((i) => i.x + i.width));
      const maxY = Math.max(...currentGroup.map((i) => i.y + i.height));
      lines.push({
        text,
        x: Math.max(0, minX - 0.003),
        y: Math.max(0, minY - 0.002),
        width: Math.min(1 - minX, (maxX - minX) + 0.006),
        height: Math.min(1 - minY, (maxY - minY) + 0.004),
        kind,
      });
    }
    currentGroup = [];
  }

  for (const item of textItems) {
    if (!currentGroup.length) {
      currentGroup.push(item);
    } else {
      const last = currentGroup[currentGroup.length - 1];
      const sameLine = Math.abs(item.y - last.y) < 0.01;
      const nearbyX = item.x - (last.x + last.width) < 0.08;
      if (sameLine && nearbyX) {
        currentGroup.push(item);
      } else {
        flushGroup();
        currentGroup.push(item);
      }
    }
  }
  flushGroup();

  if (!lines.length) return [];

  const result: NotationHighlight[] = [];

  for (let index = 0; index < lines.length; index++) {
    const line = lines[index];
    let bestCandidate = candidates[0];
    let bestDistance = Number.POSITIVE_INFINITY;

    for (const candidate of candidates) {
      const candBottom = candidate.y + candidate.height;
      const candTop = candidate.y;
      const candCenterX = candidate.x + candidate.width / 2;
      const lineCenterX = line.x + line.width / 2;

      const isBelow = line.y >= candTop - 0.02 && line.y <= candBottom + 0.35;
      const xDist = Math.abs(lineCenterX - candCenterX);
      const yDist = line.y >= candBottom ? (line.y - candBottom) : Math.abs(line.y - candTop);

      const score = (isBelow ? 0 : 10) + yDist * 2 + xDist * 1.5;

      if (score < bestDistance) {
        bestDistance = score;
        bestCandidate = candidate;
      }
    }

    result.push({
      id: "highlight-" + bestCandidate.id + "-" + (index + 1),
      candidateId: bestCandidate.id,
      kind: line.kind,
      text: line.text,
      x: line.x,
      y: line.y,
      width: line.width,
      height: line.height,
    });
  }

  return result;
}

  wk:"?", wq:"?", wr:"?", wb:"?", wn:"?", wp:"?",

function parseLine(input: string) {
  const chess = new Chess();
  const cleaned = input
    .replace(/\{[^}]*\}/g, " ")
    .replace(/\([^)]*\)/g, " ")
    .replace(/^\s*\[.*\]\s*$/gm, " ")
    .replace(/\d+\.(\.\.)?/g, " ")
    .replace(/\$\d+/g, " ")
    .replace(/\s+/g, " ")
    .trim();

  const tokens = cleaned.split(" ").filter(Boolean)
    .filter((x) => !["1-0", "0-1", "1/2-1/2", "*"].includes(x));

  const moves: string[] = [];
  for (const token of tokens) {
    try {
      const move = chess.move(token);
      moves.push(move.san);
    } catch {
      throw new Error(`Invalid or illegal move: ${token}`);
    }
  }
  return moves;
}

import { Chess, type Square } from "chess.js";

export type OpeningLineResult = {
  fen: string;
  sanMoves: string[];
  uciMoves: string[];
  error: string | null;
};

export function parseOpeningLine(input: string): OpeningLineResult {
  const chess = new Chess();
  const sanMoves: string[] = [];
  const uciMoves: string[] = [];

  const tokens = input
    .replace(/\{[^}]*\}/g, " ")
    .replace(/\([^)]*\)/g, " ")
    .replace(/\d+\.(\.\.)?/g, " ")
    .replace(/\$\d+/g, " ")
    .split(/\s+/)
    .filter(Boolean)
    .filter((token) => !["1-0", "0-1", "1/2-1/2", "*"].includes(token));

  for (const token of tokens) {
    try {
      const uci = token.match(/^([a-h][1-8])([a-h][1-8])([qrbn])?$/i);
      const move = uci
        ? chess.move({
            from: uci[1].toLowerCase() as Square,
            to: uci[2].toLowerCase() as Square,
            promotion: uci[3]?.toLowerCase(),
          })
        : chess.move(token);

      sanMoves.push(move.san);
      uciMoves.push(`${move.from}${move.to}${move.promotion ?? ""}`);
    } catch {
      return {
        fen: chess.fen(),
        sanMoves,
        uciMoves,
        error: `Invalid or illegal move: ${token}`,
      };
    }
  }

  return { fen: chess.fen(), sanMoves, uciMoves, error: null };
}

export function openingPositionAt(input: string, ply: number): OpeningLineResult {
  const parsed = parseOpeningLine(input);
  if (parsed.error) return parsed;

  const limit = Math.max(0, Math.min(parsed.sanMoves.length, Math.floor(ply)));
  const chess = new Chess();
  const sanMoves: string[] = [];
  const uciMoves: string[] = [];

  for (let i = 0; i < limit; i++) {
    const move = chess.move(parsed.sanMoves[i]);
    sanMoves.push(move.san);
    uciMoves.push(`${move.from}${move.to}${move.promotion ?? ""}`);
  }

  return { fen: chess.fen(), sanMoves, uciMoves, error: null };
}

export type OpeningExplorerMove = {
  uci: string;
  san: string;
  white: number;
  draws: number;
  black: number;
  opening?: { eco?: string; name: string } | null;
};

export type OpeningExplorerStats = {
  white: number;
  draws: number;
  black: number;
  total: number;
  moves: OpeningExplorerMove[];
  opening?: { eco?: string; name: string } | null;
};

function count(value: unknown): number {
  return typeof value === "number" && Number.isFinite(value) && value >= 0
    ? value
    : 0;
}

function opening(value: unknown): OpeningExplorerStats["opening"] {
  if (!value || typeof value !== "object") return null;
  const item = value as Record<string, unknown>;
  if (typeof item.name !== "string") return null;
  return {
    name: item.name,
    ...(typeof item.eco === "string" ? { eco: item.eco } : {}),
  };
}

export async function fetchOpeningExplorerStats(
  uciMoves: string[],
  signal?: AbortSignal,
): Promise<OpeningExplorerStats> {
  const query = new URLSearchParams();
  query.set("variant", "standard");
  query.set("moves", "8");
  if (uciMoves.length) query.set("play", uciMoves.join(","));

  const response = await fetch(
    `/api/lichess/opening-explorer?${query.toString()}`,
    {
      method: "GET",
      credentials: "same-origin",
      headers: { Accept: "application/json" },
      signal,
    },
  );

  if (!response.ok) {
    throw new Error(`Opening Explorer request failed (${response.status}).`);
  }

  const raw: unknown = await response.json();
  if (!raw || typeof raw !== "object") {
    throw new Error("Opening Explorer returned an invalid response.");
  }

  const data = raw as Record<string, unknown>;
  const white = count(data.white);
  const draws = count(data.draws);
  const black = count(data.black);

  const moves: OpeningExplorerMove[] = Array.isArray(data.moves)
    ? data.moves.flatMap((entry): OpeningExplorerMove[] => {
        if (!entry || typeof entry !== "object") return [];
        const item = entry as Record<string, unknown>;
        if (typeof item.uci !== "string" || typeof item.san !== "string") return [];
        return [{
          uci: item.uci,
          san: item.san,
          white: count(item.white),
          draws: count(item.draws),
          black: count(item.black),
          opening: opening(item.opening),
        }];
      })
    : [];

  return {
    white,
    draws,
    black,
    total: white + draws + black,
    moves,
    opening: opening(data.opening),
  };
}

export type LichessConnectionStatus = {
  connected: boolean;
  configured: boolean;
};

export async function getLichessConnectionStatus(
  signal?: AbortSignal,
): Promise<LichessConnectionStatus> {
  const response = await fetch("/api/lichess/status", {
    method: "GET",
    credentials: "same-origin",
    headers: { Accept: "application/json" },
    cache: "no-store",
    signal,
  });
  if (!response.ok) {
    throw new Error(`Connection status request failed (${response.status}).`);
  }
  const value: unknown = await response.json();
  if (!value || typeof value !== "object") {
    throw new Error("Invalid connection status response.");
  }
  const status = value as Record<string, unknown>;
  return {
    connected: status.connected === true,
    configured: status.configured === true,
  };
}

export async function disconnectLichess(): Promise<void> {
  const response = await fetch("/api/lichess/disconnect", {
    method: "POST",
    credentials: "same-origin",
    headers: { Accept: "application/json" },
  });
  if (!response.ok) {
    throw new Error(`Disconnect request failed (${response.status}).`);
  }
}

  B: "White bishop",
  R: "White rook",
  Q: "White queen",
  K: "White king",
  p: "Black pawn",
  n: "Black knight",
  b: "Black bishop",
  r: "Black rook",
  q: "Black queen",
  k: "Black king",
};

export function parseFenPlacement(placement: string): string[][] | null {
  const ranks = placement.split("/");
  if (ranks.length !== 8) return null;
  const board: string[][] = [];
  for (const rank of ranks) {
    const squares: string[] = [];
    for (const token of rank) {
      if (/^[1-8]$/.test(token)) squares.push(...Array(Number(token)).fill(""));
      else if (PIECE_CODES.test(token)) squares.push(token);
      else return null;
    }
    if (squares.length !== 8) return null;
    board.push(squares);
  }
  return board;
}

export function serializeFenPlacement(board: string[][]): string {
  if (board.length !== 8 || board.some((rank) => rank.length !== 8)) {
    throw new Error("A chess position must contain exactly 64 squares.");
  }
  return board.map((rank) => {
    let result = "";
    let empty = 0;
    for (const piece of rank) {
      if (!piece) {
        empty += 1;
      } else {
        if (!PIECE_CODES.test(piece)) throw new Error(`Invalid chess piece code: ${piece}`);
        if (empty) result += empty;
        empty = 0;
        result += piece;
      }
    }
    if (empty) result += empty;
    return result;
  }).join("/");
}

export function setFenPiece(placement: string, row: number, col: number, piece: string): string {
  const board = parseFenPlacement(placement);
  if (!board || row < 0 || row > 7 || col < 0 || col > 7) {
    throw new Error("Cannot edit an invalid chessboard position.");
  }
  if (piece && !PIECE_CODES.test(piece)) throw new Error(`Invalid chess piece code: ${piece}`);
  board[row][col] = piece;
  return serializeFenPlacement(board);
}

export function moveFenPiece(placement: string, fromRow: number, fromCol: number, toRow: number, toCol: number): string {
  const board = parseFenPlacement(placement);
  if (!board || fromRow < 0 || fromRow > 7 || fromCol < 0 || fromCol > 7 || toRow < 0 || toRow > 7 || toCol < 0 || toCol > 7) {
    throw new Error("Cannot move a piece on an invalid chessboard position.");
  }
  const piece = board[fromRow][fromCol];
  if (!piece) return placement;
  board[fromRow][fromCol] = "";
  board[toRow][toCol] = piece;
  return serializeFenPlacement(board);
}

export function flipFenPlacement(placement: string): string {
  const board = parseFenPlacement(placement);
  if (!board) throw new Error("Cannot flip an invalid chessboard position.");
  return serializeFenPlacement(board.reverse().map((rank) => rank.reverse()));
}

export function withFenPlacement(fen: string, placement: string): string {
  const fields = fen.trim().split(/\s+/);
  if (fields.length !== 6 || !parseFenPlacement(placement)) {
    throw new Error("A complete six-field FEN is required.");
  }
  fields[0] = placement;
  return fields.join(" ");
}

export function countKings(placementOrFen: string): { whiteKings: number; blackKings: number } {
  const placement = placementOrFen.trim().split(/\s+/)[0];
  const board = parseFenPlacement(placement);
  let whiteKings = 0;
  let blackKings = 0;
  if (board) {
    for (const rank of board) {
      for (const square of rank) {
        if (square === "K") whiteKings += 1;
        if (square === "k") blackKings += 1;
      }
    }
  }
  return { whiteKings, blackKings };
}

export function sanitizeCastlingRights(board: string[][], currentCastling: string): string {
  if (!currentCastling || currentCastling === "-") return "-";
  let result = "";
  // White king must be on e1 (row 7, col 4)
  const whiteKingOnE1 = board[7]?.[4] === "K";
  if (whiteKingOnE1) {
    if (currentCastling.includes("K") && board[7]?.[7] === "R") result += "K";
    if (currentCastling.includes("Q") && board[7]?.[0] === "R") result += "Q";
  }
  // Black king must be on e8 (row 0, col 4)
  const blackKingOnE8 = board[0]?.[4] === "k";
  if (blackKingOnE8) {
    if (currentCastling.includes("k") && board[0]?.[7] === "r") result += "k";
    if (currentCastling.includes("q") && board[0]?.[0] === "r") result += "q";
  }
  return result || "-";
}

export function buildFullFen(
  placement: string,
  turn: "w" | "b" = "w",
  castling?: string,
  enPassant: string = "-",
  halfmove: number = 0,
  fullmove: number = 1
): string {
  const board = parseFenPlacement(placement);
  if (!board) throw new Error("Invalid piece placement for FEN generation.");
  let effectiveCastling = castling;
  if (!effectiveCastling) {
    effectiveCastling = placement === START_PLACEMENT ? "KQkq" : "-";
  } else {
    effectiveCastling = sanitizeCastlingRights(board, effectiveCastling);
  }
  return `${placement} ${turn} ${effectiveCastling} ${enPassant} ${halfmove} ${fullmove}`;
}

export function validatePositionFen(fen: string): { isValid: boolean; error?: string } {
  const trimmed = fen.trim();
  if (!trimmed) {
    return { isValid: false, error: "Position FEN cannot be empty." };
  }
  const fields = trimmed.split(/\s+/);
  if (fields.length !== 6) {
    return { isValid: false, error: "FEN must contain exactly 6 space-separated fields." };
  }
  const [placement, turn] = fields;
  const board = parseFenPlacement(placement);
  if (!board) {
    return { isValid: false, error: "Invalid board piece placement." };
  }

  let whiteKings = 0;
  let blackKings = 0;
  let invalidPawns = 0;
  for (let r = 0; r < 8; r++) {
    for (let c = 0; c < 8; c++) {
      const piece = board[r][c];
      if (piece === "K") whiteKings += 1;
      if (piece === "k") blackKings += 1;
      if ((r === 0 || r === 7) && (piece === "P" || piece === "p")) {
        invalidPawns += 1;
      }
    }
  }

  if (whiteKings === 0) {
    return { isValid: false, error: "Position is missing a White king (K)." };
  }
  if (whiteKings > 1) {
    return { isValid: false, error: `Position has ${whiteKings} White kings. Exactly one is required.` };
  }
  if (blackKings === 0) {
    return { isValid: false, error: "Position is missing a Black king (k)." };
  }
  if (blackKings > 1) {
    return { isValid: false, error: `Position has ${blackKings} Black kings. Exactly one is required.` };
  }
  if (invalidPawns > 0) {
    return { isValid: false, error: "Pawns cannot be placed on the first or eighth rank." };
  }
  if (turn !== "w" && turn !== "b") {
    return { isValid: false, error: "Side to move must be 'w' (White) or 'b' (Black)." };
  }

  const result = validateFen(trimmed);
  if (!result.ok) {
    return { isValid: false, error: result.error || "Invalid chess position." };
  }

  try {
    new Chess(trimmed);
  } catch (err: unknown) {
    return { isValid: false, error: err instanceof Error ? err.message : String(err) };
  }

  return { isValid: true };
}

export type EngineStatus = "off" | "loading" | "ready" | "analyzing" | "error";
export type EngineScore = { kind: "cp" | "mate"; value: number; depth: number; bestMove?: string; pv?: string[] };

type UciRequest = { fen: string; depth: number };

export function parseUciInfo(line: string, sideToMove: "w" | "b"): Omit<EngineScore, "bestMove"> | null {
  if (!line.startsWith("info ")) return null;
  const depth = Number(line.match(/\bdepth (\d+)/)?.[1]);
  const score = line.match(/\bscore (cp|mate) (-?\d+)/);
  if (!score || !Number.isFinite(depth)) return null;
  const value = Number(score[2]);
  const whiteValue = sideToMove === "w" ? value : -value;
  const kind = score[1] as "cp" | "mate";
  const orientedValue = kind === "mate" && value === 0
    ? (sideToMove === "w" ? -1 : 1)
    : whiteValue;
  const pv = line.match(/\bpv (.+)$/)?.[1]?.trim().split(/\s+/);
  return {
    kind,
    value: orientedValue,
    depth,
    ...(pv?.length ? { pv } : {}),
  };
}

export function formatEvaluation(score: EngineScore): string {
  if (score.kind === "mate") return `#${score.value > 0 ? "+" : ""}${score.value}`;
  const pawns = score.value / 100;
  return `${pawns > 0 ? "+" : ""}${pawns.toFixed(2)}`;
}

export function whiteScoreShare(score: EngineScore): number {
  if (score.kind === "mate") return score.value > 0 ? 0.96 : 0.04;
  const bounded = Math.max(-1000, Math.min(1000, score.value));
  return Math.max(0.04, Math.min(0.96, 1 / (1 + 10 ** (-bounded / 400))));
}

export class StockfishEngine {
  private readonly worker: Worker;
  private ready = false;
  private searching = false;
  private awaitingStop = false;
  private queued: UciRequest | null = null;
  private currentSide: "w" | "b" = "w";
  private latestScore: EngineScore | null = null;

  constructor(
    workerUrl: string,
    private readonly onStatus: (status: EngineStatus, message?: string) => void,
    private readonly onScore: (score: EngineScore | null) => void,
  ) {
    this.worker = new Worker(workerUrl);
    this.worker.onmessage = (event: MessageEvent<string>) => this.handleLine(String(event.data));
    this.worker.onerror = (event) => {
      this.onStatus("error", event.message || "The Stockfish worker failed to load.");
      this.dispose();
    };
    this.onStatus("loading");
    this.worker.postMessage("uci");
  }

  analyze(fen: string, depth: number): void {
    this.queued = { fen, depth: Math.max(1, Math.min(24, Math.floor(depth))) };
    this.latestScore = null;
    this.onScore(null);
    if (!this.ready) return;
    if (this.searching) {
      if (!this.awaitingStop) {
        this.awaitingStop = true;
        this.worker.postMessage("stop");
      }
      return;
    }
    this.startQueuedAnalysis();
  }

  dispose(): void {
    this.worker.terminate();
    this.ready = false;
    this.searching = false;
    this.queued = null;
  }

  private handleLine(line: string): void {
    if (/^Error:/i.test(line)) {
      this.onStatus("error", line);
      this.dispose();
      return;
    }
    if (line === "uciok") {
      this.worker.postMessage("isready");
      return;
    }
    if (line === "readyok") {
      this.ready = true;
      this.onStatus("ready");
      this.startQueuedAnalysis();
      return;
    }
    if (line.startsWith("info ")) {
      if (this.awaitingStop) return;
      const score = parseUciInfo(line, this.currentSide);
      if (score) {
        // Use the current principal variation's first move for a live arrow.
        // Stockfish's final "bestmove" will replace this provisional move.
        const provisionalMove = score.pv?.[0];
        this.latestScore = {
          ...score,
          ...(provisionalMove ? { bestMove: provisionalMove } : {}),
        };
        this.onScore(this.latestScore);
      }
      return;
    }
    const bestMove = line.match(/^bestmove (\S+)/)?.[1];
    if (!bestMove) return;
    this.searching = false;
    if (this.awaitingStop) {
      this.awaitingStop = false;
      this.startQueuedAnalysis();
      return;
    }
    if (this.latestScore) {
      this.latestScore = { ...this.latestScore, bestMove: bestMove === "(none)" ? undefined : bestMove };
      this.onScore(this.latestScore);
    }
    this.onStatus("ready");
  }

  private startQueuedAnalysis(): void {
    if (!this.ready || this.searching || !this.queued) return;
    const request = this.queued;
    this.queued = null;
    this.currentSide = request.fen.split(/\s+/)[1] === "b" ? "b" : "w";
    this.searching = true;
    this.latestScore = null;
    this.onStatus("analyzing");
    this.worker.postMessage(`position fen ${request.fen}`);
    this.worker.postMessage(`go depth ${request.depth}`);
  }
}

import type { BookRecord, BookSession } from "./model";

const DB_NAME = "chessbook-reader";
const DB_VERSION = 1;
const ACTIVE_KEY = "activeBookId";

function openDatabase(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, DB_VERSION);
    request.onupgradeneeded = () => {
      const db = request.result;
      if (!db.objectStoreNames.contains("books")) {
        db.createObjectStore("books", { keyPath: "id" });
      }
      if (!db.objectStoreNames.contains("sessions")) {
        db.createObjectStore("sessions", { keyPath: "bookId" });
      }
      if (!db.objectStoreNames.contains("settings")) {
        db.createObjectStore("settings", { keyPath: "key" });
      }
    };
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error ?? new Error("Could not open local library."));
  });
}

async function transact<T>(
  storeName: string,
  mode: IDBTransactionMode,
  run: (store: IDBObjectStore) => IDBRequest<T>,
): Promise<T> {
  const db = await openDatabase();
  try {
    return await new Promise<T>((resolve, reject) => {
      const transaction = db.transaction(storeName, mode);
      let outcome: { value: T } | null = null;
      transaction.oncomplete = () => {
        if (outcome) resolve(outcome.value);
        else reject(new Error("Local library transaction completed without a result."));
      };
      transaction.onerror = () => reject(transaction.error ?? new Error("Local library transaction failed."));
      transaction.onabort = () => reject(transaction.error ?? new Error("Local library transaction aborted."));
      const request = run(transaction.objectStore(storeName));
      request.onsuccess = () => { outcome = { value: request.result }; };
      request.onerror = () => reject(request.error ?? new Error("Local library operation failed."));
    });
  } finally {
    db.close();
  }
}
