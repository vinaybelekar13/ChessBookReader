import type { EngineScore } from "./stockfish";
import { ReviewEngine, type ReviewProgress } from "./review-engine";
import type { ReviewGame } from "./review-pgn";

export type MoveClassification =
  | "Best"
  | "Excellent"
  | "Good"
  | "Inaccuracy"
  | "Mistake"
  | "Blunder";

export type ClassificationThresholds = {
  best: number;
  excellent: number;
  good: number;
  inaccuracy: number;
  mistake: number;
};

export const DEFAULT_CLASSIFICATION_THRESHOLDS: ClassificationThresholds = {
  best: 10,
  excellent: 30,
  good: 80,
  inaccuracy: 150,
  mistake: 300,
};

export type ReviewedMove = {
  ply: number;
  moveNumber: number;
  color: "w" | "b";
  san: string;
  from: string;
  to: string;
  promotion?: string;
  playedMoveUci: string;
  bestMove?: string;
  playedMoveIsBest: boolean;
  before: EngineScore;
  after: EngineScore;
  evaluationLossCp: number;
  classification: MoveClassification;
};

export type ReviewAnalysis = {
  depth: number;
  positionsAnalyzed: number;
  moves: ReviewedMove[];
  positionScores: EngineScore[];
};

export type ReviewEnginePort = Pick<ReviewEngine, "analyzePositions">;

export function scoreToWhiteCentipawns(score: EngineScore): number {
  if (score.kind === "cp") return score.value;

  const distance = Math.min(99999, Math.abs(score.value));
  return Math.sign(score.value || 1) * (100000 - distance);
}

export function calculateMoveLoss(
  before: EngineScore,
  after: EngineScore,
  color: "w" | "b",
): number {
  const beforeCp = scoreToWhiteCentipawns(before);
  const afterCp = scoreToWhiteCentipawns(after);

  const loss = color === "w"
    ? beforeCp - afterCp
    : afterCp - beforeCp;

  return Math.max(0, Math.round(loss));
}

export function classifyMove(
  lossCp: number,
  thresholds: ClassificationThresholds = DEFAULT_CLASSIFICATION_THRESHOLDS,
): MoveClassification {
  if (!Number.isFinite(lossCp) || lossCp < 0) {
    throw new Error("Evaluation loss must be a finite, non-negative number.");
  }

  if (
    thresholds.best < 0 ||
    thresholds.excellent < thresholds.best ||
    thresholds.good < thresholds.excellent ||
    thresholds.inaccuracy < thresholds.good ||
    thresholds.mistake < thresholds.inaccuracy
  ) {
    throw new Error("Classification thresholds must be non-negative and increasing.");
  }

  if (lossCp <= thresholds.best) return "Best";
  if (lossCp <= thresholds.excellent) return "Excellent";
  if (lossCp <= thresholds.good) return "Good";
  if (lossCp <= thresholds.inaccuracy) return "Inaccuracy";
  if (lossCp <= thresholds.mistake) return "Mistake";
  return "Blunder";
}

export function buildReviewAnalysis(
  review: ReviewGame,
  positionScores: EngineScore[],
  depth: number,
  thresholds: ClassificationThresholds = DEFAULT_CLASSIFICATION_THRESHOLDS,
): ReviewAnalysis {
  if (positionScores.length !== review.positions.length + 1) {
    throw new Error("Expected one engine evaluation for every game position.");
  }

  const moves = review.positions.map((move, index): ReviewedMove => {
    const before = positionScores[index];
    const after = positionScores[index + 1];

    if (!before || !after) {
      throw new Error(`Missing engine evaluation around ply ${move.ply}.`);
    }

    const playedMoveUci =
      `${move.from}${move.to}${move.promotion ?? ""}`;

    const playedMoveIsBest = before.bestMove
      ? before.bestMove.toLowerCase() === playedMoveUci.toLowerCase()
      : false;

    const evaluationLossCp = calculateMoveLoss(
      before,
      after,
      move.color,
    );

    return {
      ply: move.ply,
      moveNumber: move.moveNumber,
      color: move.color,
      san: move.san,
      from: move.from,
      to: move.to,
      ...(move.promotion ? { promotion: move.promotion } : {}),
      playedMoveUci,
      ...(before.bestMove ? { bestMove: before.bestMove } : {}),
      playedMoveIsBest,
      before,
      after,
      evaluationLossCp,
      classification: playedMoveIsBest
        ? "Best"
        : classifyMove(evaluationLossCp, thresholds),
    };
  });

  return {
    depth,
    positionsAnalyzed: positionScores.length,
    moves,
    positionScores,
  };
}

export async function analyzeReviewPositions(
  review: ReviewGame,
  engine: ReviewEnginePort,
  options: {
    depth: number;
    onProgress?: (progress: ReviewProgress) => void;
    thresholds?: ClassificationThresholds;
  },
): Promise<ReviewAnalysis> {
  const depth = Math.max(1, Math.min(24, Math.floor(options.depth)));

  const fens = [
    review.startingFen,
    ...review.positions.map((position) => position.afterFen),
  ];

  const positionScores = await engine.analyzePositions(fens, {
    depth,
    onProgress: options.onProgress,
  });

  return buildReviewAnalysis(
    review,
    positionScores,
    depth,
    options.thresholds,
  );
}

import { parseUciInfo, type EngineScore } from "./stockfish";

export type ReviewProgress = {
  completed: number;
  total: number;
  currentPly: number;
};

export type ReviewAnalysisOptions = {
  depth: number;
  onProgress?: (progress: ReviewProgress) => void;
};

type ReviewBatch = {
  fens: string[];
  depth: number;
  index: number;
  scores: EngineScore[];
  onProgress?: (progress: ReviewProgress) => void;
  resolve: (scores: EngineScore[]) => void;
  reject: (error: Error) => void;
};

export class ReviewEngine {
  private readonly worker: Worker;
  private ready = false;
  private disposed = false;
  private latestScore: EngineScore | null = null;
  private activeBatch: ReviewBatch | null = null;
  private readyResolve!: () => void;
  private readyReject!: (error: Error) => void;

  private readonly readyPromise = new Promise<void>((resolve, reject) => {
    this.readyResolve = resolve;
    this.readyReject = reject;
  });

  constructor(workerUrl: string) {
    // Mark readiness rejection as handled even when no analysis is awaiting it.
    void this.readyPromise.catch(() => undefined);

    this.worker = new Worker(workerUrl);

    this.worker.onmessage = (event: MessageEvent<string>) => {
      this.handleLine(String(event.data));
    };

    this.worker.onerror = (event) => {
      this.fail(new Error(event.message || "Stockfish review worker failed."));
    };

    this.worker.postMessage("uci");
  }

  async analyzePositions(
    fens: string[],
    options: ReviewAnalysisOptions,
  ): Promise<EngineScore[]> {
    if (this.disposed) {
      throw new Error("The review engine has been disposed.");
    }

    if (this.activeBatch) {
      throw new Error("A review analysis is already running.");
    }

    if (fens.length === 0) {
      return [];
    }

    const depth = Math.max(1, Math.min(24, Math.floor(options.depth)));

    if (fens.some((fen) => !fen.trim())) {
      throw new Error("Every review position must contain a valid FEN.");
    }

    await this.readyPromise;

    if (this.disposed) {
      throw new Error("The review engine has been disposed.");
    }

    return new Promise<EngineScore[]>((resolve, reject) => {
      this.activeBatch = {
        fens: [...fens],
        depth,
        index: 0,
        scores: [],
        onProgress: options.onProgress,
        resolve,
        reject,
      };

      this.startPosition();
    });
  }

  cancel(): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;

    this.disposed = true;
    this.worker.terminate();

    const error = new Error("Review analysis was cancelled.");
    batch?.reject(error);
    this.readyReject(error);
  }

  dispose(): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;
    this.disposed = true;

    this.worker.terminate();

    const error = new Error("The review engine has been disposed.");
    batch?.reject(error);
    this.readyReject(error);
  }

  private handleLine(line: string): void {
    if (this.disposed) return;

    if (/^Error:/i.test(line)) {
      this.fail(new Error(line));
      return;
    }

    if (line === "uciok") {
      this.worker.postMessage("isready");
      return;
    }

    if (line === "readyok") {
      this.ready = true;
      this.readyResolve();
      return;
    }

    const batch = this.activeBatch;
    if (!batch) return;

    if (line.startsWith("info ")) {
      const sideToMove = batch.fens[batch.index].split(/\s+/)[1] === "b" ? "b" : "w";
      const parsed = parseUciInfo(line, sideToMove);

      if (parsed && (!this.latestScore || parsed.depth >= this.latestScore.depth)) {
        this.latestScore = parsed;
      }

      return;
    }

    if (!line.startsWith("bestmove ")) return;

    const bestMove = line.match(/^bestmove (\S+)/)?.[1];

    if (!this.latestScore) {
      this.fail(new Error(`Stockfish returned no evaluation for ply ${batch.index + 1}.`));
      return;
    }

    const score: EngineScore = {
      ...this.latestScore,
      ...(bestMove && bestMove !== "(none)" ? { bestMove } : {}),
    };

    batch.scores.push(score);
    batch.index += 1;

    batch.onProgress?.({
      completed: batch.index,
      total: batch.fens.length,
      currentPly: batch.index,
    });

    this.latestScore = null;

    if (batch.index >= batch.fens.length) {
      this.activeBatch = null;
      batch.resolve(batch.scores);
      return;
    }

    this.startPosition();
  }

  private startPosition(): void {
    const batch = this.activeBatch;
    if (!batch || this.disposed || !this.ready) return;

    this.latestScore = null;

    const fen = batch.fens[batch.index];
    this.worker.postMessage(`position fen ${fen}`);
    this.worker.postMessage(`go depth ${batch.depth}`);
  }

  private fail(error: Error): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;
    this.disposed = true;

    this.worker.terminate();
    batch?.reject(error);
    this.readyReject(error);
  }
}

import { Chess } from "chess.js";
import { parsePgn } from "./pgn-import";

export type ReviewPosition = {
  ply: number;
  moveNumber: number;
  color: "w" | "b";
  san: string;
  beforeFen: string;
  afterFen: string;
  from: string;
  to: string;
  promotion?: string;
  flags: string;
};

export type ReviewGame = {
  headers: Record<string, string>;
  startingFen: string;
  finalFen: string;
  positions: ReviewPosition[];
};

export function reconstructReviewGame(pgn: string): ReviewGame {
  const imported = parsePgn(pgn);
  const game = new Chess(imported.lineBaseFen);
  const positions: ReviewPosition[] = [];

  imported.moves.forEach((san, index) => {
    const beforeFen = game.fen();
    const move = game.move(san);

    positions.push({
      ply: index + 1,
      moveNumber: Math.floor(index / 2) + 1,
      color: index % 2 === 0 ? "w" : "b",
      san: move.san,
      beforeFen,
      afterFen: game.fen(),
      from: move.from,
      to: move.to,
      ...(move.promotion ? { promotion: move.promotion } : {}),
      flags: move.flags,
    });
  });

  if (game.fen() !== imported.finalFen) {
    throw new Error("Review reconstruction does not match the imported PGN.");
  }

  return {
    headers: imported.headers,
    startingFen: imported.lineBaseFen,
    finalFen: game.fen(),
    positions,
  };
}

export function getReviewFenAtPly(review: ReviewGame, ply: number): string {
  const safePly = Math.max(0, Math.min(review.positions.length, Math.floor(ply)));

  return safePly === 0
    ? review.startingFen
    : review.positions[safePly - 1].afterFen;
}

function evaluationText(score?: { kind: "cp" | "mate"; value: number }) {
  if (!score) return "—";
  if (score.kind === "mate") return `${score.value > 0 ? "+" : ""}#${Math.abs(score.value)}`;
  const value = score.value / 100;
  return `${value > 0 ? "+" : ""}${value.toFixed(2)}`;
}

function resultText(result?: string) {
  if (result === "1-0") return "White won";
  if (result === "0-1") return "Black won";
  if (result === "1/2-1/2") return "Draw";
  return "Result not recorded";
}

function metadata(value?: string) {
  return value && value !== "*" && !/^[?.]+$/.test(value) ? value : "";
}

function classificationClass(value: string) {
  return value.toLowerCase().replace(/\s+/g, "-");
}

  }

  return (
    <main className={`game-review-page gr-live ${review ? "has-review" : "is-empty"}`}>
      <input ref={fileInput} type="file" accept=".pgn,text/plain" hidden onChange={handleFile} />

      {false && !review && (
        <>
          <header className="gr-live-header">
            <div>
              <div className="gr-live-eyebrow">CHESS ANALYSIS / GAME REVIEW</div>
              <h1>Game Review</h1>
              <p>Import a game, analyse every position and inspect your decisions.</p>
            </div>
            <button className="gr-live-button primary" onClick={() => fileInput.current?.click()}>
              <FileUp size={16} /> Import PGN
            </button>
          </header>

          {error && (
            <div className="gr-live-error" role="alert">
              <span>{error}</span>
              <button onClick={() => setError("")} aria-label="Dismiss error"><X size={16} /></button>
            </div>
          )}

          <section className="gr-live-import">
            <div className="gr-live-import-icon"><FileUp size={26} /></div>
            <h2>Import a chess game</h2>
            <p>Choose a PGN file or paste the PGN notation below.</p>
            <textarea
              value={pgnText}
              onChange={event => setPgnText(event.target.value)}
              placeholder={'[Event "Casual Game"]\n[White "Player 1"]\n[Black "Player 2"]\n\n1. e4 e5 2. Nf3 Nc6 *'}
              rows={9}
              spellCheck={false}
            />
            <button className="gr-live-button primary" disabled={!pgnText.trim()} onClick={() => loadPgn(pgnText)}>
              Load PGN
            </button>
          </section>
        </>
      )}

      {review && (
        <section className="gr-live-review-shell">
          <section className="gr-live-board-column">
            <section className="gr-live-game-card">
              <div className="gr-live-player">
                <span className="gr-live-avatar white">W</span>
                <div><strong>{whiteName}</strong><span>White {metadata(review.headers.WhiteElo) && `· ${review.headers.WhiteElo}`}</span></div>
              </div>
              <strong className="gr-live-result">{gameResult}</strong>
              <div className="gr-live-player black">
                <span className="gr-live-avatar black">B</span>
                <div><strong>{blackName}</strong><span>Black {metadata(review.headers.BlackElo) && `· ${review.headers.BlackElo}`}</span></div>
              </div>
              <div className="gr-live-meta">
                {[metadata(review.headers.Event), metadata(review.headers.Site), metadata(review.headers.Date), metadata(review.headers.Round), metadata(review.headers.ECO), metadata(review.headers.Opening), metadata(review.headers.TimeControl)].filter(Boolean).join(" · ") || "PGN game"}
                <span>{review.positions.length} plies</span>
                <span>{resultText(review.headers.Result)}</span>
              </div>
            </section>

            <div className="gr-live-board-wrap">
              <div className="gr-live-ranks">{[8,7,6,5,4,3,2,1].map(rank => <span key={rank}>{rank}</span>)}</div>
              <div className="gr-live-board">
                {board.flatMap((row, rowIndex) => row.map((piece, colIndex) => {
                  const square = `${"abcdefgh"[colIndex]}${8 - rowIndex}`;
                  const highlighted = currentMove && (currentMove.from === square || currentMove.to === square);
                  return (
                    <div key={square} className={`gr-live-square ${(rowIndex + colIndex) % 2 ? "dark" : "light"} ${highlighted ? "highlight" : ""}`}>
                      {piece && <img draggable={false} alt="" src={`${import.meta.env.BASE_URL}assets/pieces/standalone/${piece.color}${piece.type}.svg`} />}
                    </div>
                  );
                }))}
              </div>
              <div className="gr-live-files">{["a","b","c","d","e","f","g","h"].map(file => <span key={file}>{file}</span>)}</div>
            </div>
            <div className="gr-live-board-status">
              <span>{activePly === 0 ? "Starting position" : `${currentMove?.moveNumber ?? ""}${currentMove?.color === "w" ? "." : "..."} ${currentMove?.san ?? ""}`}</span>
              {reviewedMove && <span className="gr-live-board-eval">{reviewedMove.classification} · {evaluationText(reviewedMove.after)} · {reviewedMove.evaluationLossCp} cp</span>}
              <span>{activePly} / {review.positions.length}</span>
            </div>
          </section>

          <aside className="gr-live-analysis-panel">
            <header className="gr-live-review-header">
              <h1>Game Review</h1>
              <div className="gr-live-actions">
                <button className="gr-live-button secondary" onClick={resetReview}>
                  <RotateCcw size={15} /> New review
                </button>
                <button className="gr-live-button primary" onClick={() => fileInput.current?.click()}>
                  <FileUp size={16} /> Import PGN
                </button>
              </div>
            </header>

            {error && (
              <div className="gr-live-error" role="alert">
                <span>{error}</span>
                <button onClick={() => setError("")} aria-label="Dismiss error"><X size={16} /></button>
              </div>
            )}

            <section className="gr-live-import-panel">
              <div className="gr-live-import-heading">
                <div>
                  <strong>Load a game or position</strong>
                  <span>Import a game or analyse a FEN position</span>
                </div>
                <button
                  className="gr-live-button secondary"
                  onClick={() => setImportOpen(value => !value)}
                >
                  {importOpen ? "Hide import" : "Import game"}
                </button>
              </div>

              {importOpen && (
                <>
                  <div className="gr-live-import-tabs">
                    {([
                      ["pgn", "PGN"],
                      ["fen", "FEN"],
                      ["chesscom", "Chess.com"],
                      ["lichess", "Lichess"]
                    ] as [ImportMode, string][]).map(([mode, label]) => (
                      <button
                        key={mode}
                        className={importMode === mode ? "active" : ""}
                        onClick={() => {
                          setImportMode(mode);
                          setError("");
                          setImportedGames([]);
                        }}
                      >
                        {label}
                      </button>
                    ))}
                  </div>

                  {importMode === "pgn" && (
                    <div className="gr-live-import-content">
                      <textarea
                        value={pgnText}
                        onChange={event => setPgnText(event.target.value)}
                        placeholder={'[Event "Casual Game"]\n[White "Player 1"]\n[Black "Player 2"]\n\n1. e4 e5 2. Nf3 Nc6 *'}
                        rows={5}
                        spellCheck={false}
                      />
                      <div className="gr-live-import-actions">
                        <button className="gr-live-button secondary" onClick={() => fileInput.current?.click()}>
                          <FileUp size={15} /> Choose PGN file
                        </button>
                        <button
                          className="gr-live-button primary"
                          disabled={!pgnText.trim()}
                          onClick={() => { loadPgn(pgnText); setImportOpen(false); }}
                        >
                          Load PGN
                        </button>
                      </div>
                    </div>
                  )}

                  {importMode === "fen" && (
                    <div className="gr-live-import-content">
                      <textarea
                        value={fenText}
                        onChange={event => setFenText(event.target.value)}
                        placeholder="Paste a FEN position..."
                        rows={3}
                        spellCheck={false}
                      />
                      <button
                        className="gr-live-button primary"
                        disabled={!fenText.trim()}
                        onClick={() => { loadFenPosition(); setImportOpen(false); }}
                      >
                        Load position
                      </button>
                    </div>
                  )}

                  {(importMode === "chesscom" || importMode === "lichess") && (
                    <div className="gr-live-import-content">
                      <div className="gr-live-username-row">
                        <input
                          value={username}
                          onChange={event => setUsername(event.target.value)}
                          onKeyDown={event => {
                            if (event.key === "Enter") {
                              importMode === "chesscom"
                                ? void fetchChessComGames()
                                : void fetchLichessGames();
                            }
                          }}
                          placeholder={`${importMode === "chesscom" ? "Chess.com" : "Lichess"} username`}
                        />
                        <button
                          className="gr-live-button primary"
                          disabled={isImporting || !username.trim()}
                          onClick={() => importMode === "chesscom"
                            ? void fetchChessComGames()
                            : void fetchLichessGames()}
                        >
                          {isImporting ? <LoaderCircle className="gr-live-spinner" size={15} /> : "Find games"}
                        </button>
                      </div>
                      <p className="gr-live-muted">
                        {importMode === "chesscom"
                          ? "Fetches games from your three latest public monthly archives."
                          : "Fetches up to 25 recent public games."}
                      </p>
                      {importedGames.length > 0 && (
                        <div className="gr-live-import-game-list">
                          {importedGames.map(game => (
                            <button key={game.id} onClick={() => selectImportedGame(game)}>
                              <strong>{game.label}</strong>
                              <span>{game.detail}</span>
                            </button>
                          ))}
                        </div>
                      )}
                    </div>
                  )}
                </>
              )}
            </section>

            <section className="gr-live-toolbar">
              <div className="gr-live-depth">
                <label htmlFor="gr-live-depth">Analysis depth</label>
                <select id="gr-live-depth" value={depth} disabled={status === "running"} onChange={event => setDepth(Number(event.target.value))}>
                  {[10, 12, 14, 16, 18, 20, 22, 24].map(value => <option key={value} value={value}>{value}</option>)}
                </select>
              </div>
              {status === "running" ? (
                <button className="gr-live-button danger" onClick={cancelAnalysis}><Pause size={15} /> Cancel analysis</button>
              ) : (
                <button className="gr-live-button primary" onClick={startAnalysis}>
                  <BarChart3 size={16} /> {analysis ? "Analyse again" : "Start analysis"}
                </button>
              )}
            </section>

            {status === "running" && progress && (
              <section className="gr-live-progress">
                <div><LoaderCircle className="gr-live-spinner" size={16} /> Analysing position {progress.completed} of {progress.total}</div>
                <div className="gr-live-progress-track"><i style={{ width: `${progress.total ? progress.completed / progress.total * 100 : 0}%` }} /></div>
                <span>Stockfish depth {depth}. You can cancel at any time.</span>
              </section>
            )}

            {analysis && (
              <section className="gr-live-stats">
                <article><Target size={17} /><strong>{analysis.moves.length}</strong><span>Moves analysed</span></article>
                <article><Trophy size={17} /><strong>{counts.Best ?? 0}</strong><span>Best moves</span></article>
                <article><BarChart3 size={17} /><strong>{counts.Inaccuracy ?? 0}</strong><span>Inaccuracies</span></article>
                <article><Target size={17} /><strong>{(counts.Mistake ?? 0) + (counts.Blunder ?? 0)}</strong><span>Mistakes and blunders</span></article>
              </section>
            )}

            <nav className="gr-live-tabs">
              <button className={tab === "overview" ? "active" : ""} onClick={() => setTab("overview")}>Overview</button>
              <button className={tab === "moves" ? "active" : ""} onClick={() => setTab("moves")}>Move by move</button>
            </nav>

            <div className="gr-live-review-content">
              {tab === "overview" ? (
                <div className="gr-live-overview">
                  <section className="gr-live-panel">
                    <div className="gr-live-panel-heading"><div><h2>Key moments</h2><span>Moves with the largest evaluation losses</span></div></div>
                    {!analysis && <p className="gr-live-muted">Run Stockfish analysis to discover key moments in this game.</p>}
                    {analysis && keyMoments.length === 0 && <p className="gr-live-muted">No moves crossed the current 80 cp key-moment threshold.</p>}
                    {keyMoments.map(move => (
                      <button className="gr-live-moment" key={move.ply} onClick={() => goTo(move.ply)}>
                        <span className={`gr-live-tag ${classificationClass(move.classification)}`}>{move.classification}</span>
                        <strong>{move.moveNumber}{move.color === "w" ? "." : "..."} {move.san}</strong>
                        <span>{move.evaluationLossCp} cp loss</span>
                        <ArrowRight size={15} />
                      </button>
                    ))}
                  </section>

                  {analysis && (
                    <section className="gr-live-panel gr-live-classification-panel">
                      <div className="gr-live-panel-heading"><div><h2>Move classifications</h2><span>Counts from this Stockfish analysis</span></div></div>
                      <div className="gr-live-classification-summary">
                        {FILTERS.slice(1).map(value => (
                          <button key={value} onClick={() => { setFilter(value); setTab("moves"); }}>
                            <span className={`gr-live-tag ${classificationClass(value)}`}>{value}</span>
                            <strong>{counts[value] ?? 0}</strong>
                          </button>
                        ))}
                      </div>
                    </section>
                  )}

                  <section className="gr-live-panel gr-live-timeline">
                    <div className="gr-live-panel-heading"><div><h2>Move timeline</h2><span>Select a move to replay the position</span></div></div>
                    <div className="gr-live-move-grid">
                      {review.positions.map((move, index) => {
                        const result = analysis?.moves[index];
                        return (
                          <button key={move.ply} className={activePly === move.ply ? "selected" : ""} onClick={() => goTo(move.ply)}>
                            <span>{move.moveNumber}{move.color === "w" ? "." : "..."}</span>
                            <strong>{move.san}</strong>
                            {result && <i className={`gr-live-dot ${classificationClass(result.classification)}`} />}
                          </button>
                        );
                      })}
                    </div>
                  </section>
                </div>
              ) : (
                <section className="gr-live-panel gr-live-moves-panel">
                  <div className="gr-live-panel-heading"><div><h2>Move-by-move analysis</h2><span>{analysis ? `${analysis.moves.length} moves analysed at depth ${analysis.depth}` : "Start analysis to see engine evaluations and classifications"}</span></div></div>
                  <div className="gr-live-filters">
                    {FILTERS.map(value => <button key={value} className={filter === value ? "active" : ""} onClick={() => setFilter(value)}>{value}{value !== "All" && analysis ? ` ${counts[value] ?? 0}` : ""}</button>)}
                  </div>
                  {!analysis && <p className="gr-live-muted">No engine analysis yet.</p>}
                  {analysis && (
                    <div className="gr-live-move-list">
                      {filteredMoves.map((move: ReviewedMove) => (
                        <button key={move.ply} className={`gr-live-move-row ${activePly === move.ply ? "selected" : ""}`} onClick={() => { goTo(move.ply); setTab("overview"); }}>
                          <span>{move.moveNumber}{move.color === "w" ? "." : "..."}</span>
                          <strong>{move.san}</strong>
                          <span className={`gr-live-tag ${classificationClass(move.classification)}`}>{move.classification}</span>
                          <span>{evaluationText(move.after)}</span>
                          <span>{move.evaluationLossCp} cp</span>
                        </button>
                      ))}
                      {filteredMoves.length === 0 && <p className="gr-live-muted">No moves match this filter.</p>}
                    </div>
                  )}
                </section>
              )}
            </div>
            <div className="gr-live-controls">
              <button onClick={() => goTo(0)} title="Start">|◀</button>
              <button onClick={() => goTo(activePly - 1)} disabled={activePly === 0} title="Previous"><ArrowLeft size={17} /></button>
              <button className="primary" onClick={() => setIsPlaying(value => !value)} aria-label={isPlaying ? "Pause autoplay" : "Play moves"} title={isPlaying ? "Pause" : "Play"}>
                {isPlaying ? <Pause size={16} /> : <Play size={16} />}
              </button>
              <button onClick={() => goTo(activePly + 1)} disabled={activePly === review.positions.length} title="Next"><ArrowRight size={17} /></button>
              <button onClick={() => goTo(review.positions.length)} title="End">▶|</button>
            </div>
            <footer className="gr-live-footer">
              Stockfish {analysis ? `· Depth ${analysis.depth} · ${analysis.positionsAnalyzed} positions analysed` : "· Ready for analysis"} · Classifications use provisional centipawn thresholds, not Chess.com proprietary ratings.
            </footer>
          </aside>
        </section>
      )}
    </main>
  );
}
