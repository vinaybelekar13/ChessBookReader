 886:     if (!boardSession) return;
 887:     const activeBoard = boardSession;
 888:     try {
 889:       const currentGame = new Chess(activeBoard.currentFen);
 890:       const mover = currentGame.get(from);
 891:       const promote = mover?.type === "p" && (to.endsWith("8") || to.endsWith("1"));
 892:       if (promote) {
 893:         setPromotionRequest({ from, to });
 894:         return;
 895:       }
 896:       const result = currentGame.move({ from, to });
 897:       const nextMoves = [...activeBoard.moves.slice(0, activeBoard.moveIndex), result.san];
 898:       if (workspaceMode === "opening-explorer") {
 899:         const line = nextMoves
 900:           .map((move, index) => index % 2 === 0 ? `${Math.floor(index / 2) + 1}. ${move}` : move)
 901:           .join(" ");
 902:         setOpeningExplorerInput(line);
 903:         setOpeningExplorerPly(nextMoves.length);
 904:       } else {
 905:         patchBoardSession({
 906:           currentFen: currentGame.fen(),
 907:           moves: nextMoves,
 908:           moveIndex: nextMoves.length,
 909:         });
 910:       }
 911:       updateGameEndOverlay(currentGame);
 912:       setLastMove({ from, to });
 913:       setSelectedSquare(null);
 914:       setError("");
 915:     } catch {
 916:       const piece = game.get(to);
 917:       if (piece && piece.color === game.turn()) setSelectedSquare(to);
 918:       else setSelectedSquare(null);
 919:     }
 920:   }
