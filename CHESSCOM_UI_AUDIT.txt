function evaluationText(score?: { kind: "cp" | "mate"; value: number }) {
  if (!score) return "�";
  if (score.kind === "mate") return `${score.value > 0 ? "+" : ""}#${Math.abs(score.value)}`;
  const value = score.value / 100;
  return `${value > 0 ? "+" : ""}${value.toFixed(2)}`;

import type { EngineScore } from "./stockfish";
import { ReviewEngine, type ReviewProgress } from "./review-engine";
import type { ReviewGame } from "./review-pgn";

export type MoveClassification =
  | "Best"
  | "Excellent"
  | "Good"
  | "Inaccuracy"
  | "Mistake"
  | "Blunder";

export type ClassificationThresholds = {
  best: number;
  excellent: number;
  good: number;
  inaccuracy: number;
  mistake: number;
};

export const DEFAULT_CLASSIFICATION_THRESHOLDS: ClassificationThresholds = {
  best: 10,
  excellent: 30,
  good: 80,
  inaccuracy: 150,
  mistake: 300,
};

export type ReviewedMove = {
  ply: number;
  moveNumber: number;
  color: "w" | "b";
  san: string;
  from: string;
  to: string;
  promotion?: string;
  playedMoveUci: string;
  bestMove?: string;
  playedMoveIsBest: boolean;
  before: EngineScore;
  after: EngineScore;
  evaluationLossCp: number;
  classification: MoveClassification;
};

export type ReviewAnalysis = {
  depth: number;
  positionsAnalyzed: number;
  moves: ReviewedMove[];
  positionScores: EngineScore[];
};

export type ReviewEnginePort = Pick<ReviewEngine, "analyzePositions">;

export function scoreToWhiteCentipawns(score: EngineScore): number {
  if (score.kind === "cp") return score.value;

  const distance = Math.min(99999, Math.abs(score.value));
  return Math.sign(score.value || 1) * (100000 - distance);
}

export function calculateMoveLoss(
  before: EngineScore,
  after: EngineScore,
  color: "w" | "b",
): number {
  const beforeCp = scoreToWhiteCentipawns(before);
  const afterCp = scoreToWhiteCentipawns(after);

  const loss = color === "w"
    ? beforeCp - afterCp
    : afterCp - beforeCp;

  return Math.max(0, Math.round(loss));
}

export function classifyMove(
  lossCp: number,
  thresholds: ClassificationThresholds = DEFAULT_CLASSIFICATION_THRESHOLDS,
): MoveClassification {
  if (!Number.isFinite(lossCp) || lossCp < 0) {
    throw new Error("Evaluation loss must be a finite, non-negative number.");
  }

  if (
    thresholds.best < 0 ||
    thresholds.excellent < thresholds.best ||
    thresholds.good < thresholds.excellent ||
    thresholds.inaccuracy < thresholds.good ||
    thresholds.mistake < thresholds.inaccuracy
  ) {
    throw new Error("Classification thresholds must be non-negative and increasing.");
  }

  if (lossCp <= thresholds.best) return "Best";
  if (lossCp <= thresholds.excellent) return "Excellent";
  if (lossCp <= thresholds.good) return "Good";
  if (lossCp <= thresholds.inaccuracy) return "Inaccuracy";
  if (lossCp <= thresholds.mistake) return "Mistake";
  return "Blunder";
}

export function buildReviewAnalysis(
  review: ReviewGame,
  positionScores: EngineScore[],
  depth: number,
  thresholds: ClassificationThresholds = DEFAULT_CLASSIFICATION_THRESHOLDS,
): ReviewAnalysis {
  if (positionScores.length !== review.positions.length + 1) {
    throw new Error("Expected one engine evaluation for every game position.");
  }

  const moves = review.positions.map((move, index): ReviewedMove => {
    const before = positionScores[index];
    const after = positionScores[index + 1];

    if (!before || !after) {
      throw new Error(`Missing engine evaluation around ply ${move.ply}.`);
    }

    const playedMoveUci =
      `${move.from}${move.to}${move.promotion ?? ""}`;

    const playedMoveIsBest = before.bestMove
      ? before.bestMove.toLowerCase() === playedMoveUci.toLowerCase()
      : false;

    const evaluationLossCp = calculateMoveLoss(
      before,
      after,
      move.color,
    );

    return {
      ply: move.ply,
      moveNumber: move.moveNumber,
      color: move.color,
      san: move.san,
      from: move.from,
      to: move.to,
      ...(move.promotion ? { promotion: move.promotion } : {}),
      playedMoveUci,
      ...(before.bestMove ? { bestMove: before.bestMove } : {}),
      playedMoveIsBest,
      before,
      after,
      evaluationLossCp,
      classification: playedMoveIsBest
        ? "Best"
        : classifyMove(evaluationLossCp, thresholds),
    };
  });

  return {
    depth,
    positionsAnalyzed: positionScores.length,
    moves,
    positionScores,
  };
}

export async function analyzeReviewPositions(
  review: ReviewGame,
  engine: ReviewEnginePort,
  options: {
    depth: number;
    onProgress?: (progress: ReviewProgress) => void;
    thresholds?: ClassificationThresholds;
  },
): Promise<ReviewAnalysis> {
  const depth = Math.max(1, Math.min(24, Math.floor(options.depth)));

  const fens = [
    review.startingFen,
    ...review.positions.map((position) => position.afterFen),
  ];

  const positionScores = await engine.analyzePositions(fens, {
    depth,
    onProgress: options.onProgress,
  });

  return buildReviewAnalysis(
    review,
    positionScores,
    depth,
    options.thresholds,
  );
}

import { parseUciInfo, type EngineScore } from "./stockfish";

export type ReviewProgress = {
  completed: number;
  total: number;
  currentPly: number;
};

export type ReviewAnalysisOptions = {
  depth: number;
  onProgress?: (progress: ReviewProgress) => void;
};

type ReviewBatch = {
  fens: string[];
  depth: number;
  index: number;
  scores: EngineScore[];
  onProgress?: (progress: ReviewProgress) => void;
  resolve: (scores: EngineScore[]) => void;
  reject: (error: Error) => void;
};

export class ReviewEngine {
  private readonly worker: Worker;
  private ready = false;
  private disposed = false;
  private latestScore: EngineScore | null = null;
  private activeBatch: ReviewBatch | null = null;
  private readyResolve!: () => void;
  private readyReject!: (error: Error) => void;

  private readonly readyPromise = new Promise<void>((resolve, reject) => {
    this.readyResolve = resolve;
    this.readyReject = reject;
  });

  constructor(workerUrl: string) {
    // Mark readiness rejection as handled even when no analysis is awaiting it.
    void this.readyPromise.catch(() => undefined);

    this.worker = new Worker(workerUrl);

    this.worker.onmessage = (event: MessageEvent<string>) => {
      this.handleLine(String(event.data));
    };

    this.worker.onerror = (event) => {
      this.fail(new Error(event.message || "Stockfish review worker failed."));
    };

    this.worker.postMessage("uci");
  }

  async analyzePositions(
    fens: string[],
    options: ReviewAnalysisOptions,
  ): Promise<EngineScore[]> {
    if (this.disposed) {
      throw new Error("The review engine has been disposed.");
    }

    if (this.activeBatch) {
      throw new Error("A review analysis is already running.");
    }

    if (fens.length === 0) {
      return [];
    }

    const depth = Math.max(1, Math.min(24, Math.floor(options.depth)));

    if (fens.some((fen) => !fen.trim())) {
      throw new Error("Every review position must contain a valid FEN.");
    }

    await this.readyPromise;

    if (this.disposed) {
      throw new Error("The review engine has been disposed.");
    }

    return new Promise<EngineScore[]>((resolve, reject) => {
      this.activeBatch = {
        fens: [...fens],
        depth,
        index: 0,
        scores: [],
        onProgress: options.onProgress,
        resolve,
        reject,
      };

      this.startPosition();
    });
  }

  cancel(): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;

    this.disposed = true;
    this.worker.terminate();

    const error = new Error("Review analysis was cancelled.");
    batch?.reject(error);
    this.readyReject(error);
  }

  dispose(): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;
    this.disposed = true;

    this.worker.terminate();

    const error = new Error("The review engine has been disposed.");
    batch?.reject(error);
    this.readyReject(error);
  }

  private handleLine(line: string): void {
    if (this.disposed) return;

    if (/^Error:/i.test(line)) {
      this.fail(new Error(line));
      return;
    }

    if (line === "uciok") {
      this.worker.postMessage("isready");
      return;
    }

    if (line === "readyok") {
      this.ready = true;
      this.readyResolve();
      return;
    }

    const batch = this.activeBatch;
    if (!batch) return;

    if (line.startsWith("info ")) {
      const sideToMove = batch.fens[batch.index].split(/\s+/)[1] === "b" ? "b" : "w";
      const parsed = parseUciInfo(line, sideToMove);

      if (parsed && (!this.latestScore || parsed.depth >= this.latestScore.depth)) {
        this.latestScore = parsed;
      }

      return;
    }

    if (!line.startsWith("bestmove ")) return;

    const bestMove = line.match(/^bestmove (\S+)/)?.[1];

    if (!this.latestScore) {
      this.fail(new Error(`Stockfish returned no evaluation for ply ${batch.index + 1}.`));
      return;
    }

    const score: EngineScore = {
      ...this.latestScore,
      ...(bestMove && bestMove !== "(none)" ? { bestMove } : {}),
    };

    batch.scores.push(score);
    batch.index += 1;

    batch.onProgress?.({
      completed: batch.index,
      total: batch.fens.length,
      currentPly: batch.index,
    });

    this.latestScore = null;

    if (batch.index >= batch.fens.length) {
      this.activeBatch = null;
      batch.resolve(batch.scores);
      return;
    }

    this.startPosition();
  }

  private startPosition(): void {
    const batch = this.activeBatch;
    if (!batch || this.disposed || !this.ready) return;

    this.latestScore = null;

    const fen = batch.fens[batch.index];
    this.worker.postMessage(`position fen ${fen}`);
    this.worker.postMessage(`go depth ${batch.depth}`);
  }

  private fail(error: Error): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;
    this.disposed = true;

    this.worker.terminate();
    batch?.reject(error);
    this.readyReject(error);
  }
}

import { Chess } from "chess.js";
import { parsePgn } from "./pgn-import";

export type ReviewPosition = {
  ply: number;
  moveNumber: number;
  color: "w" | "b";
  san: string;
  beforeFen: string;
  afterFen: string;
  from: string;
  to: string;
  promotion?: string;
  flags: string;
};

export type ReviewGame = {
  headers: Record<string, string>;
  startingFen: string;
  finalFen: string;
  positions: ReviewPosition[];
};

export function reconstructReviewGame(pgn: string): ReviewGame {
  const imported = parsePgn(pgn);
  const game = new Chess(imported.lineBaseFen);
  const positions: ReviewPosition[] = [];

  imported.moves.forEach((san, index) => {
    const beforeFen = game.fen();
    const move = game.move(san);

    positions.push({
      ply: index + 1,
      moveNumber: Math.floor(index / 2) + 1,
      color: index % 2 === 0 ? "w" : "b",
      san: move.san,
      beforeFen,
      afterFen: game.fen(),
      from: move.from,
      to: move.to,
      ...(move.promotion ? { promotion: move.promotion } : {}),
      flags: move.flags,
    });
  });

  if (game.fen() !== imported.finalFen) {
    throw new Error("Review reconstruction does not match the imported PGN.");
  }

  return {
    headers: imported.headers,
    startingFen: imported.lineBaseFen,
    finalFen: game.fen(),
    positions,
  };
}

export function getReviewFenAtPly(review: ReviewGame, ply: number): string {
  const safePly = Math.max(0, Math.min(review.positions.length, Math.floor(ply)));

  return safePly === 0
    ? review.startingFen
    : review.positions[safePly - 1].afterFen;
}

function formatUciMove(fen: string, uci: string): string {
  const match = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
  if (!match) return uci;
  try {
    const position = new Chess(fen);
    const move = position.move({
      from: match[1] as Square,
      to: match[2] as Square,
      ...(match[3] ? { promotion: match[3] as "q" | "r" | "b" | "n" } : {}),
    });
    return move.san;
  } catch {
    return uci;
  }
}

function formatUciPv(fen: string, pv: string[]): string {
  const position = new Chess(fen);
  const san: string[] = [];
  for (const uci of pv) {
    const match = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
    if (!match) break;
    try {
      const move = position.move({
        from: match[1] as Square,
        to: match[2] as Square,
        ...(match[3] ? { promotion: match[3] as "q" | "r" | "b" | "n" } : {}),
      });
      san.push(move.san);
    } catch {
      break;
    }
  }
  return san.join(" ");
}

  async function addBook(file?: File) {
    if (!file) return;
    try {
      if (file.size > FILE_LIMIT) throw new Error("PDFs must be 100 MB or smaller.");
      const header = new Uint8Array(await file.slice(0, 5).arrayBuffer());
      if (new TextDecoder().decode(header) !== "%PDF-") throw new Error("The selected file does not contain a valid PDF header.");
      const now = Date.now();
      const created: BookRecord = {
        id: crypto.randomUUID(),
        title: file.name.replace(/\.pdf$/i, ""),
        file: file.slice(0, file.size, "application/pdf"),
        byteLength: file.size,
        lastOpenedAt: now,
        addedAt: now,
      };
      const blank = newSession(created.id);
      await Promise.all([putBook(created), putSession(blank), setActiveBookId(created.id)]);
      setBooks((current) => [created, ...current]);
      setSessionIndex((current) => ({ ...current, [created.id]: blank }));
      setBook(created);
      setSession(blank);
      setNotice("PDF added to this browser's local library.");
      setError("");
      if (fileInput.current) fileInput.current.value = "";
    } catch (cause) {
      setError(`Could not add PDF: ${messageOf(cause)}`);
    }
  }

      setNotice(`Removed "${item.title}" from your library.`);

    }
  }

  async function openTransfer(candidate: Candidate) {
    const canvas = canvasRef.current;
    if (!canvas || canvas.width === 0 || canvas.height === 0) {
      setError("The rendered page image is unavailable. Rerender the page before using this crop.");
      return;
    }
    let preview: HTMLCanvasElement;
    try {
      const crop = cropToPixels(candidate, canvas.width, canvas.height);
      preview = document.createElement("canvas");
      preview.width = crop.width;
      preview.height = crop.height;
      const context = preview.getContext("2d");
      if (!context) throw new Error("Could not prepare a crop preview in this browser.");
      context.drawImage(canvas, crop.x, crop.y, crop.width, crop.height, 0, 0, crop.width, crop.height);
    } catch (cause) {
      setError(`Could not create the crop preview: ${messageOf(cause)}`);
      return;
    }
    setTransferCandidate(candidate);
    setTransferPreview(preview.toDataURL("image/png"));
    setTransferFen("");
    setRecognitionResult(null);
        setRecognitionError("");
        const requestId = ++transferRequestRef.current;
    setRecognitionStatus("loading");
    try {
      const image = await new Promise<Blob>((resolve, reject) => {
        preview.toBlob((blob) => blob ? resolve(blob) : reject(new Error("Could not encode the selected crop.")), "image/png");
      });
      const form = new FormData();
      form.append("image", image, `board-crop-${candidate.id}.png`);
      const response = await fetch(`${API_BASE}/recognize`, { method: "POST", body: form });
      const result = await response.json() as RecognitionResult & { detail?: string };
      if (!response.ok) throw new Error(result.detail ?? `Recognition service returned HTTP ${response.status}.`);
      if (requestId !== transferRequestRef.current) return;
      setTransferFen(result.fen);
      setRecognitionResult(result);
      setRecognitionStatus("complete");
      // A successful scan immediately becomes the active analysis position.
      // Keep the review dialog open so the user can correct uncertain squares;
      // the explicit confirmation button remains available for corrected FENs.
      if (moveToPosition(result.fen)) {
        setNotice("Scanned position loaded automatically onto the analysis board. Review the diagram and correct any uncertain squares before continuing.");
      }
    } catch (cause) {
      if (requestId !== transferRequestRef.current) return;
      setRecognitionStatus("failed");
      setRecognitionError(messageOf(cause));
    }
  }

    setTransferFen("");
    setRecognitionStatus("idle");
    setRecognitionResult(null);

      }

    if (currentGame.isCheckmate()) {
      setGameEndOverlay({
        title: "Checkmate",
        message: `${currentGame.turn() === "w" ? "Black" : "White"} wins.`,
      });
    } else if (currentGame.isStalemate()) {
      setGameEndOverlay({
        title: "Stalemate",
        message: "The side to move has no legal moves, but its king is not in check. The game is a draw.",
      });
    } else if (currentGame.isDraw()) {
      let message = "The game is drawn.";
      if (currentGame.isThreefoldRepetition()) message = "Draw by threefold repetition.";
      else if (currentGame.isInsufficientMaterial()) message = "Draw by insufficient material.";
      else if (currentGame.isDrawByFiftyMoves()) message = "Draw by the fifty-move rule.";
      else if (currentGame.isDraw()) message = "Draw by a draw rule.";
      setGameEndOverlay({ title: "Draw", message });
    } else {
      setGameEndOverlay(null);
    }
  }

  function completePromotion(piece: "q" | "r" | "b" | "n") {
    if (!promotionRequest || !boardSession) return;
    const activeBoard = boardSession;
    try {
      const currentGame = new Chess(activeBoard.currentFen);
      const result = currentGame.move({
        from: promotionRequest.from,
        to: promotionRequest.to,
        promotion: piece,
      });
      const nextMoves = [...activeBoard.moves.slice(0, activeBoard.moveIndex), result.san];
      patchBoardSession({
        currentFen: currentGame.fen(),
        moves: nextMoves,
        moveIndex: nextMoves.length,
      });
      updateGameEndOverlay(currentGame);
      setLastMove({ from: promotionRequest.from, to: promotionRequest.to });
      setSelectedSquare(null);
      setPromotionRequest(null);
      setError("");
    } catch {
      setError("Unable to complete pawn promotion.");
      setPromotionRequest(null);
    }
  }

  function makeMove(from: Square, to: Square) {
    if (!boardSession) return;
    const activeBoard = boardSession;
    try {
      const currentGame = new Chess(activeBoard.currentFen);
      const mover = currentGame.get(from);
      const promote = mover?.type === "p" && (to.endsWith("8") || to.endsWith("1"));
      if (promote) {
        setPromotionRequest({ from, to });
        return;
      }
      const result = currentGame.move({ from, to });
      const nextMoves = [...activeBoard.moves.slice(0, activeBoard.moveIndex), result.san];
      if (workspaceMode === "opening-explorer") {
        const line = nextMoves
          .map((move, index) => index % 2 === 0 ? `${Math.floor(index / 2) + 1}. ${move}` : move)
          .join(" ");
        setOpeningExplorerInput(line);
        setOpeningExplorerPly(nextMoves.length);
      } else {
        patchBoardSession({
          currentFen: currentGame.fen(),
          moves: nextMoves,
          moveIndex: nextMoves.length,
        });
      }
      updateGameEndOverlay(currentGame);
      setLastMove({ from, to });
      setSelectedSquare(null);
      setError("");
    } catch {
      const piece = game.get(to);
      if (piece && piece.color === game.turn()) setSelectedSquare(to);
      else setSelectedSquare(null);
    }
  }

  function handleSquareClick(square: Square) {
    // A left-click removes the red highlight from that square.
    setRedSquares([]);

    if (!annotationMode && (userArrows.length > 0 || arrowStart !== null)) {
      setUserArrows([]);
      setArrowStart(null);
      setArrowHover(null);
      return;
    }
if (!boardSession) return;
    if (annotationMode) {
      if (!arrowStart) {
        setArrowStart(square);
        return;
      }
      if (arrowStart !== square) setUserArrows((arrows) => toggleArrow(arrows, { from: arrowStart, to: square }));
      setArrowStart(null);
      setArrowHover(null);
      return;
    }
    if (selectedSquare && legalTargets.has(square)) {
      makeMove(selectedSquare, square);
      return;
    }
    const piece = game.get(square);
    if (piece?.color === game.turn()) setSelectedSquare(square);
    else setSelectedSquare(null);
  }

  async function detectBoards() {
    if (detecting) {
      stopBoardScan();
      return;
    }

    if (!canvasRef.current || !session) return;
    if (!API_BASE) {
      setError("Board detection API is not configured. Configure VITE_API_BASE_URL and start the optional CV service.");
      return;
    }

    scanAbortControllerRef.current?.abort();
    const abortController = new AbortController();
    scanAbortControllerRef.current = abortController;
    const requestId = ++scanRequestIdRef.current;

    setDetecting(true);
    setError("");
    setNotice("Scanning current page...");
    patchSession((current) => ({
      ...current,
      candidatesByPage: {
        ...current.candidatesByPage,
        [String(current.page)]: [],
      },
      selectedDiagramId: null,
    }));

    try {
      const image = await new Promise<Blob>((resolve, reject) => {
        canvasRef.current?.toBlob(
          (blob) => blob
            ? resolve(blob)
            : reject(new Error("Could not encode rendered PDF page.")),
          "image/png",
        );
      });

      if (requestId !== scanRequestIdRef.current) return;

      const form = new FormData();
      form.append("image", image, `page-${session.page}.png`);

      const response = await fetch(`${API_BASE}/detect`, {
        method: "POST",
        body: form,
        signal: abortController.signal,
      });

      if (!response.ok) {
        throw new Error(`Detector returned HTTP ${response.status}.`);
      }

      const data = await response.json() as { candidates?: CandidateInput[] };
      const candidates = (data.candidates ?? []).map(
        (item, index) => normalizedCandidate({ ...item, source: "automatic" }, index),
      );

      if (requestId !== scanRequestIdRef.current) return;

      patchSession((current) => ({
        ...current,
        candidatesByPage: {
          ...current.candidatesByPage,
          [String(current.page)]: candidates,
        },
        selectedDiagramId: null,
      }));

      setNotice(
        candidates.length
          ? `Found ${candidates.length} board candidate${candidates.length === 1 ? "" : "s"}. Select a candidate to recognize its chess position and load it onto the analysis board.`
          : "No board candidates found.",
      );
    } catch (cause) {
      if (requestId !== scanRequestIdRef.current) return;

      if (abortController.signal.aborted) {
        setNotice("Scan cancelled.");
      } else {
        setError(`Board detection failed: ${messageOf(cause)}`);
      }
    } finally {
      if (requestId === scanRequestIdRef.current) {
        scanAbortControllerRef.current = null;
        setDetecting(false);
      }
    }
  }

  function moveRegion(event: React.PointerEvent<HTMLDivElement>) {
    const edit = cropEditStartRef.current;
    if (edit) {
      const point = pointerPosition(event);
      if (!point || !session) return;
      const next = editCrop(edit.candidate, edit.handle, point.x - edit.point.x, point.y - edit.point.y);
      patchSession((current) => {
        const pageKey = String(current.page);
        return {
          ...current,
          candidatesByPage: {
            ...current.candidatesByPage,
            [pageKey]: (current.candidatesByPage[pageKey] ?? []).map((candidate) =>
              candidate.id === edit.candidate.id ? { ...candidate, ...next } : candidate),
          },
        };
      });
      return;
    }
    if (!dragStartRef.current) return;
    const point = pointerPosition(event);
    if (point) {
      const selection = { ...dragStartRef.current, x2: point.x, y2: point.y };
      dragCurrentRef.current = selection;
      setDragSelection(selection);
    }
  }

  function finishRegion() {
    if (cropEditStartRef.current) {
      cropEditStartRef.current = null;
      return;
    }
    const selection = dragCurrentRef.current;
    if (!dragStartRef.current || !selection || !session) return;
    const x = Math.min(selection.x, selection.x2);
    const y = Math.min(selection.y, selection.y2);
    const width = Math.abs(selection.x2 - selection.x);
    const height = Math.abs(selection.y2 - selection.y);
    dragStartRef.current = null;
    dragCurrentRef.current = null;
    setDragSelection(null);
    if (width < 0.025 || height < 0.025) return;
    const candidate: Candidate = {
      id: `manual-${crypto.randomUUID()}`,
      x, y, width, height,
      relativeScore: 1,
      source: "manual",
    };
    patchSession((current) => ({
      ...current,
      candidatesByPage: {
        ...current.candidatesByPage,
        [String(current.page)]: [...(current.candidatesByPage[String(current.page)] ?? []), candidate],
      },
      selectedDiagramId: candidate.id,
    }));
    setManualMode(false);
    setNotice("Manual crop selected. Adjust its boundary, then use it on the board when ready.");
  }

      patchSession({ zoom: Math.max(0.45, Math.min(1.8, scale)) });
    } catch (cause) {
      setError(`Could not fit PDF page: ${messageOf(cause)}`);
    }
  }

function formatSize(bytes: number): string {
  if (bytes < 1024 * 1024) return `${Math.max(1, Math.round(bytes / 1024))} KB`;
  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
}
