import { useEffect, useRef, useState } from "react";
import { Chess } from "chess.js";
import {
  ArrowLeft, ArrowRight, BarChart3, FileUp, LoaderCircle,
  Pause, Play, RotateCcw, Target, Trophy, X
} from "lucide-react";
import { reconstructReviewGame, getReviewFenAtPly, type ReviewGame } from "./review-pgn";
import { ReviewEngine, type ReviewProgress } from "./review-engine";
import { analyzeReviewPositions, type ReviewAnalysis, type ReviewedMove } from "./review-analysis";

type ReviewTab = "overview" | "moves";
type MoveFilter = "All" | "Best" | "Excellent" | "Good" | "Inaccuracy" | "Mistake" | "Blunder";
type ImportMode = "pgn" | "fen" | "chesscom" | "lichess";
type ImportedGame = {
  id: string;
  label: string;
  detail: string;
  pgn: string;
};

const FILTERS: MoveFilter[] = ["All", "Best", "Excellent", "Good", "Inaccuracy", "Mistake", "Blunder"];

function evaluationText(score?: { kind: "cp" | "mate"; value: number }) {
  if (!score) return "â€”";
  if (score.kind === "mate") return `${score.value > 0 ? "+" : ""}#${Math.abs(score.value)}`;
  const value = score.value / 100;
  return `${value > 0 ? "+" : ""}${value.toFixed(2)}`;
}

function resultText(result?: string) {
  if (result === "1-0") return "White won";
  if (result === "0-1") return "Black won";
  if (result === "1/2-1/2") return "Draw";
  return "Result not recorded";
}

function metadata(value?: string) {
  return value && value !== "*" && !/^[?.]+$/.test(value) ? value : "";
}

function classificationClass(value: string) {
  return value.toLowerCase().replace(/\s+/g, "-");
}

export function GameReview() {
  const [pgnText, setPgnText] = useState("");
  const [importMode, setImportMode] = useState<ImportMode>("pgn");
  const [importOpen, setImportOpen] = useState(true);
  const [fenText, setFenText] = useState("");
  const [username, setUsername] = useState("");
  const [importedGames, setImportedGames] = useState<ImportedGame[]>([]);
  const [isImporting, setIsImporting] = useState(false);
  const [isFenPosition, setIsFenPosition] = useState(false);
  const EMPTY_REVIEW_BOARD: ReviewGame = { headers: {}, startingFen: new Chess().fen(), finalFen: new Chess().fen(), positions: [] };
  const [review, setReview] = useState<ReviewGame | null>(EMPTY_REVIEW_BOARD);
  const [analysis, setAnalysis] = useState<ReviewAnalysis | null>(null);
  const [activePly, setActivePly] = useState(0);
  const [isPlaying, setIsPlaying] = useState(false);
  const [tab, setTab] = useState<ReviewTab>("overview");
  const [filter, setFilter] = useState<MoveFilter>("All");
  const [depth, setDepth] = useState(14);
  const [progress, setProgress] = useState<ReviewProgress | null>(null);
  const [status, setStatus] = useState<"empty" | "ready" | "running" | "complete" | "error">("empty");
  const [error, setError] = useState("");
  const fileInput = useRef<HTMLInputElement>(null);
  const engineRef = useRef<ReviewEngine | null>(null);
  const runToken = useRef(0);

  useEffect(() => {
    return () => {
      runToken.current += 1;
      engineRef.current?.dispose();
      engineRef.current = null;
    };
  }, []);

  useEffect(() => {
    if (!review || !isPlaying) return;
    const timer = window.setInterval(() => {
      setActivePly(ply => Math.min(review.positions.length, ply + 1));
    }, 750);
    return () => window.clearInterval(timer);
  }, [isPlaying, review]);

  useEffect(() => {
    if (isPlaying && activePly >= (review?.positions.length ?? 0)) setIsPlaying(false);
  }, [activePly, isPlaying, review?.positions.length]);

  function loadPgn(text: string) {
    try {
      const parsed = reconstructReviewGame(text);
      runToken.current += 1;
      engineRef.current?.dispose();
      engineRef.current = null;
      setIsPlaying(false);
      setPgnText(text);
      setIsFenPosition(false);
      setReview(parsed);
      setAnalysis(null);
      setActivePly(0);
      setProgress(null);
      setError("");
      setStatus("ready");
      setTab("overview");
      setFilter("All");
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "Unable to read this PGN.");
      setStatus("error");
    }
  }

  async function handleFile(event: React.ChangeEvent<HTMLInputElement>) {
    const file = event.target.files?.[0];
    event.target.value = "";
    if (!file) return;
    try {
      const text = await file.text();
      loadPgn(text);
    } catch {
      setError("Unable to read the selected PGN file.");
      setStatus("error");
    }
  }

  function loadFenPosition() {
    try {
      const fen = fenText.trim();
      if (!fen) throw new Error("Paste a FEN position first.");

      const chess = new Chess(fen);
      const canonicalFen = chess.fen();

      runToken.current += 1;
      engineRef.current?.dispose();
      engineRef.current = null;
      setIsPlaying(false);
      setReview({
        headers: {},
        startingFen: canonicalFen,
        finalFen: canonicalFen,
        positions: []
      });
      setAnalysis(null);
      setActivePly(0);
      setProgress(null);
      setError("");
      setIsFenPosition(true);
      setStatus("ready");
      setTab("overview");
      setFilter("All");
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "Invalid FEN position.");
      setStatus("error");
    }
  }

  async function fetchChessComGames() {
    const player = username.trim().toLowerCase();
    if (!player) {
      setError("Enter a Chess.com username.");
      return;
    }

    setIsImporting(true);
    setError("");
    setImportedGames([]);

    try {
      const archiveResponse = await fetch(
        `https://api.chess.com/pub/player/${encodeURIComponent(player)}/games/archives`
      );
      if (!archiveResponse.ok) {
        throw new Error(archiveResponse.status === 404
          ? "Chess.com username was not found."
          : `Chess.com request failed (${archiveResponse.status}).`);
      }

      const archiveData = await archiveResponse.json();
      const archives: string[] = Array.isArray(archiveData.archives)
        ? archiveData.archives.slice(-3).reverse()
        : [];

      if (!archives.length) throw new Error("No public Chess.com game archives were found.");

      const collected: ImportedGame[] = [];

      for (const archiveUrl of archives) {
        const response = await fetch(archiveUrl);
        if (!response.ok) continue;
        const data = await response.json();

        for (const [index, game] of (data.games ?? []).entries()) {
          if (!game.pgn) continue;
          const white = game.white?.username ?? "White";
          const black = game.black?.username ?? "Black";
          const date = game.end_time
            ? new Date(game.end_time * 1000).toLocaleDateString()
            : "Date unavailable";
          const result = game.white?.result === "win"
            ? `${white} won`
            : game.black?.result === "win"
              ? `${black} won`
              : game.white?.result === "agreed" || game.black?.result === "agreed"
                ? "Draw"
                : "Game";

          collected.push({
            id: String(game.url ?? `${archiveUrl}-${index}`),
            label: `${white} vs ${black}`,
            detail: `${date} Â· ${result} Â· ${game.time_class ?? "Game"}`,
            pgn: game.pgn
          });
        }
      }

      setImportedGames(collected.slice(0, 50));
      if (!collected.length) setError("No PGN games were returned for this Chess.com account.");
    } catch (cause) {
      setError(cause instanceof Error
        ? cause.message
        : "Could not retrieve Chess.com games. Try PGN import instead.");
    } finally {
      setIsImporting(false);
    }
  }

  async function fetchLichessGames() {
    const player = username.trim();
    if (!player) {
      setError("Enter a Lichess username.");
      return;
    }

    setIsImporting(true);
    setError("");
    setImportedGames([]);

    try {
      const response = await fetch(
        `https://lichess.org/api/games/user/${encodeURIComponent(player)}?max=25&pgnInJson=true`,
        { headers: { Accept: "application/x-ndjson" } }
      );

      if (!response.ok) {
        throw new Error(response.status === 404
          ? "Lichess username was not found."
          : `Lichess request failed (${response.status}).`);
      }

      const body = await response.text();
      const games: ImportedGame[] = body
        .split(/\r?\n/)
        .filter(Boolean)
        .flatMap((line, index) => {
          try {
            const game = JSON.parse(line);
            if (!game.pgn) return [];

            const white = game.players?.white?.user?.name ?? "White";
            const black = game.players?.black?.user?.name ?? "Black";
            const date = game.createdAt
              ? new Date(game.createdAt).toLocaleDateString()
              : "Date unavailable";
            const winner = game.winner === "white"
              ? `${white} won`
              : game.winner === "black"
                ? `${black} won`
                : "Draw";

            return [{
              id: String(game.id ?? index),
              label: `${white} vs ${black}`,
              detail: `${date} Â· ${winner} Â· ${game.speed ?? "Game"}`,
              pgn: game.pgn
            }];
          } catch {
            return [];
          }
        });

      setImportedGames(games);
      if (!games.length) setError("No PGN games were returned for this Lichess account.");
    } catch (cause) {
      setError(cause instanceof Error
        ? cause.message
        : "Could not retrieve Lichess games. Try PGN import instead.");
    } finally {
      setIsImporting(false);
    }
  }

  function selectImportedGame(game: ImportedGame) {
    loadPgn(game.pgn);
    setImportOpen(false);
    setImportedGames([]);
  }

  async function startAnalysis() {
    if (!review || review.positions.length === 0 || isFenPosition || status === "running") return;

    const token = ++runToken.current;
    engineRef.current?.dispose();
    engineRef.current = null;
    setAnalysis(null);
    setProgress({ completed: 0, total: review.positions.length + 1, currentPly: 0 });
    setError("");
    setStatus("running");

    try {
      const workerUrl = `${import.meta.env.BASE_URL}stockfish/stockfish-19-lite-single.js`;
      const engine = new ReviewEngine(workerUrl);
      engineRef.current = engine;

      const result = await analyzeReviewPositions(review, engine, {
        depth,
        onProgress: value => {
          if (runToken.current === token) setProgress(value);
        }
      });

      if (runToken.current !== token) return;
      setAnalysis(result);
      setStatus("complete");
      setProgress(null);
      engine.dispose();
      engineRef.current = null;
    } catch (cause) {
      if (runToken.current !== token) return;
      engineRef.current = null;
      const message = cause instanceof Error ? cause.message : "Stockfish analysis failed.";
      if (/cancelled|disposed/i.test(message)) {
        setStatus("ready");
      } else {
        setStatus("error");
        setError(message);
      }
      setProgress(null);
    }
  }

  function cancelAnalysis() {
    runToken.current += 1;
    engineRef.current?.cancel();
    engineRef.current = null;
    setProgress(null);
    setStatus(review ? "ready" : "empty");
  }

  function resetReview() {
    cancelAnalysis();
    setIsPlaying(false);
    setReview(EMPTY_REVIEW_BOARD);
    setAnalysis(null);
    setPgnText("");
    setFenText("");
    setIsFenPosition(false);
    setImportedGames([]);
    setImportOpen(true);
    setActivePly(0);
    setError("");
    setStatus("empty");
  }

  const currentFen = review ? getReviewFenAtPly(review, activePly) : new Chess().fen();
  const board = currentFen ? new Chess(currentFen).board() : [];
  const currentMove = review?.positions[activePly - 1];
  const reviewedMove = analysis?.moves[activePly - 1];

  const filteredMoves = analysis?.moves.filter(move =>
    filter === "All" || move.classification === filter
  ) ?? [];

  const counts = analysis?.moves.reduce<Record<string, number>>((acc, move) => {
    acc[move.classification] = (acc[move.classification] ?? 0) + 1;
    return acc;
  }, {}) ?? {};

  const keyMoments = analysis?.moves.filter(move => move.evaluationLossCp > 80) ?? [];
  const whiteName = metadata(review?.headers.White) || "White";
  const blackName = metadata(review?.headers.Black) || "Black";
  const gameResult = metadata(review?.headers.Result) || "â€”";

  function goTo(ply: number) {
    if (!review) return;
    setIsPlaying(false);
    setActivePly(Math.max(0, Math.min(review.positions.length, ply)));
  }

  return (
    <main className={`game-review-page gr-live ${review ? "has-review" : "is-empty"}`}>
      <input ref={fileInput} type="file" accept=".pgn,text/plain" hidden onChange={handleFile} />

      {false && !review && (
        <>
          <header className="gr-live-header">
            <div>
              <div className="gr-live-eyebrow">CHESS ANALYSIS / GAME REVIEW</div>
              <h1>Game Review</h1>
              <p>Import a game, analyse every position and inspect your decisions.</p>
            </div>
            <button className="gr-live-button primary" onClick={() => fileInput.current?.click()}>
              <FileUp size={16} /> Import PGN
            </button>
          </header>

          {error && (
            <div className="gr-live-error" role="alert">
              <span>{error}</span>
              <button onClick={() => setError("")} aria-label="Dismiss error"><X size={16} /></button>
            </div>
          )}

          <section className="gr-live-import">
            <div className="gr-live-import-icon"><FileUp size={26} /></div>
            <h2>Import a chess game</h2>
            <p>Choose a PGN file or paste the PGN notation below.</p>
            <textarea
              value={pgnText}
              onChange={event => setPgnText(event.target.value)}
              placeholder={'[Event "Casual Game"]\n[White "Player 1"]\n[Black "Player 2"]\n\n1. e4 e5 2. Nf3 Nc6 *'}
              rows={9}
              spellCheck={false}
            />
            <button className="gr-live-button primary" disabled={!pgnText.trim()} onClick={() => loadPgn(pgnText)}>
              Load PGN
            </button>
          </section>
        </>
      )}

      {review && (
        <section className="gr-live-review-shell">
          <section className="gr-live-board-column">
            <section className="gr-live-game-card">
              <div className="gr-live-player">
                <span className="gr-live-avatar white">W</span>
                <div><strong>{whiteName}</strong><span>White {metadata(review.headers.WhiteElo) && `Â· ${review.headers.WhiteElo}`}</span></div>
              </div>
              <strong className="gr-live-result">{gameResult}</strong>
              <div className="gr-live-player black">
                <span className="gr-live-avatar black">B</span>
                <div><strong>{blackName}</strong><span>Black {metadata(review.headers.BlackElo) && `Â· ${review.headers.BlackElo}`}</span></div>
              </div>
              <div className="gr-live-meta">
                {[metadata(review.headers.Event), metadata(review.headers.Site), metadata(review.headers.Date), metadata(review.headers.Round), metadata(review.headers.ECO), metadata(review.headers.Opening), metadata(review.headers.TimeControl)].filter(Boolean).join(" Â· ") || "PGN game"}
                <span>{review.positions.length} plies</span>
                <span>{resultText(review.headers.Result)}</span>
              </div>
            </section>

            <div className="gr-live-board-wrap">
              <div className="gr-live-ranks">{[8,7,6,5,4,3,2,1].map(rank => <span key={rank}>{rank}</span>)}</div>
              <div className="gr-live-board">
                {board.flatMap((row, rowIndex) => row.map((piece, colIndex) => {
                  const square = `${"abcdefgh"[colIndex]}${8 - rowIndex}`;
                  const highlighted = currentMove && (currentMove.from === square || currentMove.to === square);
                  return (
                    <div key={square} className={`gr-live-square ${(rowIndex + colIndex) % 2 ? "dark" : "light"} ${highlighted ? "highlight" : ""}`}>
                      {piece && <img draggable={false} alt="" src={`${import.meta.env.BASE_URL}assets/pieces/standalone/${piece.color}${piece.type}.svg`} />}
                    </div>
                  );
                }))}
              </div>
              <div className="gr-live-files">{["a","b","c","d","e","f","g","h"].map(file => <span key={file}>{file}</span>)}</div>
            </div>
            <div className="gr-live-board-status">
              <span>{activePly === 0 ? "Starting position" : `${currentMove?.moveNumber ?? ""}${currentMove?.color === "w" ? "." : "..."} ${currentMove?.san ?? ""}`}</span>
              {reviewedMove && <span className="gr-live-board-eval">{reviewedMove.classification} Â· {evaluationText(reviewedMove.after)} Â· {reviewedMove.evaluationLossCp} cp</span>}
              <span>{activePly} / {review.positions.length}</span>
            </div>
          </section>

          <aside className="gr-live-analysis-panel">
            <header className="gr-live-review-header">
              <h1>Game Review</h1>
              <div className="gr-live-actions">
                <button className="gr-live-button secondary" onClick={resetReview}>
                  <RotateCcw size={15} /> New review
                </button>
                <button className="gr-live-button primary" onClick={() => fileInput.current?.click()}>
                  <FileUp size={16} /> Import PGN
                </button>
              </div>
            </header>

            {error && (
              <div className="gr-live-error" role="alert">
                <span>{error}</span>
                <button onClick={() => setError("")} aria-label="Dismiss error"><X size={16} /></button>
              </div>
            )}

            <section className="gr-live-import-panel">
              <div className="gr-live-import-heading">
                <div>
                  <strong>Load a game or position</strong>
                  <span>Import a game or analyse a FEN position</span>
                </div>
                <button
                  className="gr-live-button secondary"
                  onClick={() => setImportOpen(value => !value)}
                >
                  {importOpen ? "Hide import" : "Import game"}
                </button>
              </div>

              {importOpen && (
                <>
                  <div className="gr-live-import-tabs">
                    {([
                      ["pgn", "PGN"],
                      ["fen", "FEN"],
                      ["chesscom", "Chess.com"],
                      ["lichess", "Lichess"]
                    ] as [ImportMode, string][]).map(([mode, label]) => (
                      <button
                        key={mode}
                        className={importMode === mode ? "active" : ""}
                        onClick={() => {
                          setImportMode(mode);
                          setError("");
                          setImportedGames([]);
                        }}
                      >
                        {label}
                      </button>
                    ))}
                  </div>

                  {importMode === "pgn" && (
                    <div className="gr-live-import-content">
                      <textarea
                        value={pgnText}
                        onChange={event => setPgnText(event.target.value)}
                        placeholder={'[Event "Casual Game"]\n[White "Player 1"]\n[Black "Player 2"]\n\n1. e4 e5 2. Nf3 Nc6 *'}
                        rows={5}
                        spellCheck={false}
                      />
                      <div className="gr-live-import-actions">
                        <button className="gr-live-button secondary" onClick={() => fileInput.current?.click()}>
                          <FileUp size={15} /> Choose PGN file
                        </button>
                        <button
                          className="gr-live-button primary"
                          disabled={!pgnText.trim()}
                          onClick={() => { loadPgn(pgnText); setImportOpen(false); }}
                        >
                          Load PGN
                        </button>
                      </div>
                    </div>
                  )}

                  {importMode === "fen" && (
                    <div className="gr-live-import-content">
                      <textarea
                        value={fenText}
                        onChange={event => setFenText(event.target.value)}
                        placeholder="Paste a FEN position..."
                        rows={3}
                        spellCheck={false}
                      />
                      <button
                        className="gr-live-button primary"
                        disabled={!fenText.trim()}
                        onClick={() => { loadFenPosition(); setImportOpen(false); }}
                      >
                        Load position
                      </button>
                    </div>
                  )}

                  {(importMode === "chesscom" || importMode === "lichess") && (
                    <div className="gr-live-import-content">
                      <div className="gr-live-username-row">
                        <input
                          value={username}
                          onChange={event => setUsername(event.target.value)}
                          onKeyDown={event => {
                            if (event.key === "Enter") {
                              importMode === "chesscom"
                                ? void fetchChessComGames()
                                : void fetchLichessGames();
                            }
                          }}
                          placeholder={`${importMode === "chesscom" ? "Chess.com" : "Lichess"} username`}
                        />
                        <button
                          className="gr-live-button primary"
                          disabled={isImporting || !username.trim()}
                          onClick={() => importMode === "chesscom"
                            ? void fetchChessComGames()
                            : void fetchLichessGames()}
                        >
                          {isImporting ? <LoaderCircle className="gr-live-spinner" size={15} /> : "Find games"}
                        </button>
                      </div>
                      <p className="gr-live-muted">
                        {importMode === "chesscom"
                          ? "Fetches games from your three latest public monthly archives."
                          : "Fetches up to 25 recent public games."}
                      </p>
                      {importedGames.length > 0 && (
                        <div className="gr-live-import-game-list">
                          {importedGames.map(game => (
                            <button key={game.id} onClick={() => selectImportedGame(game)}>
                              <strong>{game.label}</strong>
                              <span>{game.detail}</span>
                            </button>
                          ))}
                        </div>
                      )}
                    </div>
                  )}
                </>
              )}
            </section>

            <section className="gr-live-toolbar">
              <div className="gr-live-depth">
                <label htmlFor="gr-live-depth">Analysis depth</label>
                <select id="gr-live-depth" value={depth} disabled={status === "running"} onChange={event => setDepth(Number(event.target.value))}>
                  {[10, 12, 14, 16, 18, 20, 22, 24].map(value => <option key={value} value={value}>{value}</option>)}
                </select>
              </div>
              {status === "running" ? (
                <button className="gr-live-button danger" onClick={cancelAnalysis}><Pause size={15} /> Cancel analysis</button>
              ) : (
                <button className="gr-live-button primary" onClick={startAnalysis}>
                  <BarChart3 size={16} /> {analysis ? "Analyse again" : "Start analysis"}
                </button>
              )}
            </section>

            {status === "running" && progress && (
              <section className="gr-live-progress">
                <div><LoaderCircle className="gr-live-spinner" size={16} /> Analysing position {progress.completed} of {progress.total}</div>
                <div className="gr-live-progress-track"><i style={{ width: `${progress.total ? progress.completed / progress.total * 100 : 0}%` }} /></div>
                <span>Stockfish depth {depth}. You can cancel at any time.</span>
              </section>
            )}

            {analysis && (
              <section className="gr-live-stats">
                <article><Target size={17} /><strong>{analysis.moves.length}</strong><span>Moves analysed</span></article>
                <article><Trophy size={17} /><strong>{counts.Best ?? 0}</strong><span>Best moves</span></article>
                <article><BarChart3 size={17} /><strong>{counts.Inaccuracy ?? 0}</strong><span>Inaccuracies</span></article>
                <article><Target size={17} /><strong>{(counts.Mistake ?? 0) + (counts.Blunder ?? 0)}</strong><span>Mistakes and blunders</span></article>
              </section>
            )}

            <nav className="gr-live-tabs">
              <button className={tab === "overview" ? "active" : ""} onClick={() => setTab("overview")}>Overview</button>
              <button className={tab === "moves" ? "active" : ""} onClick={() => setTab("moves")}>Move by move</button>
            </nav>

            <div className="gr-live-review-content">
              {tab === "overview" ? (
                <div className="gr-live-overview">
                  <section className="gr-live-panel">
                    <div className="gr-live-panel-heading"><div><h2>Key moments</h2><span>Moves with the largest evaluation losses</span></div></div>
                    {!analysis && <p className="gr-live-muted">Run Stockfish analysis to discover key moments in this game.</p>}
                    {analysis && keyMoments.length === 0 && <p className="gr-live-muted">No moves crossed the current 80 cp key-moment threshold.</p>}
                    {keyMoments.map(move => (
                      <button className="gr-live-moment" key={move.ply} onClick={() => goTo(move.ply)}>
                        <span className={`gr-live-tag ${classificationClass(move.classification)}`}>{move.classification}</span>
                        <strong>{move.moveNumber}{move.color === "w" ? "." : "..."} {move.san}</strong>
                        <span>{move.evaluationLossCp} cp loss</span>
                        <ArrowRight size={15} />
                      </button>
                    ))}
                  </section>

function formatUciMove(fen: string, uci: string): string {
  const match = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
  if (!match) return uci;
  try {
    const position = new Chess(fen);
    const move = position.move({
      from: match[1] as Square,
      to: match[2] as Square,
      ...(match[3] ? { promotion: match[3] as "q" | "r" | "b" | "n" } : {}),
    });
    return move.san;
  } catch {
    return uci;
  }
}

function formatUciPv(fen: string, pv: string[]): string {
  const position = new Chess(fen);
  const san: string[] = [];
  for (const uci of pv) {
    const match = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
    if (!match) break;
    try {
      const move = position.move({
        from: match[1] as Square,
        to: match[2] as Square,
        ...(match[3] ? { promotion: match[3] as "q" | "r" | "b" | "n" } : {}),
      });
      san.push(move.san);
    } catch {
      break;
    }
  }
  return san.join(" ");
}

function readableDate(timestamp: number): string {
  if (!timestamp) return "Not opened";
  const elapsed = Date.now() - timestamp;
  if (elapsed < 60_000) return "Just now";
  if (elapsed < 3_600_000) return `${Math.floor(elapsed / 60_000)}m ago`;
  if (elapsed < 86_400_000) return `${Math.floor(elapsed / 3_600_000)}h ago`;
  return new Date(timestamp).toLocaleDateString();
}

            setEngineMessage(message || "Stockfish could not start. Check the engine assets and browser worker support.");
            setEngineEnabled(false);
            setEngineScore(null);
            engineRef.current?.dispose();
            engineRef.current = null;
          }
        },
        setEngineScore,
      );
      engineRef.current = client;
      setEngineEnabled(true);
    } catch (cause) {
      setEngineStatus("error");
      setEngineMessage(`Stockfish could not start: ${messageOf(cause)}`);
      setEngineEnabled(false);
    }
  }

  function patchSession(change: Partial<BookSession> | ((current: BookSession) => BookSession)) {
    setSession((current) => {
      if (!current) return current;
      return typeof change === "function" ? change(current) : { ...current, ...change };
    });
  }

  function patchBoardSession(change: Partial<StudyBoardState>) {
    if (workspaceMode === "opening-explorer") {
      return;
    }

    if (workspaceMode === "analysis") {
      setAnalysisBoard((current) => ({ ...current, ...change }));
    } else if (workspaceMode === "opening-prep") {
      setOpeningPrepBoard((current) => ({ ...current, ...change }));
    } else if (studyMode) {
      setStudyBoard((current) => current ? { ...current, ...change } : current);
    } else {
      patchSession(change);
    }
  }
  async function toggleStudyMode() {
    if (!studyMode) {
      if (!session) {
        setError("Open a book before entering study mode.");
        return;
      }
      setStudyBoard({
        lineBaseFen: session.lineBaseFen,
        currentFen: session.currentFen,
        moves: [...session.moves],
        moveIndex: session.moveIndex,
        flipped: session.flipped,
        coordinates: session.coordinates,
      });
      setStudyMode(true);
    } else {
      // Preserve the latest Focus Mode board when returning to normal mode.
      if (studyBoard) {
        patchSession({
          lineBaseFen: studyBoard.lineBaseFen,
          currentFen: studyBoard.currentFen,
          moves: [...studyBoard.moves],
          moveIndex: studyBoard.moveIndex,
          flipped: studyBoard.flipped,
          coordinates: studyBoard.coordinates,
        });
      }
      setStudyMode(false);
    }
    setLastMove(null);
    setSelectedSquare(null);
    setPromotionRequest(null);

    try {
      if (!document.fullscreenElement) {
        await document.documentElement.requestFullscreen();
      } else {
        await document.exitFullscreen();
      }
    } catch {
      // Focus layout still works when browser fullscreen is blocked.
    }
}

  async function openBook(item: BookRecord) {
    try {
      const existing = sessionIndex[item.id] ?? await getSession(item.id);
      const nextSession = existing ?? newSession(item.id);
      const opened = { ...item, lastOpenedAt: Date.now() };
      await Promise.all([putBook(opened), putSession(nextSession), setActiveBookId(opened.id)]);
      setBooks((current) => [...current.filter((candidate) => candidate.id !== opened.id), opened].sort((a, b) => b.lastOpenedAt - a.lastOpenedAt));
      setSessionIndex((current) => ({ ...current, [opened.id]: nextSession }));
      setBook(opened);
      setSession(nextSession);
      setNotice(`Opened ${opened.title}`);
      setError("");
      setSelectedSquare(null);
    } catch (cause) {
      setError(`Could not open book: ${messageOf(cause)}`);
    }
  }

  async function addBook(file?: File) {
    if (!file) return;
    try {
      if (file.size > FILE_LIMIT) throw new Error("PDFs must be 100 MB or smaller.");
      const header = new Uint8Array(await file.slice(0, 5).arrayBuffer());
      if (new TextDecoder().decode(header) !== "%PDF-") throw new Error("The selected file does not contain a valid PDF header.");
      const now = Date.now();
      const created: BookRecord = {
        id: crypto.randomUUID(),
        title: file.name.replace(/\.pdf$/i, ""),
        file: file.slice(0, file.size, "application/pdf"),
        byteLength: file.size,
        lastOpenedAt: now,
        addedAt: now,
      };
      const blank = newSession(created.id);
      await Promise.all([putBook(created), putSession(blank), setActiveBookId(created.id)]);
      setBooks((current) => [created, ...current]);
      setSessionIndex((current) => ({ ...current, [created.id]: blank }));
      setBook(created);
      setSession(blank);
      setNotice("PDF added to this browser's local library.");
      setError("");
      if (fileInput.current) fileInput.current.value = "";
    } catch (cause) {
      setError(`Could not add PDF: ${messageOf(cause)}`);
    }
  }

  async function deleteBook(item: BookRecord) {
    if (!window.confirm(`Delete "${item.title}" and its saved reading progress from this browser? This cannot be undone.`)) return;

    try {
      await removeBook(item.id);
      await removeSession(item.id);

      setBooks((current) => current.filter((entry) => entry.id !== item.id));
      setSessionIndex((current) => {
        const next = { ...current };
        delete next[item.id];
        return next;
      });

      if (book?.id === item.id) {
        setBook(null);
        setSession(null);
        setPdf(null);
      }

      setNotice(`Removed "${item.title}" from your library.`);
      setError("");
    } catch (cause) {
      setError(`Could not remove book: ${messageOf(cause)}`);
    }
  }
  async function renameBook(item: BookRecord) {
    const title = window.prompt("Book title", item.title)?.trim();
    if (!title || title === item.title) return;
    const renamed = { ...item, title };
    try {
      await putBook(renamed);
      setBooks((current) => current.map((entry) => entry.id === item.id ? renamed : entry));
      if (book?.id === item.id) setBook(renamed);
    } catch (cause) {
      setError(`Could not rename book: ${messageOf(cause)}`);
    }
  }

  async function openTransfer(candidate: Candidate) {
    const canvas = canvasRef.current;
    if (!canvas || canvas.width === 0 || canvas.height === 0) {
      setError("The rendered page image is unavailable. Rerender the page before using this crop.");
      return;
    }

    try {
      const crop = cropToPixels(candidate, canvas.width, canvas.height);
      preview = document.createElement("canvas");
      preview.width = crop.width;
      preview.height = crop.height;
      const context = preview.getContext("2d");
      if (!context) throw new Error("Could not prepare a crop preview in this browser.");
      context.drawImage(canvas, crop.x, crop.y, crop.width, crop.height, 0, 0, crop.width, crop.height);
    } catch (cause) {
      setError(`Could not create the crop preview: ${messageOf(cause)}`);
      return;
    }
    setTransferCandidate(candidate);
    setTransferPreview(preview.toDataURL("image/png"));
    setTransferFen("");
    setRecognitionResult(null);
        setRecognitionError("");
        const requestId = ++transferRequestRef.current;
    setRecognitionStatus("loading");
    try {
      const image = await new Promise<Blob>((resolve, reject) => {
        preview.toBlob((blob) => blob ? resolve(blob) : reject(new Error("Could not encode the selected crop.")), "image/png");
      });
      const form = new FormData();
      form.append("image", image, `board-crop-${candidate.id}.png`);
      const response = await fetch(`${API_BASE}/recognize`, { method: "POST", body: form });
      const result = await response.json() as RecognitionResult & { detail?: string };
      if (!response.ok) throw new Error(result.detail ?? `Recognition service returned HTTP ${response.status}.`);
      if (requestId !== transferRequestRef.current) return;
      setTransferFen(result.fen);
      setRecognitionResult(result);
      setRecognitionStatus("complete");
      // A successful scan immediately becomes the active analysis position.
      // Keep the review dialog open so the user can correct uncertain squares;
      // the explicit confirmation button remains available for corrected FENs.
      if (moveToPosition(result.fen)) {
        setNotice("Scanned position loaded automatically onto the analysis board. Review the diagram and correct any uncertain squares before continuing.");
      }
    } catch (cause) {
      if (requestId !== transferRequestRef.current) return;
      setRecognitionStatus("failed");
      setRecognitionError(messageOf(cause));
    }
  }

  function updateGameEndOverlay(currentGame: Chess) {
    if (currentGame.isCheckmate()) {
      setGameEndOverlay({
        title: "Checkmate",
        message: `${currentGame.turn() === "w" ? "Black" : "White"} wins.`,
      });
    } else if (currentGame.isStalemate()) {
      setGameEndOverlay({
        title: "Stalemate",
        message: "The side to move has no legal moves, but its king is not in check. The game is a draw.",
      });
    } else if (currentGame.isDraw()) {
      let message = "The game is drawn.";
      if (currentGame.isThreefoldRepetition()) message = "Draw by threefold repetition.";
      else if (currentGame.isInsufficientMaterial()) message = "Draw by insufficient material.";
      else if (currentGame.isDrawByFiftyMoves()) message = "Draw by the fifty-move rule.";
      else if (currentGame.isDraw()) message = "Draw by a draw rule.";
      setGameEndOverlay({ title: "Draw", message });
    } else {
      setGameEndOverlay(null);
    }
  }

  function completePromotion(piece: "q" | "r" | "b" | "n") {
    if (!promotionRequest || !boardSession) return;
    const activeBoard = boardSession;
    try {
      const currentGame = new Chess(activeBoard.currentFen);
      const result = currentGame.move({
        from: promotionRequest.from,
        to: promotionRequest.to,
        promotion: piece,
      });

      patchBoardSession({
        currentFen: currentGame.fen(),
        moves: nextMoves,
        moveIndex: nextMoves.length,

      setLastMove({ from: promotionRequest.from, to: promotionRequest.to });
      setSelectedSquare(null);
      setPromotionRequest(null);
      setError("");
    } catch {
      setError("Unable to complete pawn promotion.");
      setPromotionRequest(null);
    }
  }

  function makeMove(from: Square, to: Square) {
    if (!boardSession) return;
    const activeBoard = boardSession;
    try {
      const currentGame = new Chess(activeBoard.currentFen);
      const mover = currentGame.get(from);
      const promote = mover?.type === "p" && (to.endsWith("8") || to.endsWith("1"));
      if (promote) {
        setPromotionRequest({ from, to });
        return;
      }
      const result = currentGame.move({ from, to });
      const nextMoves = [...activeBoard.moves.slice(0, activeBoard.moveIndex), result.san];
      if (workspaceMode === "opening-explorer") {
        const line = nextMoves
          .map((move, index) => index % 2 === 0 ? `${Math.floor(index / 2) + 1}. ${move}` : move)
          .join(" ");
        setOpeningExplorerInput(line);
        setOpeningExplorerPly(nextMoves.length);
      } else {
        patchBoardSession({
          currentFen: currentGame.fen(),
          moves: nextMoves,
          moveIndex: nextMoves.length,
        });
      }
      updateGameEndOverlay(currentGame);
      setLastMove({ from, to });
      setSelectedSquare(null);
      setError("");
    } catch {
      const piece = game.get(to);
      if (piece && piece.color === game.turn()) setSelectedSquare(to);
      else setSelectedSquare(null);
    }
  }

  function handleSquareClick(square: Square) {
    // A left-click removes the red highlight from that square.
    setRedSquares([]);

    if (!annotationMode && (userArrows.length > 0 || arrowStart !== null)) {
      setUserArrows([]);
      setArrowStart(null);
      setArrowHover(null);
      return;
    }
if (!boardSession) return;
    if (annotationMode) {
      if (!arrowStart) {
        setArrowStart(square);
        return;
      }
      if (arrowStart !== square) setUserArrows((arrows) => toggleArrow(arrows, { from: arrowStart, to: square }));
      setArrowStart(null);
      setArrowHover(null);
      return;
    }
    if (selectedSquare && legalTargets.has(square)) {
      makeMove(selectedSquare, square);
      return;
    }
    const piece = game.get(square);
    if (piece?.color === game.turn()) setSelectedSquare(square);
    else setSelectedSquare(null);
  }

  async function detectBoards() {
    if (detecting) {
      stopBoardScan();
      return;
    }

    if (!canvasRef.current || !session) return;
    if (!API_BASE) {
      setError("Board detection API is not configured. Configure VITE_API_BASE_URL and start the optional CV service.");
      return;
    }

    scanAbortControllerRef.current?.abort();
    const abortController = new AbortController();
    scanAbortControllerRef.current = abortController;
    const requestId = ++scanRequestIdRef.current;

    setDetecting(true);
    setError("");
    setNotice("Scanning current page...");
    patchSession((current) => ({
      ...current,
      candidatesByPage: {
        ...current.candidatesByPage,
        [String(current.page)]: [],
      },
      selectedDiagramId: null,
    }));

    try {
      const image = await new Promise<Blob>((resolve, reject) => {
        canvasRef.current?.toBlob(
          (blob) => blob
            ? resolve(blob)
            : reject(new Error("Could not encode rendered PDF page.")),
          "image/png",
        );
      });

      if (requestId !== scanRequestIdRef.current) return;

      const form = new FormData();
      form.append("image", image, `page-${session.page}.png`);

      const response = await fetch(`${API_BASE}/detect`, {
        method: "POST",
        body: form,
        signal: abortController.signal,
      });

      if (!response.ok) {
        throw new Error(`Detector returned HTTP ${response.status}.`);
      }

      const data = await response.json() as { candidates?: CandidateInput[] };
      const candidates = (data.candidates ?? []).map(
        (item, index) => normalizedCandidate({ ...item, source: "automatic" }, index),
      );

      if (requestId !== scanRequestIdRef.current) return;

      patchSession((current) => ({
        ...current,
        candidatesByPage: {
          ...current.candidatesByPage,
          [String(current.page)]: candidates,
        },
        selectedDiagramId: null,
      }));

      setNotice(
        candidates.length
          ? `Found ${candidates.length} board candidate${candidates.length === 1 ? "" : "s"}. Select a candidate to recognize its chess position and load it onto the analysis board.`
          : "No board candidates found.",
      );
    } catch (cause) {
      if (requestId !== scanRequestIdRef.current) return;

      if (abortController.signal.aborted) {
        setNotice("Scan cancelled.");
      } else {
        setError(`Board detection failed: ${messageOf(cause)}`);
      }
    } finally {
      if (requestId === scanRequestIdRef.current) {
        scanAbortControllerRef.current = null;
        setDetecting(false);
      }
    }
  }

  function moveRegion(event: React.PointerEvent<HTMLDivElement>) {

      if (!point || !session) return;

function squareCenterCoords(sq: string, flipped: boolean): { x: number; y: number } {
  const fileIndex = sq.charCodeAt(0) - 97;
  const rankIndex = parseInt(sq[1], 10) - 1;
  const col = flipped ? 7 - fileIndex : fileIndex;
  const row = flipped ? rankIndex : 7 - rankIndex;
  return { x: col + 0.5, y: row + 0.5 };
}

function pointToSquare(clientX: number, clientY: number, rect: DOMRect, flipped: boolean): Square | null {
  if (clientX < rect.left || clientX > rect.right || clientY < rect.top || clientY > rect.bottom) {
    return null;
  }
  const col = Math.min(7, Math.max(0, Math.floor(((clientX - rect.left) / rect.width) * 8)));
  const row = Math.min(7, Math.max(0, Math.floor(((clientY - rect.top) / rect.height) * 8)));
  const fileIndex = flipped ? 7 - col : col;
  const rankIndex = flipped ? row : 7 - row;
  return `${FILES[fileIndex]}${rankIndex + 1}` as Square;
}

  threat: { fill: "#882020", opacity: 0.45 },
  green: { fill: "#15781B", opacity: 0.78 },
  red: { fill: "#882020", opacity: 0.78 },
  blue: { fill: "#003088", opacity: 0.78 },
  yellow: { fill: "#e68f00", opacity: 0.78 },
};

export const ChessboardView: React.FC<ChessboardViewProps> = ({
  game,
  flipped,
  lastMove,
  engineBestMoveUci,
  threatMoveUci,
  showEngineArrow,
  showThreatArrow,
  userArrows,
  userCircles,
  onMakeMove,
  onToggleUserArrow,
  onToggleUserCircle,
  onClearUserShapes,
}) => {
  const boardRef = useRef<HTMLDivElement>(null);
  const [selectedSquare, setSelectedSquare] = useState<Square | null>(null);
  const [pendingPromotion, setPendingPromotion] = useState<{ from: Square; to: Square; color: "w" | "b" } | null>(null);
  const [drawStart, setDrawStart] = useState<Square | null>(null);
  const [drawCurrent, setDrawCurrent] = useState<Square | null>(null);
  const [drawBrush, setDrawBrush] = useState<ArrowBrush>("green");

  const boardMatrix = game.board();
  const inCheck = game.inCheck();
  const turnColor = game.turn();

  const legalDestinations = React.useMemo(() => {
    if (!selectedSquare) return new Map<Square, { isCapture: boolean; isPromotion: boolean }>();
    const moves = game.moves({ square: selectedSquare, verbose: true });
    const map = new Map<Square, { isCapture: boolean; isPromotion: boolean }>();
    for (const m of moves) {
      map.set(m.to as Square, {
        isCapture: Boolean(m.captured),
        isPromotion: Boolean(m.promotion),
      });
    }
    return map;
  }, [game, selectedSquare]);

  const attemptMove = (from: Square, to: Square) => {
    const piece = game.get(from);
    if (!piece) return;
    const isPawnPromo =
      piece.type === "p" &&
      ((piece.color === "w" && to[1] === "8") || (piece.color === "b" && to[1] === "1"));

    const legal = game.moves({ square: from, verbose: true }).some((m) => m.to === to);
    if (!legal) {
      const targetPiece = game.get(to);
      if (targetPiece && targetPiece.color === game.turn()) {
        setSelectedSquare(to);
      } else {
        setSelectedSquare(null);
      }
      return;
    }

    if (isPawnPromo) {
      setPendingPromotion({ from, to, color: piece.color });
      return;
    }

    onMakeMove(from, to);
    setSelectedSquare(null);
  };

  const handleSquareLeftClick = (sq: Square) => {
    if (userArrows.length > 0 || userCircles.length > 0) {
      onClearUserShapes();
    }
    if (selectedSquare && legalDestinations.has(sq)) {
      attemptMove(selectedSquare, sq);
      return;
    }
    const piece = game.get(sq);
    if (piece && piece.color === turnColor) {
      setSelectedSquare(sq === selectedSquare ? null : sq);
    } else {
      setSelectedSquare(null);
    }
  };

  const handlePointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
    if (e.button !== 2 || !boardRef.current) return;
    e.preventDefault();
    const sq = pointToSquare(e.clientX, e.clientY, boardRef.current.getBoundingClientRect(), flipped);
    if (!sq) return;
    const brush: ArrowBrush = e.shiftKey && e.altKey
      ? "yellow"
      : e.shiftKey
        ? "red"
        : e.altKey
          ? "blue"
          : "green";
    setDrawBrush(brush);
    setDrawStart(sq);
    setDrawCurrent(sq);
    e.currentTarget.setPointerCapture(e.pointerId);
  };

  const handlePointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
    if (!drawStart || !boardRef.current) return;
    const sq = pointToSquare(e.clientX, e.clientY, boardRef.current.getBoundingClientRect(), flipped);
    if (sq) setDrawCurrent(sq);
  };

  const handlePointerUp = (e: React.PointerEvent<HTMLDivElement>) => {
    if (!drawStart || !boardRef.current) return;
    const endSq =
      pointToSquare(e.clientX, e.clientY, boardRef.current.getBoundingClientRect(), flipped) ?? drawCurrent;
    if (endSq) {
      if (endSq === drawStart) {
        onToggleUserCircle({ square: endSq, brush: drawBrush });
      } else {
        onToggleUserArrow({ from: drawStart, to: endSq, brush: drawBrush });
      }
    }
    setDrawStart(null);
    setDrawCurrent(null);
  };

  // Combine engine best-move arrow, threat arrow, user arrows, and active preview arrow
  const allArrows: BoardArrowShape[] = [];
  if (showEngineArrow && engineBestMoveUci && engineBestMoveUci.length >= 4) {
    allArrows.push({
      from: engineBestMoveUci.slice(0, 2) as Square,
      to: engineBestMoveUci.slice(2, 4) as Square,
      brush: "engine",
    });
  }
  if (showThreatArrow && threatMoveUci && threatMoveUci.length >= 4) {
    allArrows.push({
      from: threatMoveUci.slice(0, 2) as Square,
      to: threatMoveUci.slice(2, 4) as Square,
      brush: "threat",
    });
  }
  allArrows.push(...userArrows);
  if (drawStart && drawCurrent && drawStart !== drawCurrent) {
    allArrows.push({ from: drawStart, to: drawCurrent, brush: drawBrush });
  }

  return (
    <div className="li-board-container">
      <div
        ref={boardRef}
        className="li-chessground-board"
        role="grid"
        aria-label="Lichess Analysis Chessboard"
        onPointerDown={handlePointerDown}
        onPointerMove={handlePointerMove}
        onPointerUp={handlePointerUp}
        onContextMenu={(e) => e.preventDefault()}
      >
        {Array.from({ length: 64 }, (_, idx) => {
          const row = Math.floor(idx / 8);
          const col = idx % 8;
          const fileIndex = flipped ? 7 - col : col;
          const rankIndex = flipped ? row : 7 - row;
          const sq = `${FILES[fileIndex]}${rankIndex + 1}` as Square;
          const isLight = (fileIndex + rankIndex) % 2 === 1;
          const piece = boardMatrix[7 - rankIndex]?.[fileIndex] ?? null;
          const isLastMove = lastMove?.from === sq || lastMove?.to === sq;
          const isSelected = selectedSquare === sq;
          const legalInfo = legalDestinations.get(sq);
          const isKingInCheck = inCheck && piece?.type === "k" && piece?.color === turnColor;

          return (
            <div
              key={sq}
              role="gridcell"
              aria-label={`${sq}${piece ? ` ${piece.color === "w" ? "White" : "Black"} ${piece.type}` : ""}`}
              data-square={sq}
              className={[
                "li-cg-square",
                isLight ? "li-cg-light" : "li-cg-dark",
                isLastMove ? "li-cg-last-move" : "",
                isSelected ? "li-cg-selected" : "",
                isKingInCheck ? "li-cg-check" : "",
              ]
                .filter(Boolean)
                .join(" ")}
              onClick={() => handleSquareLeftClick(sq)}
              onDragOver={(e) => e.preventDefault()}
              onDrop={(e) => {
                e.preventDefault();
                const fromSq = e.dataTransfer.getData("text/plain") as Square;
                if (fromSq && fromSq !== sq) {
                  attemptMove(fromSq, sq);
                }
              }}
            >
              {/* Rank coordinate on right-most column (col === 7) matching Lichess */}
              {col === 7 && (
                <span className={`li-coord-rank ${isLight ? "li-coord-on-light" : "li-coord-on-dark"}`}>
                  {rankIndex + 1}
                </span>
              )}

              {/* File coordinate on bottom-most row (row === 7) matching Lichess */}
              {row === 7 && (
                <span className={`li-coord-file ${isLight ? "li-coord-on-light" : "li-coord-on-dark"}`}>
                  {FILES[fileIndex]}
                </span>
              )}

              {/* Piece SVG */}
              {piece && (
                <div
                  className="li-piece-wrapper"
                  draggable={piece.color === turnColor}
                  onDragStart={(e) => {
                    e.dataTransfer.setData("text/plain", sq);
                    setSelectedSquare(sq);
                  }}
                >
                  <CburnettPiece color={piece.color} type={piece.type} />
                </div>
              )}

              {/* Legal move dot or capture corner ring */}
              {legalInfo && (
                <span className={legalInfo.isCapture ? "li-legal-capture-ring" : "li-legal-move-dot"} />
              )}
            </div>
          );
        })}

        {/* SVG Overlay for Engine Best-Move Arrow, Threat Arrow, User Arrows & Circles */}
        <svg className="li-cg-shapes-svg" viewBox="0 0 8 8" aria-hidden="true">
          {userCircles.map((circle, i) => {
            const center = squareCenterCoords(circle.square, flipped);
            const style = BRUSH_COLORS[circle.brush];
            return (
              <circle
                key={`circle-${circle.square}-${i}`}
                cx={center.x}
                cy={center.y}
                r={0.44}
                fill="none"
                stroke={style.fill}
                strokeWidth={0.075}
                opacity={style.opacity}
              />
            );
          })}

          {allArrows.map((arrow, i) => {
            const from = squareCenterCoords(arrow.from, flipped);
            const to = squareCenterCoords(arrow.to, flipped);
            const dx = to.x - from.x;
            const dy = to.y - from.y;
            const len = Math.hypot(dx, dy);
            if (len < 0.01) return null;

            const ux = dx / len;
            const uy = dy / len;
            const px = -uy;
            const py = ux;

            // Match Lichess thick translucent best-move arrow proportions in screenshot
            const isEngine = arrow.brush === "engine";
            const shaftHalfWidth = isEngine ? 0.135 : 0.095;
            const headHalfWidth = isEngine ? 0.36 : 0.27;
            const headLength = isEngine ? 0.48 : 0.38;

            const tipX = to.x - ux * 0.05;
            const tipY = to.y - uy * 0.05;
            const neckX = tipX - ux * headLength;
            const neckY = tipY - uy * headLength;
            const startX = from.x;
            const startY = from.y;

            const pts = [
              [startX + px * shaftHalfWidth, startY + py * shaftHalfWidth],
              [neckX + px * shaftHalfWidth, neckY + py * shaftHalfWidth],
              [neckX + px * headHalfWidth, neckY + py * headHalfWidth],
              [tipX, tipY],
              [neckX - px * headHalfWidth, neckY - py * headHalfWidth],
              [neckX - px * shaftHalfWidth, neckY - py * shaftHalfWidth],
              [startX - px * shaftHalfWidth, startY - py * shaftHalfWidth],
            ]
              .map(([x, y]) => `${x.toFixed(3)},${y.toFixed(3)}`)
              .join(" ");

            const brushStyle = BRUSH_COLORS[arrow.brush];
            return (
              <g key={`arrow-${arrow.from}-${arrow.to}-${i}`} opacity={brushStyle.opacity}>
                <circle cx={startX} cy={startY} r={shaftHalfWidth} fill={brushStyle.fill} />
                <polygon points={pts} fill={brushStyle.fill} />
              </g>
            );
          })}
        </svg>

        {/* Promotion Chooser Overlay */}
        {pendingPromotion && (
          <div className="li-promotion-overlay" onClick={() => setPendingPromotion(null)}>
            <div className="li-promotion-column" onClick={(e) => e.stopPropagation()}>
              {(["q", "n", "r", "b"] as const).map((promoPiece) => (
                <button
                  key={promoPiece}
                  type="button"
                  className="li-promotion-choice"
                  aria-label={`Promote to ${promoPiece}`}
                  onClick={() => {
                    onMakeMove(pendingPromotion.from, pendingPromotion.to, promoPiece);
                    setPendingPromotion(null);
                    setSelectedSquare(null);
                  }}
                >
                  <CburnettPiece color={pendingPromotion.color} type={promoPiece} />
                </button>
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  );
};

/**
 * Converts standard SAN move strings (e.g. "Nf3", "Bxb5+") into Lichess
 * figurative notation ("♘f3", "♗xb5+") used in the PV strip.
 */
export function toFigurativeSan(san: string): string {
  return san
    .replace(/^K/, "♔")
    .replace(/^Q/, "♕")
    .replace(/^R/, "♖")
    .replace(/^B/, "♗")
    .replace(/^N/, "♘");
}

import React, { useState } from "react";
import { Target, Settings, Plus } from "lucide-react";
import type { EngineEvaluation } from "./types";
import { formatLichessEval } from "./stockfishWorkerClient";
import { toFigurativeSan } from "./CburnettPieces";

interface EnginePanelProps {
  enabled: boolean;
  onToggleEnabled: () => void;
  evaluation: EngineEvaluation | null;
  depth: number;
  onChangeDepth: (d: number) => void;
  showThreat: boolean;
  onToggleThreat: () => void;
  currentTurn: "w" | "b";
  currentMoveNumber: number;
  onPlayPvFirstMove?: (uci: string) => void;
}

export const EnginePanel: React.FC<EnginePanelProps> = ({
  enabled,
  onToggleEnabled,
  evaluation,
  depth,
  onChangeDepth,
  showThreat,
  onToggleThreat,
  currentTurn,
  currentMoveNumber,
  onPlayPvFirstMove,
}) => {
  const [settingsOpen, setSettingsOpen] = useState(false);
  const [pvExpanded, setPvExpanded] = useState(false);
  const [multiPvCount, setMultiPvCount] = useState(1);

  const evalFormatted = enabled ? formatLichessEval(evaluation) : "—";

  // Build formatted PV string matching `1... e5 2. ♘f3 ♘c6 3. ♗b5 ♘f6 4. O-O ♘xe4 5. ♖e1 ♘d6 6...`
  const formattedPvTokens = React.useMemo(() => {
    if (!enabled || !evaluation || evaluation.pvSan.length === 0) return [];
    const tokens: { label: string; uci?: string }[] = [];
    let moveNum = currentTurn === "b" ? Math.max(1, currentMoveNumber) : currentMoveNumber + 1;
    let side = currentTurn;

    evaluation.pvSan.forEach((san, idx) => {
      const fig = toFigurativeSan(san);
      const uci = evaluation.pvUci[idx];
      if (idx === 0 && side === "b") {
        tokens.push({ label: `${moveNum}... ${fig}`, uci });
        side = "w";
        moveNum++;
      } else if (side === "w") {
        tokens.push({ label: `${moveNum}. ${fig}`, uci });
        side = "b";
      } else {
        tokens.push({ label: fig, uci });
        side = "w";
        moveNum++;
      }
    });
    return tokens;
  }, [enabled, evaluation, currentTurn, currentMoveNumber]);

  return (
    <div className="li-ceval-section">
      {/* Top thin green engine progress bar */}
      <div className="li-ceval-progress-track">
        <div
          className="li-ceval-progress-bar"
          style={{ width: enabled ? `${Math.min(100, ((evaluation?.depth ?? depth) / 70) * 100)}%` : "0%" }}
        />
      </div>

      {/* Main Engine Header Strip */}
      <div className="li-ceval-header">
        {/* Lichess Green Toggle Switch with Checkmark */}
        <button
          type="button"
          role="switch"
          aria-checked={enabled}
          aria-label="Toggle Stockfish evaluation"
          className={`li-ceval-switch ${enabled ? "on" : "off"}`}
          onClick={onToggleEnabled}
        >
          <span className="li-ceval-switch-knob">
            {enabled ? "✓" : ""}
          </span>
        </button>

        {/* Big Score Readout (+0.2) */}
        <output className="li-ceval-score">{evalFormatted}</output>

        {/* Engine Metadata Block (SF 19 1MB NNUE / Depth 70 CLOUD) */}
        <div className="li-ceval-meta">
          <div className="li-ceval-meta-top">
            <span className="li-sf-title">SF 19 1MB</span>
            <span className="li-nnue-badge">NNUE</span>
          </div>
          <div className="li-ceval-meta-bottom">
            <button
              type="button"
              className="li-depth-plus-btn"
              title="Increase search depth"
              onClick={() => onChangeDepth(Math.min(99, depth + 4))}
            >
              <Plus size={11} strokeWidth={3} />
            </button>
            <span className="li-depth-text">
              Depth {enabled ? evaluation?.depth ?? depth : 0}
            </span>
            {enabled && evaluation?.isCloud && (
              <span className="li-cloud-badge">CLOUD</span>
            )}
          </div>
        </div>

        {/* Right Controls: Threat Target & Settings Gear */}
        <div className="li-ceval-actions">
          <button
            type="button"
            className={`li-ceval-icon-btn ${showThreat ? "active" : ""}`}
            title="Show threat (x)"
            aria-label="Show threat"
            onClick={onToggleThreat}
          >
            <Target size={17} />
          </button>
          <button
            type="button"
            className={`li-ceval-icon-btn ${settingsOpen ? "active" : ""}`}
            title="Engine settings"
            aria-label="Engine settings"
            onClick={() => setSettingsOpen((s) => !s)}
          >
            <Settings size={17} />
          </button>
        </div>
      </div>

      {/* Collapsible Engine Settings Drawer */}
      {settingsOpen && (
        <div className="li-ceval-settings-drawer">
          <label className="li-ceval-setting-row">
            <span>Search depth ({depth})</span>
            <input
              type="range"
              min={10}
              max={70}
              value={depth}
              onChange={(e) => onChangeDepth(Number(e.target.value))}
            />
          </label>
          <label className="li-ceval-setting-row">
            <span>Multiple lines ({multiPvCount})</span>
            <input
              type="range"
              min={1}
              max={3}
              value={multiPvCount}
              onChange={(e) => setMultiPvCount(Number(e.target.value))}
            />
          </label>
        </div>
      )}

      {/* Principal Variation Box (pv_box) */}
      <div className={`li-pv-box ${pvExpanded ? "expanded" : ""}`}>
        {enabled && formattedPvTokens.length > 0 ? (
          <>
            <div
              className="li-pv-line-text"
              title="Click to play best move on board"
              onClick={() => {
                if (evaluation?.bestMoveUci && onPlayPvFirstMove) {
                  onPlayPvFirstMove(evaluation.bestMoveUci);
                }
              }}
            >
              {formattedPvTokens.map((tok, i) => (
                <span key={i} className="li-pv-token">
                  {tok.label}{" "}
                </span>
              ))}
            </div>
            <button
              type="button"
              className="li-pv-expand-caret"
              aria-label="Toggle full principal variation"
              onClick={() => setPvExpanded((e) => !e)}

              ▾

          <div className="li-pv-line-muted">
            {enabled ? "Calculating principal variation…" : "Engine disabled"}
          </div>
        )}
      </div>
    </div>
  );
};

import React from "react";
import type { MoveNode } from "./types";
import { toFigurativeSan } from "./CburnettPieces";

interface MoveListPanelProps {
  moves: MoveNode[];
  currentPly: number;
  onSelectPly: (ply: number) => void;
}

export const MoveListPanel: React.FC<MoveListPanelProps> = ({
  moves,
  currentPly,
  onSelectPly,
}) => {
  // Group moves into pairs (White + Black per full move number)
  const movePairs = React.useMemo(() => {
    const rows: { moveNumber: number; white?: MoveNode; black?: MoveNode }[] = [];
    for (const node of moves) {
      let row = rows[rows.length - 1];
      if (!row || row.moveNumber !== node.moveNumber) {
        row = { moveNumber: node.moveNumber };
        rows.push(row);
      }
      if (node.color === "w") row.white = node;
      else row.black = node;
    }
    return rows;
  }, [moves]);

  return (
    <div className="li-moves-container" role="region" aria-label="Move list">
      {movePairs.map((pair) => (
        <div key={pair.moveNumber} className="li-move-row">
          <div className="li-move-index">{pair.moveNumber}</div>

          {pair.white ? (
            <button
              type="button"
              className={`li-move-cell ${currentPly === pair.white.ply ? "active" : ""}`}
              onClick={() => onSelectPly(pair.white!.ply)}
            >
              <span className="li-move-san">{toFigurativeSan(pair.white.san)}</span>
              {pair.white.evalFormatted && (
                <span className="li-move-eval">{pair.white.evalFormatted}</span>
              )}
            </button>
          ) : (
            <div className="li-move-cell li-move-empty">...</div>
          )}

          {pair.black ? (
            <button
              type="button"
              className={`li-move-cell ${currentPly === pair.black.ply ? "active" : ""}`}
              onClick={() => onSelectPly(pair.black!.ply)}
            >
              <span className="li-move-san">{toFigurativeSan(pair.black.san)}</span>
              {pair.black.evalFormatted && (
                <span className="li-move-eval">{pair.black.evalFormatted}</span>
              )}
            </button>
          ) : (
            <div className="li-move-cell li-move-empty" />
          )}
        </div>
      ))}
    </div>
  );
};
