function evaluationText(score?: { kind: "cp" | "mate"; value: number }) {
  if (!score) return "—";
  if (score.kind === "mate") return `${score.value > 0 ? "+" : ""}#${Math.abs(score.value)}`;
  const value = score.value / 100;
  return `${value > 0 ? "+" : ""}${value.toFixed(2)}`;
}

function resultText(result?: string) {
  if (result === "1-0") return "White won";
  if (result === "0-1") return "Black won";
  if (result === "1/2-1/2") return "Draw";
  return "Result not recorded";
}

function metadata(value?: string) {
  return value && value !== "?" && value !== "*" ? value : "";
}

function classificationClass(value: string) {

}

import type { EngineScore } from "./stockfish";
import { ReviewEngine, type ReviewProgress } from "./review-engine";
import type { ReviewGame } from "./review-pgn";

export type MoveClassification =
  | "Best"
  | "Excellent"
  | "Good"
  | "Inaccuracy"
  | "Mistake"
  | "Blunder";

export type ClassificationThresholds = {
  best: number;
  excellent: number;
  good: number;
  inaccuracy: number;
  mistake: number;
};

export const DEFAULT_CLASSIFICATION_THRESHOLDS: ClassificationThresholds = {
  best: 10,
  excellent: 30,
  good: 80,
  inaccuracy: 150,
  mistake: 300,
};

export type ReviewedMove = {
  ply: number;
  moveNumber: number;
  color: "w" | "b";
  san: string;
  from: string;
  to: string;
  promotion?: string;
  playedMoveUci: string;
  bestMove?: string;
  playedMoveIsBest: boolean;
  before: EngineScore;
  after: EngineScore;
  evaluationLossCp: number;
  classification: MoveClassification;
};

export type ReviewAnalysis = {
  depth: number;
  positionsAnalyzed: number;
  moves: ReviewedMove[];
  positionScores: EngineScore[];
};

export type ReviewEnginePort = Pick<ReviewEngine, "analyzePositions">;

export function scoreToWhiteCentipawns(score: EngineScore): number {
  if (score.kind === "cp") return score.value;

  const distance = Math.min(99999, Math.abs(score.value));
  return Math.sign(score.value || 1) * (100000 - distance);
}

export function calculateMoveLoss(
  before: EngineScore,
  after: EngineScore,
  color: "w" | "b",
): number {
  const beforeCp = scoreToWhiteCentipawns(before);
  const afterCp = scoreToWhiteCentipawns(after);

  const loss = color === "w"
    ? beforeCp - afterCp
    : afterCp - beforeCp;

  return Math.max(0, Math.round(loss));
}

export function classifyMove(
  lossCp: number,
  thresholds: ClassificationThresholds = DEFAULT_CLASSIFICATION_THRESHOLDS,
): MoveClassification {
  if (!Number.isFinite(lossCp) || lossCp < 0) {
    throw new Error("Evaluation loss must be a finite, non-negative number.");
  }

  if (
    thresholds.best < 0 ||
    thresholds.excellent < thresholds.best ||
    thresholds.good < thresholds.excellent ||
    thresholds.inaccuracy < thresholds.good ||
    thresholds.mistake < thresholds.inaccuracy
  ) {
    throw new Error("Classification thresholds must be non-negative and increasing.");
  }

  if (lossCp <= thresholds.best) return "Best";
  if (lossCp <= thresholds.excellent) return "Excellent";
  if (lossCp <= thresholds.good) return "Good";
  if (lossCp <= thresholds.inaccuracy) return "Inaccuracy";
  if (lossCp <= thresholds.mistake) return "Mistake";
  return "Blunder";
}

export function buildReviewAnalysis(
  review: ReviewGame,
  positionScores: EngineScore[],
  depth: number,
  thresholds: ClassificationThresholds = DEFAULT_CLASSIFICATION_THRESHOLDS,
): ReviewAnalysis {
  if (positionScores.length !== review.positions.length + 1) {
    throw new Error("Expected one engine evaluation for every game position.");
  }

  const moves = review.positions.map((move, index): ReviewedMove => {
    const before = positionScores[index];
    const after = positionScores[index + 1];

    if (!before || !after) {
      throw new Error(`Missing engine evaluation around ply ${move.ply}.`);
    }

    const playedMoveUci =
      `${move.from}${move.to}${move.promotion ?? ""}`;

    const playedMoveIsBest = before.bestMove
      ? before.bestMove.toLowerCase() === playedMoveUci.toLowerCase()
      : false;

    const evaluationLossCp = calculateMoveLoss(
      before,
      after,
      move.color,
    );

    return {
      ply: move.ply,
      moveNumber: move.moveNumber,
      color: move.color,
      san: move.san,
      from: move.from,
      to: move.to,
      ...(move.promotion ? { promotion: move.promotion } : {}),
      playedMoveUci,
      ...(before.bestMove ? { bestMove: before.bestMove } : {}),
      playedMoveIsBest,
      before,
      after,
      evaluationLossCp,
      classification: playedMoveIsBest
        ? "Best"
        : classifyMove(evaluationLossCp, thresholds),
    };
  });

  return {
    depth,
    positionsAnalyzed: positionScores.length,
    moves,
    positionScores,
  };
}

export async function analyzeReviewPositions(
  review: ReviewGame,
  engine: ReviewEnginePort,
  options: {
    depth: number;
    onProgress?: (progress: ReviewProgress) => void;
    thresholds?: ClassificationThresholds;
  },
): Promise<ReviewAnalysis> {
  const depth = Math.max(1, Math.min(24, Math.floor(options.depth)));

  const fens = [
    review.startingFen,
    ...review.positions.map((position) => position.afterFen),
  ];

  const positionScores = await engine.analyzePositions(fens, {
    depth,
    onProgress: options.onProgress,
  });

  return buildReviewAnalysis(
    review,
    positionScores,
    depth,
    options.thresholds,
  );
}

import { parseUciInfo, type EngineScore } from "./stockfish";

export type ReviewProgress = {
  completed: number;
  total: number;
  currentPly: number;
};

export type ReviewAnalysisOptions = {
  depth: number;
  onProgress?: (progress: ReviewProgress) => void;
};

type ReviewBatch = {
  fens: string[];
  depth: number;
  index: number;
  scores: EngineScore[];
  onProgress?: (progress: ReviewProgress) => void;
  resolve: (scores: EngineScore[]) => void;
  reject: (error: Error) => void;
};

export class ReviewEngine {
  private readonly worker: Worker;
  private ready = false;
  private disposed = false;
  private latestScore: EngineScore | null = null;
  private activeBatch: ReviewBatch | null = null;
  private readyResolve!: () => void;
  private readyReject!: (error: Error) => void;

  private readonly readyPromise = new Promise<void>((resolve, reject) => {
    this.readyResolve = resolve;
    this.readyReject = reject;
  });

  constructor(workerUrl: string) {
    // Mark readiness rejection as handled even when no analysis is awaiting it.
    void this.readyPromise.catch(() => undefined);

    this.worker = new Worker(workerUrl);

    this.worker.onmessage = (event: MessageEvent<string>) => {
      this.handleLine(String(event.data));
    };

    this.worker.onerror = (event) => {
      this.fail(new Error(event.message || "Stockfish review worker failed."));
    };

    this.worker.postMessage("uci");
  }

  async analyzePositions(
    fens: string[],
    options: ReviewAnalysisOptions,
  ): Promise<EngineScore[]> {
    if (this.disposed) {
      throw new Error("The review engine has been disposed.");
    }

    if (this.activeBatch) {
      throw new Error("A review analysis is already running.");
    }

    if (fens.length === 0) {
      return [];
    }

    const depth = Math.max(1, Math.min(24, Math.floor(options.depth)));

    if (fens.some((fen) => !fen.trim())) {
      throw new Error("Every review position must contain a valid FEN.");
    }

    await this.readyPromise;

    if (this.disposed) {
      throw new Error("The review engine has been disposed.");
    }

    return new Promise<EngineScore[]>((resolve, reject) => {
      this.activeBatch = {
        fens: [...fens],
        depth,
        index: 0,
        scores: [],
        onProgress: options.onProgress,
        resolve,
        reject,
      };

      this.startPosition();
    });
  }

  cancel(): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;

    this.disposed = true;
    this.worker.terminate();

    const error = new Error("Review analysis was cancelled.");
    batch?.reject(error);
    this.readyReject(error);
  }

  dispose(): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;
    this.disposed = true;

    this.worker.terminate();

    const error = new Error("The review engine has been disposed.");
    batch?.reject(error);
    this.readyReject(error);
  }

  private handleLine(line: string): void {
    if (this.disposed) return;

    if (/^Error:/i.test(line)) {
      this.fail(new Error(line));
      return;
    }

    if (line === "uciok") {
      this.worker.postMessage("isready");
      return;
    }

    if (line === "readyok") {
      this.ready = true;
      this.readyResolve();
      return;
    }

    const batch = this.activeBatch;
    if (!batch) return;

    if (line.startsWith("info ")) {
      const sideToMove = batch.fens[batch.index].split(/\s+/)[1] === "b" ? "b" : "w";
      const parsed = parseUciInfo(line, sideToMove);

      if (parsed && (!this.latestScore || parsed.depth >= this.latestScore.depth)) {
        this.latestScore = parsed;
      }

      return;
    }

    if (!line.startsWith("bestmove ")) return;

    const bestMove = line.match(/^bestmove (\S+)/)?.[1];

    if (!this.latestScore) {
      this.fail(new Error(`Stockfish returned no evaluation for ply ${batch.index + 1}.`));
      return;
    }

    const score: EngineScore = {
      ...this.latestScore,
      ...(bestMove && bestMove !== "(none)" ? { bestMove } : {}),
    };

    batch.scores.push(score);
    batch.index += 1;

    batch.onProgress?.({
      completed: batch.index,
      total: batch.fens.length,
      currentPly: batch.index,
    });

    this.latestScore = null;

    if (batch.index >= batch.fens.length) {
      this.activeBatch = null;
      batch.resolve(batch.scores);
      return;
    }

    this.startPosition();
  }

  private startPosition(): void {
    const batch = this.activeBatch;
    if (!batch || this.disposed || !this.ready) return;

    this.latestScore = null;

    const fen = batch.fens[batch.index];
    this.worker.postMessage(`position fen ${fen}`);
    this.worker.postMessage(`go depth ${batch.depth}`);
  }

  private fail(error: Error): void {
    if (this.disposed) return;

    const batch = this.activeBatch;
    this.activeBatch = null;
    this.disposed = true;

    this.worker.terminate();
    batch?.reject(error);
    this.readyReject(error);
  }
}

import { Chess } from "chess.js";
import { parsePgn } from "./pgn-import";

export type ReviewPosition = {
  ply: number;
  moveNumber: number;
  color: "w" | "b";
  san: string;
  beforeFen: string;
  afterFen: string;
  from: string;
  to: string;
  promotion?: string;
  flags: string;
};

export type ReviewGame = {
  headers: Record<string, string>;
  startingFen: string;
  finalFen: string;
  positions: ReviewPosition[];
};

export function reconstructReviewGame(pgn: string): ReviewGame {
  const imported = parsePgn(pgn);
  const game = new Chess(imported.lineBaseFen);
  const positions: ReviewPosition[] = [];

  imported.moves.forEach((san, index) => {
    const beforeFen = game.fen();
    const move = game.move(san);

    positions.push({
      ply: index + 1,
      moveNumber: Math.floor(index / 2) + 1,
      color: index % 2 === 0 ? "w" : "b",
      san: move.san,
      beforeFen,
      afterFen: game.fen(),
      from: move.from,
      to: move.to,
      ...(move.promotion ? { promotion: move.promotion } : {}),
      flags: move.flags,
    });
  });

  if (game.fen() !== imported.finalFen) {
    throw new Error("Review reconstruction does not match the imported PGN.");
  }

  return {
    headers: imported.headers,
    startingFen: imported.lineBaseFen,
    finalFen: game.fen(),
    positions,
  };
}

export function getReviewFenAtPly(review: ReviewGame, ply: number): string {
  const safePly = Math.max(0, Math.min(review.positions.length, Math.floor(ply)));

  return safePly === 0
    ? review.startingFen
    : review.positions[safePly - 1].afterFen;
}
