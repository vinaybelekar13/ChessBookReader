function formatUciMove(fen: string, uci: string): string {
  const match = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
  if (!match) return uci;
  try {
    const position = new Chess(fen);
    const move = position.move({
      from: match[1] as Square,
      to: match[2] as Square,
      ...(match[3] ? { promotion: match[3] as "q" | "r" | "b" | "n" } : {}),
    });
    return move.san;
  } catch {
    return uci;
  }
}

function formatUciPv(fen: string, pv: string[]): string {
  const position = new Chess(fen);
  const san: string[] = [];
  for (const uci of pv) {
    const match = /^([a-h][1-8])([a-h][1-8])([qrbn])?$/.exec(uci);
    if (!match) break;
    try {
      const move = position.move({
        from: match[1] as Square,
        to: match[2] as Square,
        ...(match[3] ? { promotion: match[3] as "q" | "r" | "b" | "n" } : {}),
      });
      san.push(move.san);
    } catch {
      break;
    }
  }
  return san.join(" ");
}

function readableDate(timestamp: number): string {
  if (!timestamp) return "Not opened";
  const elapsed = Date.now() - timestamp;
  if (elapsed < 60_000) return "Just now";
  if (elapsed < 3_600_000) return `${Math.floor(elapsed / 60_000)}m ago`;
  if (elapsed < 86_400_000) return `${Math.floor(elapsed / 3_600_000)}h ago`;
  return new Date(timestamp).toLocaleDateString();
}

function squareName(fileIndex: number, rankIndex: number): Square {
  return `${"abcdefgh"[fileIndex]}${rankIndex + 1}` as Square;
}

function normalizedCandidate(value: CandidateInput, index: number): Candidate {
  const { relative_score, confidence, ...candidate } = value;
  return {
    ...candidate,
    id: candidate.id || `candidate-${index + 1}`,
    source: candidate.source ?? "automatic",
    relativeScore: candidate.relativeScore ?? relative_score ?? confidence ?? 0,
  };
}

  async function addBook(file?: File) {
    if (!file) return;
    try {
      if (file.size > FILE_LIMIT) throw new Error("PDFs must be 100 MB or smaller.");
      const header = new Uint8Array(await file.slice(0, 5).arrayBuffer());
      if (new TextDecoder().decode(header) !== "%PDF-") throw new Error("The selected file does not contain a valid PDF header.");
      const now = Date.now();
      const created: BookRecord = {
        id: crypto.randomUUID(),
        title: file.name.replace(/\.pdf$/i, ""),
        file: file.slice(0, file.size, "application/pdf"),
        byteLength: file.size,
        lastOpenedAt: now,
        addedAt: now,
      };
      const blank = newSession(created.id);
      await Promise.all([putBook(created), putSession(blank), setActiveBookId(created.id)]);
      setBooks((current) => [created, ...current]);
      setSessionIndex((current) => ({ ...current, [created.id]: blank }));
      setBook(created);
      setSession(blank);
      setNotice("PDF added to this browser's local library.");
      setError("");
      if (fileInput.current) fileInput.current.value = "";
    } catch (cause) {

  async function deleteBook(item: BookRecord) {
    if (!window.confirm(`Delete "${item.title}" and its saved reading progress from this browser? This cannot be undone.`)) return;

    try {
      await removeBook(item.id);
      await removeSession(item.id);

      setBooks((current) => current.filter((entry) => entry.id !== item.id));
      setSessionIndex((current) => {

        delete next[item.id];

      });

        setBook(null);
        setSession(null);
        setPdf(null);
      }

      setNotice(`Removed "${item.title}" from your library.`);
      setError("");
    } catch (cause) {
      setError(`Could not remove book: ${messageOf(cause)}`);
    }
  }

    }

  async function openTransfer(candidate: Candidate) {
    const canvas = canvasRef.current;
    if (!canvas || canvas.width === 0 || canvas.height === 0) {
      setError("The rendered page image is unavailable. Rerender the page before using this crop.");
      return;
    }
    let preview: HTMLCanvasElement;
    try {
      const crop = cropToPixels(candidate, canvas.width, canvas.height);
      preview = document.createElement("canvas");
      preview.width = crop.width;
      preview.height = crop.height;
      const context = preview.getContext("2d");
      if (!context) throw new Error("Could not prepare a crop preview in this browser.");
      context.drawImage(canvas, crop.x, crop.y, crop.width, crop.height, 0, 0, crop.width, crop.height);
    } catch (cause) {
      setError(`Could not create the crop preview: ${messageOf(cause)}`);
      return;
    }
    setTransferCandidate(candidate);
    setTransferPreview(preview.toDataURL("image/png"));
    setTransferFen("");
    setRecognitionResult(null);
        setRecognitionError("");
        const requestId = ++transferRequestRef.current;
    setRecognitionStatus("loading");
    try {
      const image = await new Promise<Blob>((resolve, reject) => {
        preview.toBlob((blob) => blob ? resolve(blob) : reject(new Error("Could not encode the selected crop.")), "image/png");
      });
      const form = new FormData();
      form.append("image", image, `board-crop-${candidate.id}.png`);
      const response = await fetch(`${API_BASE}/recognize`, { method: "POST", body: form });
      const result = await response.json() as RecognitionResult & { detail?: string };
      if (!response.ok) throw new Error(result.detail ?? `Recognition service returned HTTP ${response.status}.`);
      if (requestId !== transferRequestRef.current) return;
      setTransferFen(result.fen);
      setRecognitionResult(result);
      setRecognitionStatus("complete");
      // A successful scan immediately becomes the active analysis position.
      // Keep the review dialog open so the user can correct uncertain squares;
      // the explicit confirmation button remains available for corrected FENs.
      if (moveToPosition(result.fen)) {
        setNotice("Scanned position loaded automatically onto the analysis board. Review the diagram and correct any uncertain squares before continuing.");
      }
    } catch (cause) {
      if (requestId !== transferRequestRef.current) return;
      setRecognitionStatus("failed");
      setRecognitionError(messageOf(cause));
    }
  }

  function updateGameEndOverlay(currentGame: Chess) {
    if (currentGame.isCheckmate()) {
      setGameEndOverlay({
        title: "Checkmate",
        message: `${currentGame.turn() === "w" ? "Black" : "White"} wins.`,
      });
    } else if (currentGame.isStalemate()) {
      setGameEndOverlay({
        title: "Stalemate",
        message: "The side to move has no legal moves, but its king is not in check. The game is a draw.",
      });
    } else if (currentGame.isDraw()) {
      let message = "The game is drawn.";
      if (currentGame.isThreefoldRepetition()) message = "Draw by threefold repetition.";
      else if (currentGame.isInsufficientMaterial()) message = "Draw by insufficient material.";
      else if (currentGame.isDrawByFiftyMoves()) message = "Draw by the fifty-move rule.";
      else if (currentGame.isDraw()) message = "Draw by a draw rule.";
      setGameEndOverlay({ title: "Draw", message });
    } else {
      setGameEndOverlay(null);
    }
  }

  function completePromotion(piece: "q" | "r" | "b" | "n") {
    if (!promotionRequest || !boardSession) return;

  }

  function makeMove(from: Square, to: Square) {
    if (!boardSession) return;
    const activeBoard = boardSession;
    try {
      const currentGame = new Chess(activeBoard.currentFen);
      const mover = currentGame.get(from);
      const promote = mover?.type === "p" && (to.endsWith("8") || to.endsWith("1"));
      if (promote) {
        setPromotionRequest({ from, to });
        return;
      }
      const result = currentGame.move({ from, to });
      const nextMoves = [...activeBoard.moves.slice(0, activeBoard.moveIndex), result.san];
      if (workspaceMode === "opening-explorer") {
        const line = nextMoves
          .map((move, index) => index % 2 === 0 ? `${Math.floor(index / 2) + 1}. ${move}` : move)
          .join(" ");
        setOpeningExplorerInput(line);
        setOpeningExplorerPly(nextMoves.length);
      } else {
        patchBoardSession({
          currentFen: currentGame.fen(),
          moves: nextMoves,
          moveIndex: nextMoves.length,
        });
      }
      updateGameEndOverlay(currentGame);
      setLastMove({ from, to });
      setSelectedSquare(null);
      setError("");
    } catch {
      const piece = game.get(to);
      if (piece && piece.color === game.turn()) setSelectedSquare(to);
      else setSelectedSquare(null);
    }
  }

  function handleSquareClick(square: Square) {
    // A left-click removes the red highlight from that square.
    setRedSquares([]);

    if (!annotationMode && (userArrows.length > 0 || arrowStart !== null)) {
      setUserArrows([]);
      setArrowStart(null);
      setArrowHover(null);
      return;
    }
if (!boardSession) return;
    if (annotationMode) {
      if (!arrowStart) {
        setArrowStart(square);
        return;
      }
      if (arrowStart !== square) setUserArrows((arrows) => toggleArrow(arrows, { from: arrowStart, to: square }));
      setArrowStart(null);
      setArrowHover(null);
      return;
    }
    if (selectedSquare && legalTargets.has(square)) {
      makeMove(selectedSquare, square);
      return;
    }
    const piece = game.get(square);
    if (piece?.color === game.turn()) setSelectedSquare(square);
    else setSelectedSquare(null);
  }

  async function detectBoards() {
    if (detecting) {
      stopBoardScan();
      return;
    }

    if (!canvasRef.current || !session) return;
    if (!API_BASE) {
      setError("Board detection API is not configured. Configure VITE_API_BASE_URL and start the optional CV service.");
      return;
    }

    scanAbortControllerRef.current?.abort();
    const abortController = new AbortController();
    scanAbortControllerRef.current = abortController;
    const requestId = ++scanRequestIdRef.current;

    setDetecting(true);
    setError("");
    setNotice("Scanning current page...");
    patchSession((current) => ({
      ...current,
      candidatesByPage: {
        ...current.candidatesByPage,
        [String(current.page)]: [],
      },
      selectedDiagramId: null,
    }));

    try {
      const image = await new Promise<Blob>((resolve, reject) => {
        canvasRef.current?.toBlob(
          (blob) => blob
            ? resolve(blob)
            : reject(new Error("Could not encode rendered PDF page.")),
          "image/png",
        );
      });

      if (requestId !== scanRequestIdRef.current) return;

      const form = new FormData();
      form.append("image", image, `page-${session.page}.png`);

      const response = await fetch(`${API_BASE}/detect`, {
        method: "POST",
        body: form,
        signal: abortController.signal,
      });

      if (!response.ok) {
        throw new Error(`Detector returned HTTP ${response.status}.`);
      }

      const data = await response.json() as { candidates?: CandidateInput[] };
      const candidates = (data.candidates ?? []).map(
        (item, index) => normalizedCandidate({ ...item, source: "automatic" }, index),
      );

      if (requestId !== scanRequestIdRef.current) return;

      patchSession((current) => ({
        ...current,
        candidatesByPage: {
          ...current.candidatesByPage,
          [String(current.page)]: candidates,
        },
        selectedDiagramId: null,
      }));

      setNotice(
        candidates.length
          ? `Found ${candidates.length} board candidate${candidates.length === 1 ? "" : "s"}. Select a candidate to recognize its chess position and load it onto the analysis board.`
          : "No board candidates found.",
      );
    } catch (cause) {
      if (requestId !== scanRequestIdRef.current) return;

      if (abortController.signal.aborted) {
        setNotice("Scan cancelled.");
      } else {
        setError(`Board detection failed: ${messageOf(cause)}`);
      }
    } finally {
      if (requestId === scanRequestIdRef.current) {
        scanAbortControllerRef.current = null;
        setDetecting(false);
      }
    }
  }

      patchSession({ zoom: Math.max(0.45, Math.min(1.8, scale)) });

    }
  }

function formatSize(bytes: number): string {
  if (bytes < 1024 * 1024) return `${Math.max(1, Math.round(bytes / 1024))} KB`;
  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
}

import React, { useEffect, useMemo, useState } from "react";
import {
  buildFullFen,
  countKings,
  EMPTY_PLACEMENT,
  flipFenPlacement,
  moveFenPiece,
  parseFenPlacement,
  PIECE_LABELS,
  sanitizeCastlingRights,
  setFenPiece,
  START_PLACEMENT,
  validatePositionFen,
} from "./position-editor";

const WHITE_PIECES = ["K", "Q", "R", "B", "N", "P"] as const;
const BLACK_PIECES = ["k", "q", "r", "b", "n", "p"] as const;

export interface ManualBoardEditorProps {
  initialFen: string;
  onConfirm: (fen: string) => void;
  onCancel: () => void;
  confirmLabel: string;
  title: string;
  subtitle?: string;
  // Specific to recognition review (Feature A)
  cropPreviewUrl?: string;
  recognitionStatus?: "idle" | "loading" | "complete" | "failed";
  recognitionResult?: {
    board_confidence: number;
    mean_piece_confidence: number | null;
    piece_count: number;
    orientation: "normal" | "flipped" | "unknown";
    warnings: string[];
  } | null;
  recognitionError?: string;
  onRetryRecognition?: () => void;
  // Specific to setup mode (Feature B)
  showPresets?: boolean;
  currentAnalysisFen?: string;
}

export function ManualBoardEditor({
  initialFen,
  onConfirm,
  onCancel,
  confirmLabel,
  title,
  subtitle,
  cropPreviewUrl,
  recognitionStatus,
  recognitionResult,
  recognitionError,
  onRetryRecognition,
  showPresets = false,
  currentAnalysisFen,
}: ManualBoardEditorProps) {
  // Parse initial fields
  const parsedInitial = useMemo(() => {
    const trimmed = initialFen.trim();
    const fields = trimmed ? trimmed.split(/\s+/) : [];
    const placement = fields[0] && parseFenPlacement(fields[0]) ? fields[0] : START_PLACEMENT;
    const turn = fields[1] === "b" ? ("b" as const) : ("w" as const);
    const castling = fields[2] ?? "-";
    const ep = fields[3] ?? "-";
    const halfmove = Number(fields[4]) || 0;
    const fullmove = Number(fields[5]) || 1;
    const fen = buildFullFen(placement, turn, castling, ep, halfmove, fullmove);
    return { placement, turn, castling, ep, halfmove, fullmove, fen };
  }, [initialFen]);

  const [placement, setPlacement] = useState(parsedInitial.placement);
  const [turn, setTurn] = useState<"w" | "b">(parsedInitial.turn);
  const [castling, setCastling] = useState(parsedInitial.castling);
  const [activeTool, setActiveTool] = useState<string>("P"); // Piece symbol or "" for erase
  const [selectedSquare, setSelectedSquare] = useState<{ row: number; col: number } | null>(null);
  const [flipped, setFlipped] = useState(false);
  const [history, setHistory] = useState<string[]>([parsedInitial.fen]);
  const [historyIndex, setHistoryIndex] = useState(0);
  const [fenInputText, setFenInputText] = useState(parsedInitial.fen);
  const [validationError, setValidationError] = useState<string>("");

  // When initialFen changes externally (e.g. recognition completes or retries), reset editor
  useEffect(() => {
    setPlacement(parsedInitial.placement);
    setTurn(parsedInitial.turn);
    setCastling(parsedInitial.castling);
    setHistory([parsedInitial.fen]);
    setHistoryIndex(0);
    setFenInputText(parsedInitial.fen);
    setValidationError("");
    setSelectedSquare(null);
  }, [parsedInitial]);

  const currentBoard = useMemo(() => parseFenPlacement(placement) ?? parseFenPlacement(EMPTY_PLACEMENT)!, [placement]);
  const currentFen = useMemo(() => buildFullFen(placement, turn, castling), [placement, turn, castling]);
  const validation = useMemo(() => validatePositionFen(currentFen), [currentFen]);
  const kingCounts = useMemo(() => countKings(placement), [placement]);

  // Push new state to history
  function pushState(newPlacement: string, newTurn: "w" | "b" = turn, newCastling: string = castling) {
    const board = parseFenPlacement(newPlacement);
    const sanitizedCastling = board ? sanitizeCastlingRights(board, newCastling) : "-";
    const newFen = buildFullFen(newPlacement, newTurn, sanitizedCastling);

    setPlacement(newPlacement);
    setTurn(newTurn);
    setCastling(sanitizedCastling);
    setFenInputText(newFen);
    setValidationError("");

    if (newFen !== history[historyIndex]) {
      const nextHistory = history.slice(0, historyIndex + 1);
      nextHistory.push(newFen);
      setHistory(nextHistory);
      setHistoryIndex(nextHistory.length - 1);
    }
  }

  function handleUndo() {
    if (historyIndex > 0) {
      const prevIndex = historyIndex - 1;
      const prevFen = history[prevIndex];
      const fields = prevFen.split(/\s+/);
      setPlacement(fields[0]);
      setTurn(fields[1] === "b" ? "b" : "w");
      setCastling(fields[2] ?? "-");
      setFenInputText(prevFen);
      setHistoryIndex(prevIndex);
      setValidationError("");
    }
  }

  function handleRedo() {
    if (historyIndex < history.length - 1) {
      const nextIndex = historyIndex + 1;
      const nextFen = history[nextIndex];
      const fields = nextFen.split(/\s+/);
      setPlacement(fields[0]);
      setTurn(fields[1] === "b" ? "b" : "w");
      setCastling(fields[2] ?? "-");
      setFenInputText(nextFen);
      setHistoryIndex(nextIndex);
      setValidationError("");
    }
  }

  function handleSquareClick(row: number, col: number) {
    setSelectedSquare({ row, col });
    if (activeTool === "") {
      // Erase tool active
      const nextPlacement = setFenPiece(placement, row, col, "");
      pushState(nextPlacement);
    } else {
      // Place / Replace with active piece
      const nextPlacement = setFenPiece(placement, row, col, activeTool);
      pushState(nextPlacement);
    }
  }

  function handleToolSelect(tool: string) {
    setActiveTool(tool);
    // If a square is already selected and user picks a piece or erase, apply to that square
    if (selectedSquare) {
      const nextPlacement = setFenPiece(placement, selectedSquare.row, selectedSquare.col, tool);
      pushState(nextPlacement);
    }
  }

  function handleClearBoard() {
    pushState(EMPTY_PLACEMENT, turn, "-");
    setSelectedSquare(null);
  }

  function handleStartingPosition() {
    pushState(START_PLACEMENT, "w", "KQkq");
    setSelectedSquare(null);
  }

  function handleFlipBoard() {
    setFlipped((f) => !f);
  }

  function handleInvertPieces() {
    const nextPlacement = flipFenPlacement(placement);
    pushState(nextPlacement);
  }

  function handleTurnChange(newTurn: "w" | "b") {
    pushState(placement, newTurn, castling);
  }

  function handleFenInputChange(value: string) {
    setFenInputText(value);
    const trimmed = value.trim();
    const fields = trimmed.split(/\s+/);
    if (fields.length === 6) {
      const parsed = parseFenPlacement(fields[0]);
      if (parsed) {
        setPlacement(fields[0]);
        setTurn(fields[1] === "b" ? "b" : "w");
        setCastling(fields[2] ?? "-");
        setValidationError("");
        if (trimmed !== history[historyIndex]) {
          const nextHistory = history.slice(0, historyIndex + 1);
          nextHistory.push(trimmed);
          setHistory(nextHistory);
          setHistoryIndex(nextHistory.length - 1);
        }
        return;
      }
    }
    const val = validatePositionFen(trimmed);
    if (!val.isValid) {
      setValidationError(val.error || "Invalid FEN syntax.");
    }
  }

  // Keyboard navigation for selected square
  function handleKeyDown(event: React.KeyboardEvent) {
    if (!selectedSquare) return;
    const { row, col } = selectedSquare;
    if (event.key === "Delete" || event.key === "Backspace") {
      event.preventDefault();
      const nextPlacement = setFenPiece(placement, row, col, "");
      pushState(nextPlacement);
      return;
    }
    const key = event.key;
    if (/^[prnbqkPRNBQK]$/.test(key)) {
      event.preventDefault();
      const nextPlacement = setFenPiece(placement, row, col, key);
      pushState(nextPlacement);
      setActiveTool(key);
      return;
    }
    if (event.key === "Escape") {
      setSelectedSquare(null);
    }
  }

  function handleConfirmClick() {
    if (validationError) {
      return;
    }
    if (!validation.isValid) {
      setValidationError(validation.error || "Cannot confirm: position is not valid.");
      return;
    }
    onConfirm(currentFen);
  }

  const pieceAssetUrl = (piece: string) => {
    const color = piece === piece.toUpperCase() ? "w" : "b";
    const type = piece.toLowerCase();
    return `${import.meta.env.BASE_URL}assets/pieces/standalone/${color}${type}.svg`;
  };

  return (
    <div className="modal-backdrop" role="presentation" onKeyDown={handleKeyDown}>
      <section
        className="transfer-dialog manual-editor-modal"
        role="dialog"
        aria-modal="true"
        aria-label={title}
      >
        <header>
          <div>
            <h2>{title}</h2>
            {subtitle && <p>{subtitle}</p>}
          </div>
          <button
            type="button"
            aria-label="Close dialog"
            title="Cancel and close"
            onClick={onCancel}
          >
            ×
          </button>
        </header>

        {showPresets && (
          <div className="editor-presets-bar">
            <span>START FROM:</span>
            {currentAnalysisFen && (
              <button
                type="button"
                className="preset-btn"
                onClick={() => {
                  const fields = currentAnalysisFen.trim().split(/\s+/);
                  if (fields.length === 6 && parseFenPlacement(fields[0])) {
                    pushState(fields[0], fields[1] === "b" ? "b" : "w", fields[2]);
                  }
                }}
              >
                Edit Current Board
              </button>
            )}
            <button type="button" className="preset-btn" onClick={handleClearBoard}>
              Empty Board
            </button>
            <button type="button" className="preset-btn" onClick={handleStartingPosition}>
              Standard Starting Board
            </button>
          </div>
        )}

        {cropPreviewUrl && (
          <div className="recognition-layout">
            <img className="crop-preview" src={cropPreviewUrl} alt="Selected chess diagram crop" />
            <div className="recognition-summary" role="status">
              {recognitionStatus === "loading" && (
                <>
                  <span className="spinner" /> Recognizing pieces…
                </>
              )}
              {recognitionStatus === "complete" && (
                <>
                  <strong>Model-recognized position · Editable below</strong>
                  <span>
                    {recognitionResult?.piece_count ?? 0} pieces recognized · confidence{" "}
                    {recognitionResult?.board_confidence?.toFixed(2) ?? "—"}
                  </span>
                  <span>
                    Orientation: {recognitionResult?.orientation ?? "normal"}. Use "Flip position 180°" if needed.
                  </span>
                  {onRetryRecognition && (
                    <button
                      type="button"
                      className="mini-button retry-btn"
                      onClick={onRetryRecognition}
                    >
                      ↻ Retry recognition
                    </button>
                  )}
                </>
              )}
              {recognitionStatus === "failed" && (
                <>
                  <strong>Automatic recognition did not complete.</strong>
                  <span>{recognitionError || "Check the diagram crop and set up the pieces manually."}</span>
                  {onRetryRecognition && (
                    <button
                      type="button"
                      className="mini-button retry-btn"
                      onClick={onRetryRecognition}
                    >
                      ↻ Retry recognition
                    </button>
                  )}
                </>
              )}
            </div>
          </div>
        )}

        <div className="manual-editor-workspace">
          {/* Left: The Board */}
          <div className="manual-board-container">
            <div
              className={`manual-chessboard ${flipped ? "flipped" : ""}`}
              role="grid"
              aria-label="Interactive chess position editor"
            >
              {Array.from({ length: 64 }, (_, index) => {
                const visibleRow = Math.floor(index / 8);
                const visibleCol = index % 8;
                const boardRow = flipped ? 7 - visibleRow : visibleRow;

import { Chess } from "chess.js";

export function makePosition(baseFen: string, moves: string[], moveIndex: number): Chess {
  const game = new Chess(baseFen);
  for (const san of moves.slice(0, moveIndex)) {
    const result = game.move(san);
    if (!result) throw new Error(`Saved move could not be replayed: ${san}`);
  }
  return game;
}

export function fenWithTurn(fen: string, turn: "w" | "b"): string {
  const fields = fen.split(/\s+/);
  if (fields.length !== 6) throw new Error("FEN must contain six fields.");
  fields[1] = turn;
  fields[3] = "-";
  fields[4] = "0";
  return fields.join(" ");
}

import type { Square } from "chess.js";

export type BoardArrow = {
  from: Square;
  to: Square;
  color?: string;
};

export type SquareHighlight = {
  square: Square;
  color?: string;
};

export const ANNOTATION_COLORS = {
  red: "#EF5350",
  yellow: "#FFB52E",
  blue: "#29B6E6",
  green: "#8BC34A",
  engine: "#f0a23b",
} as const;

export function getAnnotationColor(event: {
  shiftKey?: boolean;
  ctrlKey?: boolean;
  altKey?: boolean;
  metaKey?: boolean;
}): string {
  if (event.shiftKey) return ANNOTATION_COLORS.green;
  if (event.ctrlKey || event.metaKey) return ANNOTATION_COLORS.yellow;
  if (event.altKey) return ANNOTATION_COLORS.blue;
  return ANNOTATION_COLORS.red;
}

export function boardSquareAtPoint(
  clientX: number,
  clientY: number,
  rect: Pick<DOMRect, "left" | "top" | "width" | "height">,
  flipped = false,
): Square | null {
  if (rect.width <= 0 || rect.height <= 0) return null;
  const x = clientX - rect.left;
  const y = clientY - rect.top;
  if (x < 0 || y < 0 || x >= rect.width || y >= rect.height) return null;
  const column = Math.floor((x / rect.width) * 8);
  const row = Math.floor((y / rect.height) * 8);
  const file = flipped ? 7 - column : column;
  const rank = flipped ? row + 1 : 8 - row;
  return ("abcdefgh"[file] + rank) as Square;
}

export function squareCenter(square: Square, flipped = false): { x: number; y: number } {
  const file = square.charCodeAt(0) - 97;
  const rank = Number(square[1]) - 1;
  return {
    x: (flipped ? 7 - file : file) + 0.5,
    y: (flipped ? rank : 7 - rank) + 0.5,
  };
}

export function parseBestMove(move: string | undefined): BoardArrow | null {
  const match = move?.match(/^([a-h][1-8])([a-h][1-8])[qrbn]?$/);
  return match ? { from: match[1] as Square, to: match[2] as Square, color: ANNOTATION_COLORS.engine } : null;
}

export function toggleArrow(arrows: BoardArrow[], next: BoardArrow): BoardArrow[] {
  const existing = arrows.findIndex((arrow) => arrow.from === next.from && arrow.to === next.to);
  if (existing < 0) return [...arrows, next];
  if (next.color && arrows[existing].color !== next.color) {
    return arrows.map((arrow, index) => (index === existing ? next : arrow));
  }
  return arrows.filter((_, index) => index !== existing);
}

export function toggleSquareHighlight(
  highlights: SquareHighlight[],
  square: Square,
  color: string = ANNOTATION_COLORS.red,
): SquareHighlight[] {
  const existing = highlights.findIndex((h) => h.square === square);
  if (existing < 0) return [...highlights, { square, color }];
  if (highlights[existing].color !== color) {
    return highlights.map((h, index) => (index === existing ? { square, color } : h));
  }
  return highlights.filter((_, index) => index !== existing);
}

export function getArrowPath(
  fromSquare: Square,
  toSquare: Square,
  flipped = false,
  shaftWidth = 0.22,
  headLength = 0.36,
  headWidth = 0.50,
): string {
  const from = squareCenter(fromSquare, flipped);
  const to = squareCenter(toSquare, flipped);
  const dx = to.x - from.x;
  const dy = to.y - from.y;
  const len = Math.hypot(dx, dy);
  if (len < 0.2) return "";

  const ux = dx / len;
  const uy = dy / len;
  const nx = -uy;
  const ny = ux;

  const actualHeadLen = len < 0.7 ? len * 0.48 : headLength;
  const actualHeadWidth = len < 0.7 ? len * 0.65 : headWidth;
  const actualShaftWidth = len < 0.7 ? len * 0.28 : shaftWidth;

  const hs = actualShaftWidth / 2;
  const hw = actualHeadWidth / 2;

  const bx = to.x - actualHeadLen * ux;
  const by = to.y - actualHeadLen * uy;

  const tipX = to.x;
  const tipY = to.y;

  const headLeftX = bx + hw * nx;
  const headLeftY = by + hw * ny;
  const headRightX = bx - hw * nx;
  const headRightY = by - hw * ny;

  const shaftEndLeftX = bx + hs * nx;
  const shaftEndLeftY = by + hs * ny;
  const shaftEndRightX = bx - hs * nx;
  const shaftEndRightY = by - hs * ny;

  const shaftStartLeftX = from.x + hs * nx;
  const shaftStartLeftY = from.y + hs * ny;
  const shaftStartRightX = from.x - hs * nx;
  const shaftStartRightY = from.y - hs * ny;

  return [
    "M " + shaftStartLeftX.toFixed(4) + " " + shaftStartLeftY.toFixed(4),
    "L " + shaftEndLeftX.toFixed(4) + " " + shaftEndLeftY.toFixed(4),
    "L " + headLeftX.toFixed(4) + " " + headLeftY.toFixed(4),
    "L " + tipX.toFixed(4) + " " + tipY.toFixed(4),
    "L " + headRightX.toFixed(4) + " " + headRightY.toFixed(4),
    "L " + shaftEndRightX.toFixed(4) + " " + shaftEndRightY.toFixed(4),
    "L " + shaftStartRightX.toFixed(4) + " " + shaftStartRightY.toFixed(4),
    "A " + hs.toFixed(4) + " " + hs.toFixed(4) + " 0 0 0 " + shaftStartLeftX.toFixed(4) + " " + shaftStartLeftY.toFixed(4),
    "Z",
  ].join(" ");
}
