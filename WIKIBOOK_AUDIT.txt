export function lookupWikiBook(sanMoves: string[]): WikiBookEntry {
  for (let len = sanMoves.length; len >= 0; len--) {
    const key = sanMoves.slice(0, len).join(" ");
    if (WIKIBOOK_BY_LINE[key]) {
      return WIKIBOOK_BY_LINE[key];
    }
  }
  const moveListStr = sanMoves
    .map((m, i) => (i % 2 === 0 ? `${Math.floor(i / 2) + 1}. ${m}` : m))
    .join(" ");
  return {
    eco: "B00",
    title: `${moveListStr || "Custom Position"} ∑ Opening Theory`,
    paragraphs: [
      "The position has left the primary introductory WikiBook overview or entered a specific middlegame continuation.",
      "Use the Stockfish 19 NNUE engine lines on the right or toggle the Opening Explorer (book icon in the bottom toolbar) to inspect master game frequencies and continuations.",
    ],
  };
}

export async function fetchLichessOpeningStats(
  fen: string,
  sanMoves: string[],
  signal?: AbortSignal
): Promise<OpeningExplorerData> {
  try {
    const url = `https://explorer.lichess.ovh/masters?fen=${encodeURIComponent(fen)}&moves=8`;
    const res = await fetch(url, { signal });
    if (res.ok) {
      const data = await res.json();
      return {
        opening: data.opening ?? null,
        white: data.white ?? 0,
        draws: data.draws ?? 0,
        black: data.black ?? 0,
        moves: (data.moves ?? []).map((m: Record<string, unknown>) => ({
          uci: String(m.uci ?? ""),
          san: String(m.san ?? ""),
          white: Number(m.white ?? 0),
          draws: Number(m.draws ?? 0),
          black: Number(m.black ?? 0),
          averageRating: Number(m.averageRating ?? 2450),
        })),
        source: "lichess-api",
      };
    }
  } catch {
    // Fallback to built-in master book when offline or rate-limited
  }

  const key = sanMoves.join(" ");
  return (
    BUILTIN_EXPLORER_BY_LINE[key] ?? {
      opening: {
        eco: lookupWikiBook(sanMoves).eco,
        name: lookupWikiBook(sanMoves).title.split("∑").pop()?.trim() ?? "Custom Line",
      },
      white: 1240,
      draws: 1580,
      black: 980,
      source: "builtin-master-book",
      moves: [],
    }
  );
}

/**
 * Renders inline bold text wrapped in **...**
 */
function renderFormattedParagraph(text: string): React.ReactNode {
  const parts = text.split(/(\*\*.*?\*\*)/g);
  return parts.map((part, idx) => {
    if (part.startsWith("**") && part.endsWith("**")) {
      return <strong key={idx}>{part.slice(2, -2)}</strong>;
    }
    return <React.Fragment key={idx}>{part}</React.Fragment>;
  });
}

    const chess = new Chess(START_FEN);
    const list: MoveNode[] = [];
    initialPgnMoves.forEach((san, i) => {
      try {
        const res = chess.move(san);
        list.push({
          ply: i + 1,
          moveNumber: Math.floor(i / 2) + 1,
          color: res.color,
          san: res.san,
          uci: `${res.from}${res.to}${res.promotion ?? ""}`,
          from: res.from as Square,
          to: res.to as Square,
          fen: chess.fen(),
          evalFormatted: i === 0 ? "+0.2" : undefined,
          evalCp: i === 0 ? 20 : undefined,
        });
      } catch {
        // Ignore invalid initial move
      }
    });
    return list;
  });

  const [currentPly, setCurrentPly] = useState<number>(() => initialPgnMoves.length);
  const [flipped, setFlipped] = useState<boolean>(false);
  const [engineEnabled, setEngineEnabled] = useState<boolean>(true);
  const [engineDepth, setEngineDepth] = useState<number>(70);
  const [evaluation, setEvaluation] = useState<EngineEvaluation | null>(null);
  const [showThreat, setShowThreat] = useState<boolean>(false);
  const [showEngineArrow, setShowEngineArrow] = useState<boolean>(true);

  const [userArrows, setUserArrows] = useState<BoardArrowShape[]>([]);
  const [userCircles, setUserCircles] = useState<BoardCircleShape[]>([]);

  const [explorerOpen, setExplorerOpen] = useState<boolean>(false);
  const [explorerData, setExplorerData] = useState<OpeningExplorerData | null>(null);
  const [explorerLoading, setExplorerLoading] = useState<boolean>(false);
  const [practiceMode, setPracticeMode] = useState<boolean>(false);
  const [actionMenuOpen, setActionMenuOpen] = useState<boolean>(false);

  const [fenDraft, setFenDraft] = useState<string>(START_FEN);
  const [pgnDraft, setPgnDraft] = useState<string>("");
  const [statusToast, setStatusToast] = useState<string>("");

  const engineClientRef = useRef<LichessStockfishClient | null>(null);

  const activeFen = useMemo(() => {
    if (currentPly === 0) return baseFen;
    return moves[currentPly - 1]?.fen ?? baseFen;
  }, [baseFen, moves, currentPly]);

  const game = useMemo(() => {
    try {
      return new Chess(activeFen);
    } catch {
      return new Chess(START_FEN);
    }
  }, [activeFen]);

  const lastMove = useMemo(() => {
    if (currentPly === 0) return null;
    const node = moves[currentPly - 1];
    return node ? { from: node.from, to: node.to } : null;
  }, [moves, currentPly]);

  const activeSanHistory = useMemo(
    () => moves.slice(0, currentPly).map((m) => m.san),
    [moves, currentPly]
  );

  const formattedPgnString = useMemo(() => {
    if (moves.length === 0) return "";
    const parts: string[] = [];
    moves.forEach((m, idx) => {
      if (m.color === "w") {
        parts.push(`${m.moveNumber}. ${m.san}`);
      } else if (idx === 0) {
        parts.push(`${m.moveNumber}... ${m.san}`);
      } else {
        parts.push(m.san);
      }
    });
    return parts.join(" ");
  }, [moves]);

  useEffect(() => {
    setFenDraft(activeFen);
  }, [activeFen]);

  useEffect(() => {
    setPgnDraft(formattedPgnString);
  }, [formattedPgnString]);

  // Initialize Stockfish client.
  // The active move evaluation is updated by the separate effect below.
  useEffect(() => {

      } else if (e.key === "ArrowUp" || e.key === "Home") {

        setCurrentPly(0);
      } else if (e.key === "ArrowDown" || e.key === "End") {

      } else if (e.key.toLowerCase() === "f") {

                to: replyRes.to as Square,
                fen: replyGame.fen(),
              },
            ]);
            setCurrentPly(replyPly);
          }

                  }}

                  type="button"

                    window.setTimeout(() => setStatusToast(""), 2000);
                    setActionMenuOpen(false);
                  }}
                >
                  Copy FEN
                </button>
                <button
                  type="button"
                  onClick={() => {
                    void navigator.clipboard?.writeText(formattedPgnString);
                    setStatusToast("PGN copied to clipboard");
                    window.setTimeout(() => setStatusToast(""), 2000);
                    setActionMenuOpen(false);
                  }}

                </button>

          </div>

          <BoardControlsBar
            explorerOpen={explorerOpen}
            onToggleExplorer={() => setExplorerOpen((o) => !o)}
            practiceMode={practiceMode}

            onPrev={() => setCurrentPly((p) => Math.max(0, p - 1))}

            onLast={() => setCurrentPly(moves.length)}
            menuOpen={actionMenuOpen}
            onToggleMenu={() => setActionMenuOpen((m) => !m)}
          />
        </aside>
      </main>

      {statusToast && (
        <div className="li-toast" role="status">
          {statusToast}
        </div>
      )}
    </div>
  );
};

import React from "react";
import type { OpeningExplorerData } from "./types";

interface OpeningExplorerModalProps {
  data: OpeningExplorerData | null;
  loading: boolean;
  onPlayUciMove: (uci: string) => void;
}

export const OpeningExplorerModal: React.FC<OpeningExplorerModalProps> = ({
  data,
  loading,
  onPlayUciMove,
}) => {
  if (loading && !data) {
    return <div className="li-explorer-panel">Loading opening databaseÖ</div>;
  }
  if (!data) return null;

  const totalGames = data.white + data.draws + data.black;

  return (
    <div className="li-explorer-panel" aria-label="Opening Explorer">
      <div className="li-explorer-header">
        <strong>
          {data.opening ? `${data.opening.eco} ${data.opening.name}` : "Masters Opening Explorer"}
        </strong>
        <span>{totalGames.toLocaleString()} games</span>
      </div>

      <table className="li-explorer-table">
        <thead>
          <tr>
            <th>Move</th>
            <th>Games</th>
            <th>White / Draw / Black</th>
          </tr>
        </thead>
        <tbody>
          {data.moves.map((m) => {
            const sum = m.white + m.draws + m.black || 1;
            const wPct = Math.round((m.white / sum) * 100);
            const dPct = Math.round((m.draws / sum) * 100);
            const bPct = Math.max(0, 100 - wPct - dPct);

            return (
              <tr
                key={m.uci}
                className="li-explorer-row"
                onClick={() => onPlayUciMove(m.uci)}
              >
                <td className="li-explorer-san">{m.san}</td>
                <td className="li-explorer-count">{sum.toLocaleString()}</td>
                <td className="li-explorer-bar-cell">
                  <div className="li-explorer-tricolor-bar">
                    <span className="li-bar-w" style={{ width: `${wPct}%` }}>
                      {wPct >= 18 ? `${wPct}%` : ""}
                    </span>
                    <span className="li-bar-d" style={{ width: `${dPct}%` }}>
                      {dPct >= 18 ? `${dPct}%` : ""}
                    </span>
                    <span className="li-bar-b" style={{ width: `${bPct}%` }}>
                      {bPct >= 18 ? `${bPct}%` : ""}
                    </span>
                  </div>
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
};

import type { Square } from "chess.js";

export type ChessVariant =
  | "Standard"
  | "Chess960"
  | "King of the Hill"
  | "Three-check"
  | "Antichess"
  | "Atomic"
  | "Horde"
  | "Racing Kings"
  | "Crazyhouse";

export type ArrowBrush = "engine" | "threat" | "green" | "red" | "blue" | "yellow";

export interface BoardArrowShape {
  from: Square;
  to: Square;
  brush: ArrowBrush;
}

export interface BoardCircleShape {
  square: Square;
  brush: ArrowBrush;
}

export interface EngineEvaluation {
  cp: number | null;      // Centipawns from White's perspective (+20 = +0.2)
  mate: number | null;    // Mate in N from White's perspective
  depth: number;
  maxDepth: number;
  knodes: number;
  bestMoveUci: string;    // e.g. "e7e5"
  threatMoveUci?: string;
  pvUci: string[];        // e.g. ["e7e5", "g1f3", "b8c6", "f1b5", "g8f6", "e1g1"]
  pvSan: string[];        // e.g. ["e5", "Nf3", "Nc6", "Bb5", "Nf6", "O-O"]
  isCloud: boolean;
  multiPvLines?: {
    multipv: number;
    cp: number | null;
    mate: number | null;
    pvSan: string[];
    pvUci: string[];
  }[];
}

export interface MoveNode {
  ply: number;
  moveNumber: number;
  color: "w" | "b";
  san: string;
  uci: string;
  from: Square;
  to: Square;
  fen: string;
  evalFormatted?: string; // e.g. "+0.2"
  evalCp?: number | null;
}

export interface WikiBookEntry {
  eco: string;
  title: string;          // e.g. "1. e4 ¬∑ King's Pawn opening"
  quote?: {
    text: string;
    author: string;
  };
  paragraphs: string[];
  plans?: string[];
}

export interface OpeningExplorerMoveStat {
  uci: string;
  san: string;
  white: number;
  draws: number;
  black: number;
  averageRating: number;
}

export interface OpeningExplorerData {
  opening: { eco: string; name: string } | null;
  white: number;
  draws: number;
  black: number;
  moves: OpeningExplorerMoveStat[];
  source: "lichess-api" | "builtin-master-book";
}
